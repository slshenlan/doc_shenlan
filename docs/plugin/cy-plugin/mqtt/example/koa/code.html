ts
import CyMqtt, {
  enumMqttEvent,
  type IMqttEventMap,
  type IMqttPublishOptions,
} from "cy-plugin-mqtt";

import type { PendingAction, ScopedTopic } from "./types";

import { MAX_PENDING, MQTT_SCOPE_SYMBOL, mqttClientMsg } from "./config";

/** 生成作用域内订阅登记 key，保证 subscribe / unsubscribe / dispose 对同一主题 */
function topicKey(id: string, st?: string): string {
  return `${st ?? ""}::${id}`;
}

/**
 * 模块 / 任务级 MQTT 作用域（Koa / Node）。
 *
 * 职责：
 * - 将 publish / subscribe / unsubscribe 绑定到唯一 `scopeId`
 * - 登记本 scope 发起过的订阅，便于结束时统一退订
 * - `dispose()`：取消未执行队列 + unsubscribe 已登记主题
 *
 * 普通 HTTP 请求一般用全局 `mqtt.publish` 即可；
 * 仅在「某业务模块临时订阅、结束后要退订」时使用 Scope。
 *
 * @example
 * ```ts
 * const scope = mqtt.createScope();
 * scope.subscribe('sensor/001');
 * // 模块卸载 / 任务结束
 * scope.dispose();
 * ```
 */
export class MqttScope {
  private disposed = false;

  /**
   * 本作用域已发起的订阅（含仍在队列中、尚未真正 subscribe 成功的）。
   * dispose 时按此表执行 unsubscribe。
   */
  private readonly topics = new Map<string, ScopedTopic>();

  constructor(
    private readonly client: MqttClient,
    /** 本作用域唯一标识，用于标记 / 取消队列任务 */
    readonly scopeId: symbol,
  ) {}

  /** 是否已 dispose；dispose 后所有操作直接忽略 */
  get isDisposed(): boolean {
    return this.disposed;
  }

  /**
   * 发布消息（自动带上本 scopeId）。
   * 已连接立即发；未连接入队，CONNECT 后冲刷；dispose 后调用无效。
   */
  publish(msg: unknown, pt?: string, options?: IMqttPublishOptions): void {
    if (this.disposed) {
      return;
    }
    this.client.publish(msg, pt, options, this.scopeId);
  }

  /**
   * 订阅主题，并登记到本 scope，供 dispose 时自动退订。
   * 最终 topic = `(st ?? subscribeTopic) + id`
   */
  subscribe(id: string, st?: string): void {
    if (this.disposed) {
      return;
    }
    this.topics.set(topicKey(id, st), { id, st });
    this.client.subscribe(id, st, this.scopeId);
  }

  /**
   * 取消订阅，并从本 scope 登记表移除。
   * `id` / `st` 须与对应 `subscribe` 一致。
   */
  unsubscribe(id: string, st?: string): void {
    if (this.disposed) {
      return;
    }
    this.topics.delete(topicKey(id, st));
    this.client.unsubscribe(id, st, this.scopeId);
  }

  /**
   * 销毁本作用域：
   * 1. `cancelScope`：取消仍在队列 / 正在冲刷中的本 scope 任务
   * 2. 对本 scope 登记过的主题执行 `unsubscribe`（走无 scope 通道，避免刚被 cancel 掉）
   *
   * 注意：不会卸载 `mqtt.on` 的事件监听，模块内 `on` 须自行 `off`。
   */
  dispose(): void {
    if (this.disposed) {
      return;
    }
    this.disposed = true;
    this.client.cancelScope(this.scopeId);

    for (const { id, st } of this.topics.values()) {
      // 退订走无 scope 通道，避免刚被 cancelScope 清掉
      this.client.unsubscribe(id, st);
    }
    this.topics.clear();
  }
}

/**
 * Koa / Node 业务侧 MQTT 客户端封装（进程级单例由 `index.ts` 创建）。
 *
 * 能力：
 * - 未连接时 publish / subscribe / unsubscribe 自动入队，CONNECT（含重连）后冲刷
 * - `createScope()`：模块 / 任务级作用域，避免销毁后仍被冲刷
 * - 事件 API 透传底层 CyMqtt（基于 emittery；监听须自行 off）
 *
 * @example
 * ```ts
 * mqtt.connect();
 * mqtt.on(enumMqttEvent.MESSAGE, ({ topic, data }) => {
 *   logger.mqttClient.debug(topic, data);
 * });
 * ```
 */
export class MqttClient {
  /** 未连接时的待发队列 */
  private pending: PendingAction[] = [];

  /**
   * 当前正在冲刷的批次。
   * dispose / cancelScope 时可同步打 cancelled，避免「已 splice 出队列但仍被执行」。
   */
  private flushing: PendingAction[] | null = null;

  /** 队列任务自增 id */
  private pendingSeq = 0;

  constructor(private readonly raw: CyMqtt) {
    // 每次连接成功（含重连）冲刷待发队列
    this.raw.on(enumMqttEvent.CONNECT, () => {
      this.flushPending();
    });
  }

  /** 当前是否已与 broker 建立连接 */
  get isConnected(): boolean {
    return this.raw.isConnected;
  }

  /** 底层 CyMqtt 实例（一般业务无需直接使用） */
  get rawClient(): CyMqtt {
    return this.raw;
  }

  /**
   * 创建模块 / 任务作用域。
   * 结束时务必 `scope.dispose()`。
   */
  createScope(): MqttScope {
    return new MqttScope(this, Symbol(MQTT_SCOPE_SYMBOL));
  }

  /** 连接到 MQTT 服务器 */
  connect(): void {
    this.raw.connect();
  }

  /**
   * 主动断开连接，并清空全部待发队列（含正在冲刷批次标记取消）。
   * 可再次 `connect`；底层已成功订阅的记录仍由 CyMqtt 保留，重连后可恢复。
   */
  disconnect(): void {
    this.clearPending();
    this.raw.disconnect();
  }

  /**
   * 销毁底层实例：清空队列、关闭连接，之后不可再 `connect`。
   * 进程退出前由 `stopMqttClient`（`registerGracefulShutdown` 任务）统一调用。
   */
  destroy(): void {
    this.clearPending();
    this.raw.destroy();
  }

  /**
   * 任意逻辑：已连接立即执行，否则等下一次 CONNECT。
   * 不走 pending 队列，也无 scope 绑定。
   */
  whenConnected(fn: () => void): void {
    if (this.raw.isConnected) {
      fn();
      return;
    }
    this.raw.once(enumMqttEvent.CONNECT, fn);
  }

  /**
   * 发布消息。
   * 已连接立即发；未连接入队。`scopeId` 由 MqttScope 传入，业务一般不必手传。
   */
  publish(
    msg: unknown,
    pt?: string,
    options?: IMqttPublishOptions,
    scopeId?: symbol,
  ): void {
    if (this.raw.isConnected) {
      this.raw.publish(msg, pt, options);
      return;
    }
    this.enqueue({
      id: ++this.pendingSeq,
      kind: "publish",
      scopeId,
      cancelled: false,
      msg,
      pt,
      options,
    });
  }

  /**
   * 订阅主题。最终 topic = `(st ?? subscribeTopic) + id`。
   * 已连接立即订；未连接入队。临时订阅推荐走 `MqttScope.subscribe`。
   */
  subscribe(id: string, st?: string, scopeId?: symbol): void {
    if (this.raw.isConnected) {
      this.raw.subscribe(id, st);
      return;
    }
    this.enqueue({
      id: ++this.pendingSeq,
      kind: "subscribe",
      scopeId,
      cancelled: false,
      topicId: id,
      st,
    });
  }

  /**
   * 取消订阅。`id` / `st` 须与对应 `subscribe` 一致。
   * 已连接立即退订；未连接入队。
   */
  unsubscribe(id: string, st?: string, scopeId?: symbol): void {
    if (this.raw.isConnected) {
      this.raw.unsubscribe(id, st);
      return;
    }
    this.enqueue({
      id: ++this.pendingSeq,
      kind: "unsubscribe",
      scopeId,
      cancelled: false,
      topicId: id,
      st,
    });
  }

  /**
   * 取消某作用域下尚未执行的队列任务（含正在冲刷批次中尚未执行到的项）。
   * 由 `MqttScope.dispose` 调用。
   */
  cancelScope(scopeId: symbol): void {
    for (const action of this.pending) {
      if (action.scopeId === scopeId) {
        action.cancelled = true;
      }
    }
    if (this.flushing) {
      for (const action of this.flushing) {
        if (action.scopeId === scopeId) {
          action.cancelled = true;
        }
      }
    }
    this.pending = this.pending.filter((item) => {
      return !item.cancelled;
    });
  }

  /**
   * 监听事件（挂在进程级单例上，基于 emittery）。
   * 模块卸载时务必 `off` 同一函数引用，否则会泄漏 / 重复回调。
   */
  on<Key extends keyof IMqttEventMap>(
    event: Key,
    listener: (data: IMqttEventMap[Key]) => void | Promise<void>,
  ): void {
    this.raw.on(event, listener);
  }

  /** 移除事件监听（须与 `on` 传入同一 `listener` 引用） */
  off<Key extends keyof IMqttEventMap>(
    event: Key,
    listener: (data: IMqttEventMap[Key]) => void | Promise<void>,
  ): void {
    this.raw.off(event, listener);
  }

  /** 只触发一次的事件监听 */
  once<Key extends keyof IMqttEventMap>(
    event: Key,
    listener: (data: IMqttEventMap[Key]) => void | Promise<void>,
  ): void {
    this.raw.once(event, listener);
  }

  /** 清空全部事件监听（慎用，会影响其他业务挂载的监听） */
  offAll(): void {
    this.raw.offAll();
  }

  /** 写入待发队列；超出上限丢弃最早一条 */
  private enqueue(action: PendingAction): void {
    if (this.pending.length >= MAX_PENDING) {
      this.pending.shift();
      logger.mqttClient.warn(
        mqttClientMsg(`待发送队列已满(${MAX_PENDING})，丢弃最早一条`),
      );
    }
    this.pending.push(action);
  }

  /**
   * 清空待发队列，并将 pending / flushing 中任务标记为 cancelled，
   * 防止冲刷过程中仍被执行。
   */
  private clearPending(): void {
    for (const action of this.pending) {
      action.cancelled = true;
    }
    this.pending = [];
    if (this.flushing) {
      for (const action of this.flushing) {
        action.cancelled = true;
      }
    }
  }

  /**
   * 冲刷待发队列：按入队顺序执行未 cancelled 的任务。
   * 在 CONNECT 时触发；冲刷期间 dispose 可通过 flushing 批次取消剩余项。
   */
  private flushPending(): void {
    if (this.pending.length === 0) {
      return;
    }

    this.flushing = this.pending.splice(0);
    for (const action of this.flushing) {
      if (action.cancelled) {
        continue;
      }
      switch (action.kind) {
        case "publish":
          this.raw.publish(action.msg, action.pt, action.options);
          break;
        case "subscribe":
          this.raw.subscribe(action.topicId!, action.st);
          break;
        case "unsubscribe":
          this.raw.unsubscribe(action.topicId!, action.st);
          break;
      }
    }
    this.flushing = null;
  }
}

import type { ICyMqtt } from "cy-plugin-mqtt";

import { coreConfigEnv } from "../../../core/config/env";

/** 待发队列上限，超出时丢弃最早一条并告警 */
export const MAX_PENDING = 100;

/** MqttScope 的 Symbol description，便于调试区分 */
export const MQTT_SCOPE_SYMBOL = "mqtt-scope";

/** 模块展示名（日志 / 错误文案前缀） */
export const MQTT_CLIENT_MODULE_NAME = "MQTT Client";

/** 默认订阅 / 发布主题前缀 */
export const DEFAULT_TOPIC_PREFIX = "device/";

/** 重连：初始间隔（ms） */
export const RECONNECT_INTERVAL = 5_000;

/** 重连：最大间隔（ms） */
export const RECONNECT_MAX_INTERVAL = 60_000;

/** 重连：最大次数（Infinity 表示不限制） */
export const RECONNECT_MAX_COUNT = Infinity;

/**
 * 拼接 MQTT Client 模块展示名文案
 *
 * @param text - 文案后缀
 * @returns 如 `MQTT Client 连接失败`
 */
export function mqttClientMsg(text: string): string {
  return `${MQTT_CLIENT_MODULE_NAME} ${text}`;
}

/**
 * 从 `coreConfigEnv` 组装连接配置（仅供创建全局单例）。
 * `url` / `protocol` 由 `cy-plugin-mqtt` 解析，此处原样透传。
 *
 * @returns 满足 `ICyMqtt` 的连接选项
 */
export function createMqttOptions(): ICyMqtt {
  const env = coreConfigEnv;
  const debug =
    env.ENV_MQTT_CLIENT_DEBUG === "true" || env.ENV_MQTT_CLIENT_DEBUG === "1";
  const protocol = env.ENV_MQTT_CLIENT_PROTOCOL as
    | ICyMqtt["protocol"]
    | undefined;

  return {
    url: env.ENV_MQTT_CLIENT_URL,
    username: env.ENV_MQTT_CLIENT_USERNAME,
    password: env.ENV_MQTT_CLIENT_PASSWORD,
    debug,
    subscribeTopic: DEFAULT_TOPIC_PREFIX,
    publishTopic: DEFAULT_TOPIC_PREFIX,
    ...(protocol ? { protocol } : {}),
    connectOption: {
      clientId: env.ENV_MQTT_CLIENT_ID,
      clean: true,
      maxCount: RECONNECT_MAX_COUNT,
      interval: RECONNECT_INTERVAL,
      maxInterval: RECONNECT_MAX_INTERVAL,
    },
  };
}

import type { IMqttPublishOptions } from "cy-plugin-mqtt";

/** 待发队列中的任务类型 */
export type PendingKind = "publish" | "subscribe" | "unsubscribe";

/**
 * 未连接时入队的单条任务。
 * 带 `scopeId` 的任务可在对应 MqttScope.dispose 时被取消，
 * 避免模块销毁后仍被 CONNECT 冲刷执行。
 */
export interface PendingAction {
  id: number;
  kind: PendingKind;
  /** 所属作用域；全局调用不带此字段 */
  scopeId?: symbol;
  /** 为 true 时冲刷阶段会跳过 */
  cancelled: boolean;
  /** publish 载荷 */
  msg?: unknown;
  /** publish 指定 topic（完整路径） */
  pt?: string;
  options?: IMqttPublishOptions;
  /** subscribe / unsubscribe 的 id 段 */
  topicId?: string;
  /** subscribe / unsubscribe 的主题前缀，不传则用配置里的 subscribeTopic */
  st?: string;
}

使用说明（摘要） ​

在 server.ts 中启动 + 优雅退出 ​

ts
import http from "http";

import app from "./app";
import { coreConfigEnv } from "./core/config/env";
import { registerGracefulShutdown } from "./core/lifecycle";
import { mqtt, stopMqttClient, mqttClientMsg } from "./plugins/mqtt/client";

const server = http.createServer(app.callback());

registerGracefulShutdown({
  server,
  tasks: [
    {
      name: "mqtt-client",
      run: () => {
        stopMqttClient();
      },
    },
  ],
});

server.listen(coreConfigEnv.ENV_API_SERVER_PORT, async () => {
  try {
    mqtt.connect();
  } catch (error) {
    logger.mqttClient.error(mqttClientMsg("启动失败"), error);
  }
});

进程内收发 / Scope ​

ts
import { mqtt, enumMqttEvent } from "./plugins/mqtt/client";

mqtt.on(enumMqttEvent.CONNECT, () => {
  mqtt.subscribe("sensor/001"); // → device/sensor/001
});

mqtt.on(enumMqttEvent.MESSAGE, ({ topic, data }) => {
  logger.mqttClient.debug("收到消息", { topic, data });
});

function startDeviceWatcher(deviceId: string) {
  const scope = mqtt.createScope();
  scope.subscribe(deviceId);
  const onMessage = ({ topic, data }: { topic: string; data: unknown }) => {
    if (!topic.endsWith(deviceId)) return;
    logger.mqttClient.debug("设备消息", data);
  };
  mqtt.on(enumMqttEvent.MESSAGE, onMessage);
  return () => {
    mqtt.off(enumMqttEvent.MESSAGE, onMessage);
    scope.dispose();
  };
}

注意事项 ​

  • Scope 的 publish / subscribe：dispose() 自动清理队列并退订
  • mqtt.on：挂在进程单例，不会随 Scope.dispose 清除，必须配对 off
  • 全局 mqtt.subscribe（未走 scope）：结束时需自行 unsubscribe
  • 日志文案前缀统一走 MQTT_CLIENT_MODULE_NAME（mqttClientMsg），改名只改 config
  • 进程退出用 registerGracefulShutdown，任务里调 stopMqttClient()；勿再叠加 SIGINT/SIGTERM 钩子
  • 优雅退出建议先 stopMqttClient 再 stopMqttServerPlugin
  • 单例在首次加载 index 时创建，须保证此前已执行 bootstrap（dotenv / coreConfigEnv）
