/**
 * 解析最终可连接的 WebSocket 地址（浏览器端）
 */
export function resolveWsUrl() {
  const raw = import.meta.env.VITE_APP_WEBSOCKET_URL || "";
  if (!raw) {
    return "";
  }
  // 已是完整 ws/wss
  if (/^wss?:\/\//i.test(raw)) {
    return raw;
  }
  // 相对路径 → 拼当前 host
  if (typeof location === "undefined") {
    return "";
  }
  const protocol = location.protocol === "https:" ? "wss:" : "ws:";
  return `${protocol}//${location.host}${raw}`;
}

/**
 * 业务页面中使用 $socket
 */
import useHooksGlobalProperty from "@/hooks/use-global-property";
import { enumSocketEvent } from "@/libs/ts/socket";

const { $socket } = useHooksGlobalProperty();

const onMessage = (data: unknown) => {
  console.log("收到消息:", data);
};

onMounted(() => {
  $socket.on(enumSocketEvent.OPEN, () => {
    console.log("连接已建立");
  });

  $socket.on(enumSocketEvent.MESSAGE, onMessage);

  $socket.on(enumSocketEvent.CLOSE, (event) => {
    console.log("连接关闭", event);
  });

  $socket.on(enumSocketEvent.ERROR, (error) => {
    console.log("连接错误", error);
  });
});

onUnmounted(() => {
  // 只移除本页监听，不要随意 close（全局连接由登录态管理）
  $socket.off(enumSocketEvent.MESSAGE, onMessage);
});

// 发送
$socket.send({ type: "chat", message: "Hello" });
$socket.send("Hello World");

import PluginSocket, { enumSocketEvent } from "cy-plugin-websocket";
// 定义一个函数作为监听器（注意要保存引用，方便 off 时移除）
function handleMessage(data: any) {
  console.log("📨 收到消息:", data);
}
// 创建客户端实例
const client = new PluginSocket("ws://localhost:9527/ws-api/socket", {
  debug: true,
  heartOption: {
    interval: 10000,
  },
  reconnectOption: {
    maxCount: 3,
  },
});
// 监听
client.on(enumSocketEvent.OPEN, () => {
  console.log("已经打开连接");
});
client.on(enumSocketEvent.MESSAGE, handleMessage);
client.on(enumSocketEvent.CLOSE, () => {
  console.log("已经关闭连接");
});
client.on(enumSocketEvent.ERROR, (error: any) => {
  console.log("出现错误", error);
});
// 开始连接
client.connect();
// 发送消息
client.send({
  a: 1,
});
// 关闭连接
client.close();
// 移除监听
client.off(enumSocketEvent.MESSAGE, handleMessage);
// 移除所有监听
client.offAll();

/**
 * 自定义 WebSocket 参数
 * @param socketOption
 * @property {boolean} [socketOption.debug=false] - 是否开启调试模式(可选)
 * @property {boolean} [socketOption.connectResend=false] - 是否开启重连机制(可选)
 * @property {ISocketHeartOption} [socketOption.heartOption] - 心跳配置(可选)
 * @property {ISocketReconnectOption} [socketOption.reconnectOption] - 重试配置(可选)
 * @property {(data: any) => any} [socketOption.handleMessage] - 处理推送数据的回调函数(可选)
 */
export interface ISocketOption {
  /** 是否开启调试模式 */
  debug?: boolean;

  /** 是否开启重连机制 */
  connectResend?: boolean;

  /** 心跳配置 */
  heartOption?: ISocketHeartOption;

  /** 重试配置 */
  reconnectOption?: ISocketReconnectOption;

  /** 处理推送数据的回调函数 */
  handleMessage?: (data: any) => any;
}

/**
 * 心跳配置
 * @property {Record<string, unknown>} message - 心跳数据
 * @property {number} interval - 间隔时间
 * @property {number | null} heartbeatTimer - 定时器
 */
export interface ISocketHeartOption {
  /** 心跳数据 */
  message?: any;

  /** 间隔时间 */
  interval?: number;

  /** 定时器 */
  timer?: number | null;
}

/**
 * 重试配置
 * @property {number} [maxCount] - 最大重试次数(默认 Infinity)
 * @property {number} [interval] - 重试间隔 (默认 5_1000)
 * @property {number} [maxInterval] - 最大重试间隔 (默认60_000)
 */
export interface ISocketReconnectOption {
  /**
   * 最大重试次数
   */
  maxCount?: number;

  /** 重试间隔 */
  interval?: number;

  /** 最大重试间隔 */
  maxInterval?: number;
}

// 枚举
/** websocket状态枚举 */
export enum enumSocketStatus {
  /** 正在连接 */
  CONNECTING,

  /** 已连接 */
  OPEN,

  /** 正在关闭 */
  CLOSING,

  /** 已关闭 */
  CLOSED,
}

/** websocket事件枚举 */
export enum enumSocketEvent {
  /** 开启 */
  OPEN = "open",

  /** 消息 */
  MESSAGE = "message",

  /** 关闭 */
  CLOSE = "close",

  /** 错误 */
  ERROR = "error",
}

/*
 *  全局 websocket
 */

import { ref } from "vue";
import { defineStore } from "pinia";
import pluginSocket from "cy-plugin-websocket";

export const useSocketStore = defineStore("socket", () => {
  // socket 初始化的对象
  const storeSocket = ref({
    ready: false,
    ws: null,
  });

  // 当前返回的消息
  const storeSocketMsg = ref({});

  // 初始化 socket
  function storeSocketInit() {
    storeSocket.value.ws = pluginSocket({
      onMessage: (message) => {
        storeSocketMsg.value = message;
        console.log("socket推送了数据:");
        console.log(JSON.stringify(storeSocketMsg.value, null, 2));
      },
      // 异常断开重新连接
      onErrorClose: () => {
        console.log("全局socket异常断开，5s后重新连接");
        setTimeout(() => {
          storeSocketInit();
        }, 5000);
      },
    });

    storeSocket.value.ws.connect().then((res) => {
      storeSocket.value.ready = true;
      console.log("全局socket连接成功");

      // 连接成功需要处理的事情
    });
  }

  // 发送消息
  function storeSocketSend(message) {
    if (storeSocket.value.ready) {
      storeSocket.value.ws.send(message, true);
    }
  }

  return {
    storeSocket,
    storeSocketMsg,
    storeSocketInit,
    storeSocketSend,
  };
});
