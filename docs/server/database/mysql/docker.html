txt
mysql/
├─ docker-compose.yml          # Compose 主文件
├─ etc/
│   └─ mysql/
│       └─ my.cnf              # MySQL 配置文件
├─ var/
│   ├─ lib/

│   └─ log/
│       └─ mysql/              # MySQL 日志目录
├─ mysql-init/
│   └─ init.sql                # 容器启动后初始化脚本

启动容器 ​

yaml
version: "3.9"

services:
  mysql:
    image: mysql:5.7
    container_name: mysql
    restart: always
    ports:
      - "3306:3306"
    environment:
      MYSQL_ROOT_PASSWORD: xxxxxxx
      # MYSQL_DATABASE: ${MYSQL_DATABASE}
      TZ: Asia/Shanghai
    command: >
      --default-authentication-plugin=mysql_native_password
    volumes:
      - ./etc/mysql/my.cnf:/etc/mysql/my.cnf:ro # my.cnf 挂载为只读
      - ./var/lib/mysql:/var/lib/mysql # 数据目录
      - ./var/log/mysql:/var/log/mysql # 日志目录
      - ./mysql-init:/docker-entrypoint-initdb.d # 初始化脚本
    networks:
      - net_mysql

networks:
  net_mysql:
    driver: bridge

ini
[mysqld]
# 基本配置
user=mysql                    # MySQL 服务运行用户
datadir=/var/lib/mysql        # 数据存储目录
socket=/var/run/mysqld/mysqld.sock  # Unix socket 文件，用于本地客户端连接
pid-file=/var/run/mysqld/mysqld.pid # MySQL PID 文件位置


skip-host-cache               # 禁用 host 缓存，提高连接速度
skip-name-resolve             # 禁止 DNS 解析，减少连接延迟
symbolic-links=0              # 禁止符号链接，提高安全性
secure-file-priv=/var/lib/mysql-files  # 限制 LOAD DATA / OUTFILE 目录

# 性能优化
max_connections = 2000        # 最大连接数
thread_cache_size = 100       # 缓存线程，减少频繁创建销毁开销
table_open_cache = 4000       # 打开表缓存，提高查询性能
open_files_limit = 65535      # 打开文件限制

# InnoDB 内存优化
# innodb_buffer_pool_size = 12G      # 数据和索引缓存，建议占 70%-80% RAM
# innodb_buffer_pool_instances = 12  # 分区缓冲池，提高并发性能
# innodb_log_file_size = 1G          # redo log 大小，提高写入性能
# innodb_log_buffer_size = 64M       # redo log 缓冲区
# innodb_flush_log_at_trx_commit = 1 # 每事务同步到磁盘，最安全
# innodb_file_per_table = 1           # 每表独立 ibd 文件，方便管理
# innodb_flush_method = O_DIRECT      # 避免双重缓存，提高性能

# 其他缓存
join_buffer_size = 2M
sort_buffer_size = 4M
read_rnd_buffer_size = 4M

# 字符集与排序规则
character-set-server = utf8mb4          # 默认服务器字符集，支持多语言和 Emoji
collation-server = utf8mb4_unicode_ci   # 默认服务器校对规则，Unicode 排序更准确


# 每次客户端连接初始化字符集
init_connect='SET NAMES utf8mb4'               # 设置连接字符集为 utf8mb4
init_connect='SET collation_connection = utf8mb4_unicode_ci'  # 设置连接校对规则


# 二进制日志 (binlog) 配置
log-bin=mysql-bin               # 开启 binlog
server_id=1                     # 主从复制唯一 ID
expire_logs_days = 30           # 自动清理 binlog
max_binlog_size = 1G            # 单个 binlog 文件最大大小
binlog_format = ROW             # 使用行格式
sync_binlog = 1                 # 每次事务同步 binlog，保证数据安全


# SQL 模式
# STRICT_TRANS_TABLES      : 插入非法数据时报错
# NO_ZERO_IN_DATE           : 禁止 '2023-00-01' 这种零月/零日
# NO_ZERO_DATE              : 禁止 '0000-00-00'
# ERROR_FOR_DIVISION_BY_ZERO: 除零报错
# NO_AUTO_CREATE_USER       : 禁止 GRANT 自动创建用户
# NO_ENGINE_SUBSTITUTION    : 不允许引擎替换，保持严格表类型
sql-mode=STRICT_TRANS_TABLES,NO_ZERO_IN_DATE,NO_ZERO_DATE,ERROR_FOR_DIVISION_BY_ZERO,NO_AUTO_CREATE_USER,NO_ENGINE_SUBSTITUTION


# 日志配置
log-error=/var/log/mysql/mysqld.log      # 错误日志
slow_query_log = 1                        # 开启慢查询日志
slow_query_log_file = /var/log/mysql/mysql-slow.log
long_query_time = 2                        # 超过 2 秒的查询记录
log_queries_not_using_indexes = 1         # 记录未使用索引的查询


# 安全与维护
# lower_case_table_names = 1              # Windows 或大小写敏感环境可配置
# max_allowed_packet = 64M                 # 最大数据包，避免大字段插入失败


!includedir /etc/mysql/conf.d/
!includedir /etc/mysql/mysql.conf.d/

sql
-- 创建数据库
CREATE DATABASE IF NOT EXISTS 数据库 CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- 创建用户
CREATE USER IF NOT EXISTS '账户'@'%' IDENTIFIED BY '密码';

-- 授权用户访问对应数据库
GRANT ALL PRIVILEGES ON 数据库.* TO '账户'@'%';

-- 刷新权限
FLUSH PRIVILEGES;

dockerfile
# 使用 Node.js 24.13.0 作为基础镜像
FROM node:24.13.0-slim

# 设置工作目录
WORKDIR /app

# 复制依赖文件和配置
COPY package.json package-lock.json ./

# 复制 Prisma 配置文件
COPY prisma.config.ts ./

# Prisma 必须文件
COPY prisma ./prisma

# 复制 dist 文件到 src 目录
COPY dist ./src

# 复制启动脚本
COPY docker-entrypoint.sh  ./

# 安装依赖(不安装  devDependencies)
RUN npm ci --omit=dev --no-audit --no-fund

# 安装 openssl、 mysql 客户端和 gzip
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    openssl \
    default-mysql-client \
    gzip && \
    rm -rf /var/lib/apt/lists/*

# # 暴露端口
# EXPOSE 9527

# 启动
# CMD ["node","src/server.js"]
CMD ["sh", "docker-entrypoint.sh"]
