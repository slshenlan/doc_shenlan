Docker 安装 redis ​

文件夹结构 ​

txt
redis/
├─ docker-compose.yml        # Compose 主文件
├─ conf/
│   └─ redis.conf            # Redis 配置文件
├─ data/                     # Redis 数据持久化目录（RDB/AOF）
├─ var/
│   └─ log/
│       └─ redis/            # Redis 日志目录

启动容器 ​

yaml
services:
  redis:
    image: redis:7.2.0
    container_name: redis # 容器名
    user: "1001:1001"
    restart: always
    ports:
      - "6379:6379"
    volumes:
      - ./conf/redis.conf:/usr/local/etc/redis/redis.conf:ro # 配置
      - ./data:/data # 数据
      - ./var/log/redis:/var/log/redis # 日志

    # 指定容器启动时执行的命令，使用挂载的配置文件启动 Redis
    command: ["redis-server", "/usr/local/etc/redis/redis.conf"]

    environment:
      - TZ=Asia/Shanghai
    networks:
      - net_redis
networks:
  net_redis:
    driver: bridge

# Redis 配置文件示例

# 允许所有 IP 访问 Redis
bind 0.0.0.0

#Redis 访问密码
requirepass xxx

# 进程 ID 文件存放路径(docker内不用配置)
# pidfile /var/run/redis_6379.pid

# Redis 工作目录，RDB 和 AOF 文件存放路径
dir /data

# 指定日志文件路径。
logfile /var/log/redis/redis.log

# Redis 监听的端口号，默认 6379
port 6379

# 启用保护模式。如果没有明确的 bind 规则且无密码，会拒绝远程访问。
protected-mode yes

# 指定 TCP 连接的 backlog（等待队列），如果客户端连接过多，可以适当调大
tcp-backlog 511

# 客户端空闲超时断开时间（秒），0 表示永不超时
timeout 0

# TCP 保持连接的时间（秒），用于检测断开的连接，默认 300 秒
tcp-keepalive 300

# 是否以守护进程方式运行，no 代表前台运行（适用于 Docker）
daemonize no

# 进程监督方式，no 代表不使用 systemd 或 upstart 进行管理
supervised no

# 日志级别（debug、verbose、notice、warning)
loglevel notice

# 设置数据库数量，默认 16
databases 16

# 关闭 Redis 启动 Logo
always-show-logo no

# 允许修改 Redis 进程名称，便于 ps 查找
set-proc-title yes

#进程标题格式
proc-title-template "{title} {listen-addr} {server-mode}"

#影响 Redis SORT 指令的本地化排序规则
# locale-collate ""

# RDB 规则
# 900 1：900 秒内至少 1 次修改就触发快照。
# 300 10：300 秒内 10 次修改触发快照。
# 60 10000：60 秒内 10000 次修改触发快照。
save 900 1 300 10 60 10000

# RDB 失败时停止写入，避免数据丢失
stop-writes-on-bgsave-error yes

# 是否对 RDB 进行压缩存储
rdbcompression yes

# 计算 RDB 文件的校验和，防止损坏
rdbchecksum yes

# RDB 数据文件名称。
dbfilename dump.rdb

#是否同步删除 dump.rdb 文件
rdb-del-sync-files no

# 允许副本节点提供过期数据
replica-serve-stale-data yes

# 副本只读，防止误修改
replica-read-only yes

# 是否启用无磁盘同步
repl-diskless-sync no

# 无磁盘同步的等待时间
repl-diskless-sync-delay 5

# 最大无磁盘同步副本数，0 表示不限制
# repl-diskless-sync-max-replicas 0

# 副本加载方式
repl-diskless-load disabled

# 复制过程中是否启用 TCP_NODELAY，no 代表减少延迟
repl-disable-tcp-nodelay no

# 副本优先级，0 代表永不提升为主节点
replica-priority 100

# 记录 ACL 日志的最大条数
acllog-max-len 128

# 控制 lazyfree 相关功能，默认 no（同步释放内存）
lazyfree-lazy-eviction no
lazyfree-lazy-expire no
lazyfree-lazy-server-del no
replica-lazy-flush no
lazyfree-lazy-user-del no
lazyfree-lazy-user-flush no

# OOM 调整分值
oom-score-adj no
oom-score-adj-values 0 200 800

# 禁用透明大页（推荐开启以提高性能)
disable-thp yes

# 关闭 AOF 持久化
appendonly no

# AOF 文件名称。
appendfilename "appendonly.aof"

# AOF 目录名称
# appenddirname "appendonlydir"

# AOF 写入磁盘策略
# always：每次写入都同步到磁盘（性能损耗大）。
# everysec：每秒同步（推荐）。
# no：由操作系统决定。
#
appendfsync everysec

# AOF 重写时是否禁用 fsync
no-appendfsync-on-rewrite no


# AOF 重写条件
# 100：AOF 文件大小增长 100% 触发重写
# 64mb：AOF 文件至少达到 64MB 才会触发重写
#
auto-aof-rewrite-percentage 100
auto-aof-rewrite-min-size 64mb

# AOF 读取损坏数据时是否继续加载
aof-load-truncated yes

# AOF 文件是否使用 RDB 预加载格式
aof-use-rdb-preamble yes

# 是否在 AOF 记录 Unix 时间戳
# aof-timestamp-enabled no

# Lua 脚本执行最大时间（毫秒）
lua-time-limit 5000

# 超过 10ms 记录慢查询日志
slowlog-log-slower-than 10000

# 记录最大慢查询条数
slowlog-max-len 128

# 延迟监控阈值（0 代表关闭）
latency-monitor-threshold 0

# 监听键空间事件（默认为空）
notify-keyspace-events ""

# 数据结构优化
hash-max-ziplist-entries 512
hash-max-ziplist-value 64
set-max-intset-entries 512

zset-max-ziplist-entries 128
zset-max-ziplist-value 64
hll-sparse-max-bytes 3000
stream-node-max-bytes 4096
stream-node-max-entries 100


# 开启 rehashing 以优化内存回收
activerehashing yes

# Both the hard or the soft limit can be disabled by setting them to zero.
client-output-buffer-limit normal 0 0 0
client-output-buffer-limit replica 256mb 64mb 60
client-output-buffer-limit pubsub 32mb 8mb 60

# 控制后台任务执行频率
hz 10

# 动态调整频率
dynamic-hz yes

# 允许增量 fsync 提高性能
aof-rewrite-incremental-fsync yes
rdb-save-incremental-fsync yes

# 开启 jemalloc 后台线程优化内存管理
jemalloc-bg-thread yes
