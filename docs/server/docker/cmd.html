shell
# 启动 示例1
docker-compose up -d

# 启动 示例2
docker compose \
  --env-file .env.test \
  -f docker-compose-test.yml \
  up -d

# 停止所有服务容器，但不删除它们。
# 下次启动时，数据卷和网络还在，
# 可以直接 docker-compose start 恢复
docker-compose stop

# 停止并删除容器、网络，但保留卷
docker-compose down

# 进入容器
docker exec -it 容器  /bin/bash

# 退出容器
exit

# 查找启动文件地址
docker inspect 容器 --format '{{json .Config.Labels}}' | jq

# 查看容器详细信息
docker inspect 容器

# 只显示某个容器挂载路径
docker inspect -f '{{range .Mounts}}{{.Source}} -> {{.Destination}}{{println}}{{end}}' 容器

# 显示所有容器完整挂载路径
docker ps --no-trunc --format "table {{.Names}}\t{{.Mounts}}"

docker logs 容器

# 查看容器实时日志
docker logs -f 容器

# 查看容器实时日志(指定行数)
docker logs -f --tail 100 <容器名>

# 查看容器实时日志（最近 10 分钟内日志） Docker 20.10+
docker logs --since 10m -f <容器名>

shell
# 查看所有镜像
docker images

# 设置镜像名称
docker tag xxxx 镜像名:v1.0

# 删除镜像
docker rmi 镜像

# 删除旧标签
docker rmi aaa:latest

# 重新命名
docker tag aaa:latest bbb:latest

# 保存到本地
docker save -o /Users/xiebin/Downloads/自定义文件名.tar 镜像名

# 载入
docker load < 名称.tar

# 查询 容器 所在网络
docker inspect 容器 | grep -i networks -A 10

docker network ls

docker network create 网络

# 删除指定的网络

docker network connect 网络 容器

docker network disconnect 网络 容器
