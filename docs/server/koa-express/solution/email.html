/*
 * 邮件
 */

import nodemailer from "nodemailer";
import SMTPPool from "nodemailer/lib/smtp-pool";
import SMTPTransport from "nodemailer/lib/smtp-transport";
import pLimit from "p-limit";

import logger from "./logger";
import { IBatchSendProgress, IMailConfig } from "./types";

import { createTraceId, runWithTraceId } from "@/dev-set/core/modules/trace";

export class Mailer {
  /** 默认配置 */
  private defaultConfig = {
    /** 最大连接数 */
    maxConnections: 3,

    /** 每秒发送的最大邮件数 */
    rateLimit: 5,

    /** 池中单个连接最大请求数 */
    maxMessages: 100,

    /** 重试次数 */
    retry: 3,

    /** 批量发送大小 */
    batchSize: 50,

    /** 并发数量 */
    limiterSize: 5,
  };

  /** 邮件发送器 */
  private transporter: nodemailer.Transporter<SMTPPool.SentMessageInfo>;

  /** 标记是否已验证过 */
  private verified = false;

  private verifyPromise: Promise<boolean> | null = null;

  /** 应用层并发限制器 */
  private limiter: ReturnType<typeof pLimit>;

  /** 当前正在执行的 send 数量 */
  private runningCount = 0;

  constructor(private config: IMailConfig) {
    const dc = this.defaultConfig;
    const smtpOptions: SMTPPool.Options = {
      host: config.host,
      port: config.port,
      auth: config.auth,
      debug: false,
      secure: config?.secure || true,
      pool: config?.pool || true,
      maxConnections: config?.maxConnections || dc.maxConnections,
      rateLimit: config?.rateLimit || dc.rateLimit,
      maxMessages: config?.maxMessages || dc.maxMessages,
      logger: logger as SMTPTransport.Options["logger"],
    };

    // 创建邮件传输器
    this.transporter = nodemailer.createTransport(smtpOptions);

    // 应用层并发控制（与 SMTP pool 无关）
    this.limiter = pLimit(config.concurrency ?? dc.limiterSize);
  }

  /**
   * 单个 安全发送邮件（不会抛错，失败返回 null）
   */
  async send(
    options: nodemailer.SendMailOptions,
    retry = this.defaultConfig.retry,
  ): Promise<string | null> {
    return this.limiter(async () => {
      this.runningCount++;
      hcLogger.email.info("【 单个发送邮件开始 】", {
        当前并发数: this.runningCount,
        to: options.to,
      });

      try {
        const result = await this.safeSend(options, retry);
        return result;
      } finally {
        this.runningCount--;
        hcLogger.email.info("【 单个发送邮件结束 】", {
          当前并发数: this.runningCount,
          to: options.to,
        });
      }
    });
  }
  /**
   * 批量发送（一次 100 人也没问题）
   */
  async batchSend(
    mails: nodemailer.SendMailOptions[],
    onProgress?: (progress: IBatchSendProgress) => void,
  ): Promise<{
    success: Array<{ to: string; messageId: string }>;
    failed: Array<{ to: string; error: any }>;
  }> {
    const batchTraceId = createTraceId(); // 批次追踪 ID
    const batchSize = this.defaultConfig.batchSize; // 批量拆分大小，每批 Promise 数量
    const retry = this.defaultConfig.retry; // 重试次数
    const success: Array<{ to: string; messageId: string }> = []; // 成功的邮件
    const failed: Array<{ to: string; error: any }> = []; // 失败的邮件
    const totalBatches = Math.ceil(mails.length / batchSize); // 总批次数

    hcLogger.email.info("【 批量发送邮件开始 】 ", {
      total: mails.length,
      batchTraceId,
    });

    for (let i = 0; i < mails.length; i += batchSize) {
      const batch = mails.slice(i, i + batchSize);
      const results = await Promise.allSettled(
        batch.map((mail) => this.send(mail, retry)),
      );

      results.forEach((res, idx) => {
        const to = String(batch[idx].to);
        if (res.status === "fulfilled" && res.value) {
          success.push({ to, messageId: res.value });
        } else {
          failed.push({
            to,
            error:
              res.status === "rejected" ? res.reason : "send returned null",
          });
        }
      });

      // 进度回调
      onProgress?.({
        batchIndex: i / batchSize + 1,
        totalBatches,
        successCount: success.length,
        failedCount: failed.length,
      });

      hcLogger.email.info("【 批量发送邮件结束 】 ", {
        success: success.length,
        failed: failed.length,
        batchTraceId,
      });
    }

    return { success, failed };
  }

  /**
   * 真正的安全发送实现
   */
  private async safeSend(
    options: nodemailer.SendMailOptions,
    retry: number,
  ): Promise<string | null> {
    const traceId = createTraceId();

    try {
      return await runWithTraceId(traceId, async () => {
        if (!this.verified) {
          if (!this.verifyPromise) {
            // 第一次触发验证
            this.verifyPromise = this.verify();
          }
          this.verified = await this.verifyPromise;
          this.verifyPromise = null; // 清理缓存
        }

        // traceId 贯穿整个发送流程
        return this.sendMethod(options, retry);
      });
    } catch (err) {
      hcLogger.email.error("邮件发送最终失败（safeSend 已捕获）", err);
      return null;
    }
  }

  /** 发送邮件（并发安全、重试不丢失 sendId） */
  private async sendMethod(
    options: nodemailer.SendMailOptions,
    retry = 3,
  ): Promise<string> {
    hcLogger.email.info("开始发送邮件", {
      to: options.to,
      subject: options.subject,
      html: options.html,
    });
    let lastErr: any = null;
    for (let i = 0; i <= retry; i++) {
      try {
        if (i > 0) {
          const delay = Math.pow(2, i) * 500 + Math.random() * 200; // 指数退避 + 随机抖动
          hcLogger.email.info(
            `重试 ${i}/${retry} for ${options.to}, 等待 ${delay.toFixed(0)}ms`,
          );
          await new Promise((r) => setTimeout(r, delay));
        }

        const info = await this.transporter.sendMail(options);

        hcLogger.email.info(
          `发送到: ${options.to}, messageId: ${info.messageId}`,
        );
        return info.messageId;
      } catch (err: any) {
        lastErr = err;
        hcLogger.email.error(
          `[发送失败] attempt ${i}/${retry} to: ${options.to}, error: ${err.message}`,
        );
      }
    }

    hcLogger.email.info("邮件发送结束 (RETRY EXHAUSTED) ");

    throw lastErr;
  }

  /** 验证 SMTP 连接配置是否正确 */
  async verify(): Promise<boolean> {
    try {
      await this.transporter.verify();
      hcLogger.email.info("SMTP 配置验证成功");
      return true;
    } catch (err) {
      hcLogger.email.error("SMTP 配置验证失败", err);
      return false;
    }
  }
}

const mailer = new Mailer({
  host: "smtp.feishu.cn",
  port: 465,
  auth: {
    user: "xxx@xxx.com",
    pass: "xxxxxxx",
  },
});

export default mailer;

import type { ILogger } from "./types";

function isBlockedBeforeSend(msg: any): boolean {
  if (!msg || typeof msg !== "object") {
    return false;
  }

  const { component, tnx, stage, action } = msg;

  // 发送前阶段需要屏蔽
  if (component === "mail" && tnx === "create") {
    return true;
  }
  if (
    component === "smtp-connection" &&
    (tnx === "dns" || tnx === "network" || tnx === "smtp" || tnx === "message")
  ) {
    return true;
  }
  if (component === "smtp-transport" && action !== "send") {
    return true;
  }
  if (!component && stage === "sent") {
    return true;
  }
  if (component === "mail" && action === "send") {
    return true;
  }

  return false;
}

/**
 * 拦截 SMTPTransport 发送前的日志
 */
function handler(level: string, msg: string) {
  if (isBlockedBeforeSend(msg)) {
    return;
  }
  hcLogger.email.nodemailer(level, msg);
  // hcLogger.stdout(level, msg);
}

// 创建日志
const logger: ILogger = {
  // 必须字段
  level: (msg) => {
    handler("LEVEL", msg);
  },
  debug: (msg) => {
    handler("DEBUG", msg);
  },
  info: (msg) => {
    handler("INFO", msg);
  },
  warn: (msg) => {
    handler("WARN", msg);
  },
  error: (msg) => {
    hcLogger.email.error(msg);
  },
  trace: () => {},
  fatal: () => {},
};

export default logger;

/**
 * 邮件配置
 */
export interface IMailConfig {
  /** 邮件服务器地址 */
  host: string;

  /** 邮件服务器端口 */
  port: number;

  /** 邮件服务器认证信息 */
  auth: {
    user: string;
    pass: string;
  };

  /** 是否使用 SSL */
  secure?: boolean;

  /** 是否使用连接池 */
  pool?: boolean;

  /** 连接池最大连接数 */
  maxConnections?: number;

  /** 池中单个连接最大请求数 */
  maxMessages?: number;

  /** 每秒发送的最大邮件数 */
  rateLimit?: number;

  /** 并发数 */
  concurrency?: number;
}

/**
 * 批量发送进度
 */
export interface IBatchSendProgress {
  batchIndex: number;
  totalBatches: number;
  successCount: number;
  failedCount: number;
}

/**
 * 日志
 */
export interface ILogger {
  level: string | ((level: string) => void);
  debug: (msg: any) => void;
  info: (msg: any) => void;
  warn: (msg: any) => void;
  error: (msg: any) => void;
  trace: () => void;
  fatal: () => void;
}

import { loggers } from "../config/log4";
import {
  formatDebugData,
  formatConsoleData,
  formatErrorData,
  getStackTrace,
} from "../config/utils";

const log = {
  /**
   * 信息
   */
  info: (...args: any[]) => {
    const stack = getStackTrace();
    const msg = formatInfoData(stack, ...args);

    loggers.logEmailInfo.info(msg);
  },

  /**
   * 调试
   */
  debug: (...args: any[]) => {
    const stack = getStackTrace();
    const msg = formatDebugData(stack, ...args);

    loggers.logEmailDebug.debug(msg);
  },

  /**
   * 错误
   */
  error: (...args: any[]) => {
    const stack = getStackTrace();
    const msg = formatErrorData(stack, ...args);

    loggers.logEmailError.error(msg);
  }

  /**
   * nodemailer打印
   */
  nodemailer: (...args: any[]) => {
    const stack = getStackTrace();
    const msg = formatConsoleData(stack, ...args);

    loggers.logEmailNodemailer.info(msg);
  },
};

export default log;

export interface IPubLogger {
  email: {
    /**
     * 信息日志
     *
     * @example
     * ```ts
     * hcLogger.email.info('任务开始');
     * hcLogger.email.info('执行结果', { demo: 1 });
     * ```
     */
    info: (...args: any[]) => void;

    /**
     * 调试日志

     * @example
     * ```ts
     * hcLogger.email.debug('任务开始');

     * ```

     * 错误日志

  };
}
