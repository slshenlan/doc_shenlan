ts
import { requestDefaultConfig } from "../config";

import { errorCodes, requestMethods } from "./enum";
import type { TErrorCode, TRequestMethod } from "./enum";
import { RequestError } from "./error";
import type {
  IErrorHandler,
  IRequestConfig,
  IRequestContext,
  IRequestInterceptor,
  IRequestOptions,
  IResponseInterceptor,
  RequestHeaders,
} from "./types";

/** 可自动重试的错误码 */
const RETRYABLE_CODES: TErrorCode[] = [
  errorCodes.TIMEOUT,
  errorCodes.NETWORK_ERROR,
];

/**
 * 请求客户端
 */
export class Request {
  private baseUrl: string;
  private timeout: number;
  private credentials: RequestCredentials;
  private headers: RequestHeaders | (() => RequestHeaders);
  private retry: number;
  private retryDelay: number;
  private requestInterceptor?: IRequestInterceptor;
  private responseInterceptor?: IResponseInterceptor;
  private errorHandler?: IErrorHandler;

  /**
   * 创建请求
   *
   * @param config 请求配置
   */
  constructor(config: IRequestConfig = {}) {
    this.baseUrl = config.baseUrl ?? "";
    this.timeout = config.timeout ?? requestDefaultConfig.timeout;
    this.credentials = config.credentials ?? "same-origin";
    this.headers = config.headers ?? {};
    this.retry = config.retry ?? requestDefaultConfig.retry;
    this.retryDelay = config.retryDelay ?? requestDefaultConfig.retryDelay;
    this.requestInterceptor = config.requestInterceptor;
    this.responseInterceptor = config.responseInterceptor;
    this.errorHandler = config.errorHandler;
  }

  /**
   * 获取请求头（支持工厂函数）
   */
  private getHeaders(): RequestHeaders {
    return typeof this.headers === "function" ? this.headers() : this.headers;
  }

  /**
   * 拼接 url 与查询参数
   */
  private buildUrl(url: string, params?: Record<string, unknown>): string {
    let full = this.baseUrl + url;

    if (params && Object.keys(params).length > 0) {
      const query = new URLSearchParams(
        params as Record<string, string>
      ).toString();

      if (query) {
        full += `?${query}`;
      }
    }

    return full;
  }

  /**
   * 构建请求上下文
   */
  private buildContext(
    method: TRequestMethod,
    url: string,
    data: unknown,
    options: IRequestOptions
  ): IRequestContext {
    const isBodyMethod = method !== requestMethods.GET;

    return {
      url: isBodyMethod
        ? this.baseUrl + url
        : this.buildUrl(url, data as Record<string, unknown>),
      headers: {
        ...this.getHeaders(),
        ...options.headers,
      },
      method,
      options,
      body: isBodyMethod ? data : undefined,
    };
  }

  /**
   * 序列化请求体
   *
   * - FormData / Blob / ArrayBuffer / URLSearchParams / 字符串 原样传递
   * - 普通对象 / 数组 序列化为 JSON
   */
  private serializeBody(body: unknown): BodyInit | undefined {
    if (body == null) {
      return undefined;
    }

    if (
      typeof body === "string" ||
      body instanceof FormData ||
      body instanceof Blob ||
      body instanceof ArrayBuffer ||
      ArrayBuffer.isView(body) ||
      body instanceof URLSearchParams
    ) {
      return body as BodyInit;
    }

    return JSON.stringify(body);
  }

  /**
   * 解析响应体
   */
  private async parseResponse(
    response: Response,
    responseType?: IRequestOptions["responseType"]
  ): Promise<unknown> {
    const type = responseType ?? this.guessResponseType(response);

    switch (type) {
      case "text":
        return response.text();
      case "blob":
        return response.blob();
      case "arrayBuffer":
        return response.arrayBuffer();
      default:
        return response.json();
    }
  }

  /**
   * 根据 Content-Type 猜测响应类型
   */
  private guessResponseType(response: Response): "json" | "text" {
    const contentType = response.headers.get("content-type") ?? "";
    return contentType.includes("application/json") ? "json" : "text";
  }

  /**
   * 创建 HTTP 错误（非 2xx）
   */
  private async createHttpError(response: Response): Promise<RequestError> {
    const text = await response.text();
    let message = "请求失败";

    if (text) {
      try {
        const body = JSON.parse(text);

        if (body && typeof body.message === "string") {
          message = body.message;
        }
      } catch {
        // 非 JSON 响应体，保留默认提示
      }
    }

    return new RequestError({
      code: errorCodes.HTTP_ERROR,
      message,
      status: response.status,
      response,
    });
  }

  /**
   * 归一化错误
   */
  private normalizeError(
    error: unknown,
    externalSignal?: AbortSignal
  ): RequestError {
    if (error instanceof RequestError) {
      return error;
    }

    const isAbort = error instanceof Error && error.name === "AbortError";

    return new RequestError({
      code: isAbort
        ? externalSignal?.aborted
          ? errorCodes.CANCELLED
          : errorCodes.TIMEOUT
        : errorCodes.NETWORK_ERROR,
      message: error instanceof Error ? error.message : String(error),
    });
  }

  /**
   * 延迟等待（外部信号触发取消时提前返回）
   */
  private sleep(ms: number, signal?: AbortSignal): Promise<void> {
    return new Promise((resolve) => {
      if (signal?.aborted) {
        resolve();
        return;
      }

      const timer = setTimeout(() => {
        cleanup();
        resolve();
      }, ms);

      const onAbort = () => {
        cleanup();
        resolve();
      };

      const cleanup = () => {
        clearTimeout(timer);
        signal?.removeEventListener("abort", onAbort);
      };

      signal?.addEventListener("abort", onAbort, { once: true });
    });
  }

  /**
   * 执行单次请求
   */
  private async execute<Res>(
    method: TRequestMethod,
    url: string,
    data: unknown,
    options: IRequestOptions,
    timeout: number
  ): Promise<Res> {
    let config = this.buildContext(method, url, data, options);

    /**
     * 请求拦截
     */
    if (this.requestInterceptor) {
      config = await this.requestInterceptor(config);
    }

    const controller = new AbortController();
    const externalSignal = options.signal;

    // 外部取消时中止内部控制器
    const onExternalAbort = () => controller.abort();

    if (externalSignal) {
      if (externalSignal.aborted) {
        controller.abort();
      } else {
        externalSignal.addEventListener("abort", onExternalAbort, {
          once: true,
        });
      }
    }

    // 超时覆盖整个请求（含响应体读取）
    const timer = setTimeout(() => controller.abort(), timeout);

    try {
      const response = await fetch(config.url, {
        method: config.method,
        headers: config.headers,
        body: this.serializeBody(config.body),
        signal: controller.signal,
        credentials: options.credentials ?? this.credentials,
        mode: options.mode,
        cache: options.cache,
        redirect: options.redirect,
      });

      if (!response.ok) {
        throw await this.createHttpError(response);
      }

      const result = await this.parseResponse(response, options.responseType);

      console.log("result:", result);

      /**
       * 响应拦截
       */
      if (this.responseInterceptor) {
        return await (this.responseInterceptor as IResponseInterceptor<Res>)({
          response,
          data: result as Res,
          config,
        });
      }

      return result as Res;
    } finally {
      clearTimeout(timer);

      if (externalSignal) {
        externalSignal.removeEventListener("abort", onExternalAbort);
      }
    }
  }

  /**
   * 发送请求（含重试与统一错误处理）
   */
  private async request<Req, Res>(
    method: TRequestMethod,
    url: string,
    data?: Req,
    options: IRequestOptions = {}
  ): Promise<Res> {
    const timeout = options.timeout ?? this.timeout;
    const retry = options.retry ?? this.retry;
    const retryDelay = options.retryDelay ?? this.retryDelay;

    let attempt = 0;

    while (true) {
      try {
        return await this.execute<Res>(method, url, data, options, timeout);
      } catch (error) {
        const requestError = this.normalizeError(error, options.signal);

        const shouldRetry =
          attempt < retry && RETRYABLE_CODES.includes(requestError.code);

        if (!shouldRetry) {
          if (this.errorHandler) {
            try {
              await this.errorHandler(requestError);
            } catch {
              // 错误处理器自身异常不影响错误抛出
            }
          }

          throw requestError;
        }

        attempt += 1;

        if (retryDelay > 0) {
          await this.sleep(retryDelay, options.signal);
        }

        // 等待期间被外部取消，不再重试
        if (options.signal?.aborted) {
          const cancelled = new RequestError({
            code: errorCodes.CANCELLED,
            message: "请求已取消",
          });

          if (this.errorHandler) {
            await this.errorHandler(cancelled);
          }

          throw cancelled;
        }
      }
    }
  }

  /**
   * 发送 get 请求
   * @param url 请求地址
   * @param params 查询参数
   * @param options 请求选项
   */
  get<Res = unknown>(
    url: string,
    params?: Record<string, unknown>,
    options?: IRequestOptions
  ): Promise<Res> {
    return this.request<Record<string, unknown>, Res>(
      requestMethods.GET,
      url,
      params,
      options
    );
  }

  /**
   * 发送 post 请求
   * @param url 请求地址
   * @param data 请求数据
   * @param options 请求选项
   */
  post<Req = unknown, Res = unknown>(
    url: string,
    data?: Req,
    options?: IRequestOptions
  ): Promise<Res> {
    return this.request<Req, Res>(requestMethods.POST, url, data, options);
  }

  /**
   * 发送 put 请求
   * @param url 请求地址
   * @param data 请求数据
   * @param options 请求选项
   */
  put<Req = unknown, Res = unknown>(
    url: string,
    data?: Req,
    options?: IRequestOptions
  ): Promise<Res> {
    return this.request<Req, Res>(requestMethods.PUT, url, data, options);
  }

  /**
   * 发送 patch 请求
   * @param url 请求地址
   * @param data 请求数据
   * @param options 请求选项
   */
  patch<Req = unknown, Res = unknown>(
    url: string,
    data?: Req,
    options?: IRequestOptions
  ): Promise<Res> {
    return this.request<Req, Res>(requestMethods.PATCH, url, data, options);
  }

  /**
   * 发送 delete 请求
   * @param url 请求地址
   * @param data 请求数据
   * @param options 请求选项
   */
  delete<Req = unknown, Res = unknown>(
    url: string,
    data?: Req,
    options?: IRequestOptions
  ): Promise<Res> {
    return this.request<Req, Res>(requestMethods.DELETE, url, data, options);
  }
}

/**
 * 请求错误类
 */
export class RequestError extends Error implements IRequestError {
  code: TErrorCode;
  status?: number;
  response?: Response;
  cause?: unknown;

  constructor(error: IRequestError) {
    super(error.message);

    this.name = "RequestError";
    this.code = error.code;
    this.status = error.status;
    this.response = error.response;

    if (error.cause !== undefined) {
      this.cause = error.cause;
    }
  }
}

import type { TErrorCode, TRequestMethod } from "./enum";

export type { TErrorCode, TRequestMethod } from "./enum";

/** 请求头 */
export type RequestHeaders = Record<string, string>;

/** 响应体解析类型 */
export type ResponseType = "json" | "text" | "blob" | "arrayBuffer";

/**
 * 请求配置 —— 用于实例化 Request
 */
export interface IRequestConfig {
  /** base url */
  baseUrl?: string;
  /** 超时时间 (ms) */
  timeout?: number;

  /** 凭据模式 */
  credentials?: RequestCredentials;
  /**
   * 请求头
   *
   * 支持对象，也支持工厂函数（如读取登录态 token）
   */
  headers?: RequestHeaders | (() => RequestHeaders);
  /** 自动重试次数（仅超时 / 网络错误会重试） */
  retry?: number;
  /** 重试间隔 (ms) */
  retryDelay?: number;
  /** 请求拦截器 */
  requestInterceptor?: IRequestInterceptor;
  /** 响应拦截器 */
  responseInterceptor?: IResponseInterceptor;
  /** 全局错误处理（最终错误只会回调一次） */
  errorHandler?: IErrorHandler;
}

/**
 * 单次请求选项 —— 可覆盖实例配置
 */
export interface IRequestOptions {
  /** 本次请求头（与默认头合并） */
  headers?: RequestHeaders;
  /** 本次超时时间，覆盖实例配置 */
  timeout?: number;
  /** 凭据模式 */
  credentials?: RequestCredentials;
  /** 请求模式 */
  mode?: RequestMode;
  /** 缓存策略 */
  cache?: RequestCache;
  /** 重定向策略 */
  redirect?: RequestRedirect;
  /** 外部取消信号（支持调用方主动取消） */
  signal?: AbortSignal;
  /** 响应体解析类型，默认按 Content-Type 自动识别 json / text */
  responseType?: ResponseType;
  /** 本次请求重试次数 */
  retry?: number;
  /** 本次请求重试间隔 */
  retryDelay?: number;
}

/**
 * 请求上下文 —— 进入拦截器
 */
export interface IRequestContext {
  /** 完整 url（已拼接 baseUrl 与查询参数） */
  url: string;
  /** 请求方法 */
  method: TRequestMethod;
  /** 请求头 */
  headers: RequestHeaders;
  /** 请求体（GET 时为 undefined） */
  body?: unknown;
  /** 本次请求选项 */
  options: IRequestOptions;
}

/**
 * 请求拦截器
 */
export interface IRequestInterceptor {
  (config: IRequestContext): Promise<IRequestContext> | IRequestContext;
}

/**
 * 响应上下文
 */
export interface IResponseContext<T = unknown> {
  response: Response;
  data: T;
  config: IRequestContext;
}

/**
 * 响应拦截器 —— 返回的值即为请求方法的最终结果
 */
export interface IResponseInterceptor<T = unknown> {
  (ctx: IResponseContext<T>): Promise<T> | T;
}

/**
 * 请求错误
 */
export interface IRequestError {
  /** 错误码 */
  code: TErrorCode;
  /** 错误信息 */
  message: string;
  /** HTTP 状态码 */
  status?: number;
  /** 响应对象 */
  response?: Response;
  /** 原始错误 */
  cause?: unknown;
}

/**
 * 全局错误处理函数
 */
export interface IErrorHandler {
  (error: IRequestError): void | Promise<void>;
}

/**

export interface IRequestDefaultConfig {

  headers: RequestHeaders | (() => RequestHeaders);

  /** 重试次数 */
  retry: number;
  /** 重试间隔 (ms) */
  retryDelay: number;
}

使用说明（摘要） ​

实例化 Request（默认配置 + 拦截器） ​

ts
import { Request } from "./core";

const http = new Request({
  baseUrl: "192.168.1.1:9090",
  timeout: 10_000,
  credentials: "same-origin",
  headers: { "Content-Type": "application/json" },
  retry: 2,
  retryDelay: 500,

  /**
   * 请求拦截器：可注入 token、改写 url / headers
   */
  requestInterceptor(config) {
    config.headers.Authorization = `Bearer ${getToken()}`;
    return config;
  },

  /**
   * 响应拦截器：返回值即请求方法的最终结果
   */
  responseInterceptor(ctx) {
    return ctx.data;
  },

  /**
   * 全局错误处理（最终错误只会回调一次）
   */
  errorHandler(error) {
    console.error(`[${error.code}]`, error.message, error.status ?? "");
  },
});

注意事项 ​

  • Request 实例持有默认配置；单次请求 options 可覆盖（headers / timeout / credentials / retry / retryDelay / responseType / signal ...）
  • 请求拦截器返回新的 IRequestContext；响应拦截器返回值即最终结果，未设置时返回解析后的响应体
  • 只有 TIMEOUT / NETWORK_ERROR 会触发自动重试，最多 retry 次；重试间隔可取消
  • GET 的 data 作为查询参数拼接，其余方法作为请求体（普通对象 / 数组 JSON 序列化，二进制原样传递）
  • 外部 signal 取消 → 抛 CANCELLED；内部超时 → 抛 TIMEOUT；非 2xx → 抛 HTTP_ERROR（优先取响应体 message）
  • resolveHaBaseUrl() 读 VITE_APP_HA_BASE_URL：绝对 URL 原样 / 相对前缀拼 origin / 空取 location.origin
  • 业务侧一般通过 src/http/*（如 auth.ts）导出具名实例，不要到处 new Request
