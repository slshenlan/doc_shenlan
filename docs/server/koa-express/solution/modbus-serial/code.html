另有 demo.ts（调用示例，非插件实现）与 readme.md / desc.md / code.md（文档），不参与运行。

文件 / 目录说明
index.ts导出 ModbusPlugin、EnumClientState / EnumModbusOptionType、类型
types.ts对外类型；运行时内部结构下沉到各模块的 types.ts
config/只负责从 coreConfigEnv 取值，插件层不做默认值兜底
modules/plugin/只做编排：持有运行时与上下文，串起连接 / 采集 / 重连
modules/client/运行时工厂、generation 代次工具、对外状态快照
modules/connection/connectTCP、safeCloseClient（含兜底解锁）
modules/scheduler/定时器只入队，slQueue 串行消费；背压与陈旧结果拦截
modules/reconnect/安全关闭 → 延迟重连
modules/methods/功能码到 modbus-serial 读方法的映射，集中处理不支持的功能码
modules/errors/关键字匹配判定连接类错误，决定是否重连
modules/logger/统一 logger.modbus；getClientLabel 统一日志前缀
modules/constants/枚举 + 与实现细节强相关的内部常量

ts
/*
 * Modbus Serial 插件 — plugins/serialport
 *
 * 基于 modbus-serial 封装的 Modbus TCP 采集插件：多设备、串行队列、断线自动重连。
 * 连接参数默认取环境配置（.env 中「Modbus TCP 配置」段），调用方只需声明 clients 与回调。
 *
 * @example
 * ```ts
 * import { EnumModbusOptionType, ModbusPlugin } from './plugins/serialport';
 *
 * const modbus = new ModbusPlugin({
 *   clients: [
 *     {
 *       id: 1,
 *       slave: 1,
 *       readConfigs: [
 *         { type: EnumModbusOptionType.READ_HOLDING_REGISTERS, start: 0, range: 2 }
 *       ],
 *       onTaskSuccess: (result, config) => {
 *         logger.modbus.info('读取成功', { result, config });
 *       }
 *     }
 *   ],
 *   onConnect: (id, slave) => {
 *     logger.modbus.info(`设备 ID=${id} (slave=${slave}) 已连接`);
 *   }
 * });
 *
 * modbus.init();
 * ```
 *
 * 需要连接非默认设备时，按需覆盖连接参数：
 * ```ts
 * const modbus = new ModbusPlugin({
 *   serverIp: '192.168.1.20',
 *   interval: 0, // 只读取一次
 *   clients: [...]
 * });
 * ```
 */

export type {
  IClientOption,
  IClientSnapshot,
  IModbusPluginHandlers,
  IModbusPluginOptions,
  IReadConfig,
  ITaskAddressConfig,
} from "./types";

export { EnumClientState, EnumModbusOptionType } from "./modules/constants";

export { ModbusPlugin } from "./modules/plugin";

/*
 * Modbus 插件对外类型
 * 仅声明调用方需要感知的类型；运行时内部结构见 modules/<模块>/types.ts
 */

import type { IModbusPluginEnvConfig } from "./config/types";
import type {
  EnumClientState,
  EnumModbusOptionType,
} from "./modules/constants";

/** 寄存器读取配置 */
export interface IReadConfig {
  /** 操作类型（功能码） */
  type: EnumModbusOptionType;
  /** 寄存器起始地址 */
  start: number;
  /** 读取寄存器数量 */
  range: number;
}

/** 任务回调回传的地址信息 */
export interface ITaskAddressConfig {
  /** 寄存器起始地址 */
  start: number;
  /** 读取寄存器数量 */
  range: number;
}

/** 单个客户端（设备）配置，每个客户端对应一条独立连接与任务队列 */
export interface IClientOption {
  /** 客户端 ID，用于业务侧区分设备，同时作为日志标识 */
  id: number;
  /** 设备地址（slave），一台设备默认为 1 */
  slave: number;
  /** 读取数据配置，每轮采集按顺序全部入队 */
  readConfigs: IReadConfig[];
  /** 单个任务成功回调（设备维度），与 onQueueSuccess 同时触发 */
  onTaskSuccess?: (result: unknown, config: ITaskAddressConfig) => void;
  /** 单个任务失败回调（设备维度），与 onQueueError 同时触发 */
  onTaskError?: (err: unknown, config: ITaskAddressConfig) => void;
}

/** 客户端运行状态快照（对外只读视图） */
export interface IClientSnapshot {
  /** 客户端 ID */
  id: number;
  /** 设备地址 */
  slave: number;
  /** 当前连接状态 */
  status: EnumClientState;
  /** 队列中待执行的任务数 */
  pendingTasks: number;
  /** 是否有任务正在执行 */
  taskExecuting: boolean;
}

/** 插件级事件回调，所有客户端共用 */
export interface IModbusPluginHandlers {
  /** 任一设备连接成功 */
  onConnect?: (clientId: number, slave: number) => void;
  /** 任一设备连接关闭（含主动关闭与断线） */
  onDisconnect?: (clientId: number, slave: number) => void;
  /** 队列任务成功的统一回调 */
  onQueueSuccess?: (
    clientId: number,
    slave: number,
    data: unknown,
    config: IReadConfig
  ) => void;
  /** 队列任务失败的统一回调 */
  onQueueError?: (
    clientId: number,
    slave: number,
    error: unknown,
    config: IReadConfig
  ) => void;
}

/**
 * Modbus 插件配置（对外）
 *
 * 只有 `clients` 必填；连接类参数默认由插件内部从 coreConfigEnv 读取，
 * 需要连接多台不同设备时再按需覆盖。
 */
export interface IModbusPluginOptions
  extends Partial<IModbusPluginEnvConfig>,
    IModbusPluginHandlers {
  /** 客户端配置 */
  clients: IClientOption[];
}

/** 插件内部使用的完整配置（连接参数已按环境配置补全） */
export interface IModbusPluginResolvedOptions
  extends IModbusPluginEnvConfig,
    IModbusPluginHandlers {
  /** 客户端配置 */
  clients: IClientOption[];
}

/*
 * Modbus 插件配置
 *
 * 取值直接来自全局环境配置 coreConfigEnv，插件层不再做默认值兜底；
 * 各项默认值统一维护在 .env.* 中（见「Modbus TCP 配置」段）。
 *
 * @example
 * ```ts
 * const modbus = new ModbusPlugin({
 *   ...resolveModbusPluginConfig(),
 *   clients: [{ id: 1, slave: 1, readConfigs: [...] }]
 * });
 * ```
 */

import type { IModbusPluginEnvConfig } from "./types";

import { coreConfigEnv } from "../../../core/config/env";

export type { IModbusPluginEnvConfig } from "./types";

/**
 * 解析 Modbus 连接配置
 * @param overrides 覆盖配置（优先级高于环境变量）
 * @returns 连接配置（不含 clients）
 */
export function resolveModbusPluginConfig(
  overrides: Partial<IModbusPluginEnvConfig> = {}
): IModbusPluginEnvConfig {
  const env = coreConfigEnv;

  return {
    serverIp: env.ENV_MODBUS_TCP_SERVER_IP,
    serverPort: env.ENV_MODBUS_TCP_SERVER_IP_PORT,
    timeout: env.ENV_MODBUS_TCP_TIMEOUT,
    reconnectDelay: env.ENV_MODBUS_TCP_RETRYDELAY,
    interval: env.ENV_MODBUS_TCP_INTERVAL,
    maxPendingRounds: env.ENV_MODBUS_TCP_MAX_PENDING_ROUNDS,
    ...overrides,
  };
}

ts
/*
 * ModbusPlugin 本体
 * 只做编排：持有客户端运行时与上下文，把连接 / 采集 / 重连三个环节串成闭环
 */

import type {
  IClientSnapshot,
  IModbusPluginOptions,
  IModbusPluginResolvedOptions,
} from "../../types";
import type { IClientRuntime, IModbusRuntimeContext } from "../client/types";

import {
  bumpGeneration,
  createClientRuntime,
  toClientSnapshot,
} from "../client";
import { connectClient, safeCloseClient } from "../connection";
import { EnumClientState } from "../constants";
import { modbusLog } from "../logger";
import { clearReconnectTimer, scheduleReconnect } from "../reconnect";
import { clearClientTimer, startClientTimer } from "../scheduler";
import { resolveModbusPluginOptions } from "./options";

/**
 * Modbus TCP 插件
 *
 * 设计要点：
 * 1. 每个设备独立持有 ModbusRTU + slQueue + 连接状态机，互不影响
 * 2. 定时器只负责入队，队列串行消费，保证同一连接一问一答
 * 3. 连接失败或运行中出现连接类错误，统一走「安全关闭 → 延迟重连」
 * 4. 重连 / 销毁通过递增运行代次，丢弃飞行中任务的陈旧回调
 *
 * 连接参数（IP / 端口 / 超时 / 重连延迟 / 采集间隔）默认取环境配置，无需手动传入。
 *
 * @example
 * ```ts
 * const modbus = new ModbusPlugin({
 *   clients: [{ id: 1, slave: 1, readConfigs: [...] }]
 * });
 *
 * modbus.init();
 * // 进程退出前
 * modbus.destroy();
 * ```
 */
export class ModbusPlugin {
  private readonly options: IModbusPluginResolvedOptions;
  private readonly clients: IClientRuntime[];
  private readonly context: IModbusRuntimeContext;
  private destroyed = false;

  /**
   * @param options 插件配置；除 `clients` 外均为可选
   *
   * @param options.serverIp modbus 服务器 IP，缺省取 `ENV_MODBUS_TCP_SERVER_IP`
   * @param options.serverPort modbus 服务器端口，缺省取 `ENV_MODBUS_TCP_SERVER_IP_PORT`
   * @param options.timeout 单次请求超时 ms，缺省取 `ENV_MODBUS_TCP_TIMEOUT`
   * @param options.reconnectDelay 断线后延迟多久重连 ms，缺省取 `ENV_MODBUS_TCP_RETRYDELAY`
   * @param options.interval 定时读取间隔 ms，缺省取 `ENV_MODBUS_TCP_INTERVAL`；
   *   `> 0` 表示连接成功后立即读一次再按间隔循环，`0` 表示只读一次
   * @param options.maxPendingRounds 队列积压上限（单位：轮，一轮 = readConfigs.length 个任务），
   *   缺省取 `ENV_MODBUS_TCP_MAX_PENDING_ROUNDS`；待执行任务数达到上限时跳过本轮入队
   *
   * @param options.clients 设备列表，每个元素对应一条独立的 TCP 连接与任务队列
   * @param options.clients[].id 客户端 ID，仅用于业务侧区分设备与日志标识
   * @param options.clients[].slave 从站地址，一台设备默认为 1
   * @param options.clients[].readConfigs 该设备的寄存器读取配置，每轮采集按顺序全部入队
   * @param options.clients[].readConfigs[].type 操作类型（功能码），取值见 `EnumModbusOptionType`
   * @param options.clients[].readConfigs[].start 寄存器起始地址
   * @param options.clients[].readConfigs[].range 读取寄存器数量
   * @param options.clients[].onTaskSuccess 该设备单条读取成功回调，入参为 `(result, { start, range })`
   * @param options.clients[].onTaskError 该设备单条读取失败回调，入参为 `(err, { start, range })`
   *
   * @param options.onConnect 任一设备连接成功，入参为 `(clientId, slave)`
   * @param options.onDisconnect 任一设备连接关闭，入参为 `(clientId, slave)`
   * @param options.onQueueSuccess 队列任务成功的统一回调，入参为 `(clientId, slave, data, config)`
   * @param options.onQueueError 队列任务失败的统一回调，入参为 `(clientId, slave, error, config)`
   *
   * @example
   * ```ts
   * const modbus = new ModbusPlugin({
   *   // 连接参数按需覆盖，不写则全部走 .env 的「Modbus TCP 配置」段
   *   serverIp: '192.168.1.137',
   *   interval: 5000,
   *   clients: [
   *     {
   *       id: 1,
   *       slave: 1,
   *       readConfigs: [
   *         { type: EnumModbusOptionType.READ_HOLDING_REGISTERS, start: 0, range: 2 }
   *       ],
   *       onTaskSuccess: (result, config) => {
   *         logger.modbus.info('读取成功', { result, config });
   *       },
   *       onTaskError: (error, config) => {
   *         logger.modbus.error('读取失败', { error, config });
   *       }
   *     }
   *   ],
   *   onConnect: (id, slave) => {
   *     logger.modbus.info(`设备 ID=${id} (slave=${slave}) 已连接`);
   *   },
   *   onDisconnect: (id, slave) => {
   *     logger.modbus.info(`设备 ID=${id} (slave=${slave}) 已断开`);
   *   },
   *   onQueueSuccess: (id, slave, data, config) => {
   *     logger.modbus.debug('队列任务成功', { id, slave, data, config });
   *   },
   *   onQueueError: (id, slave, error, config) => {
   *     logger.modbus.warn('队列任务失败', { id, slave, error, config });
   *   }
   * });
   * ```
   */
  constructor(options: IModbusPluginOptions) {
    this.options = resolveModbusPluginOptions(options);
    this.context = {
      options: this.options,
      isActive: () => {
        return !this.destroyed;
      },
    };
    this.clients = this.options.clients.map((option) => {
      return createClientRuntime(option);
    });
  }

  /**
   * 初始化所有客户端并发起连接
   * @throws 插件已销毁时抛出
   */
  public init(): void {
    if (this.destroyed) {
      throw new Error("ModbusPlugin has been destroyed");
    }

    modbusLog("初始化 modbus 客户端", { total: this.clients.length });
    this.clients.forEach((clientObj) => {
      this.connect(clientObj);
    });
  }

  /**
   * 销毁插件：停止定时器、清空队列、关闭连接，并丢弃飞行中任务的回调
   */
  public destroy(): void {
    if (this.destroyed) {
      return;
    }

    this.destroyed = true;
    modbusLog("销毁 modbus 客户端", { total: this.clients.length });

    this.clients.forEach((clientObj) => {
      bumpGeneration(clientObj);
      clearReconnectTimer(clientObj);
      clearClientTimer(clientObj);
      clientObj.status = EnumClientState.DISCONNECTED;
      safeCloseClient(clientObj, this.context);
    });
  }

  /**
   * 获取所有客户端的运行状态快照
   * @returns 客户端快照列表
   */
  public getClients(): IClientSnapshot[] {
    return this.clients.map((clientObj) => {
      return toClientSnapshot(clientObj);
    });
  }

  /**
   * 发起连接
   * @param clientObj 客户端运行时
   */
  private connect(clientObj: IClientRuntime): void {
    if (this.destroyed) {
      return;
    }

    connectClient(clientObj, {
      context: this.context,
      scheduleReconnect: (target) => {
        this.handleReconnect(target);
      },
      startTimer: (target) => {
        this.handleStartTimer(target);
      },
    });
  }

  /**
   * 调度重连
   * @param clientObj 客户端运行时
   */
  private handleReconnect(clientObj: IClientRuntime): void {
    if (this.destroyed) {
      return;
    }

    scheduleReconnect(clientObj, {
      context: this.context,
      connect: (target) => {
        this.connect(target);
      },
    });
  }

  /**
   * 启动定时采集
   * @param clientObj 客户端运行时
   */
  private handleStartTimer(clientObj: IClientRuntime): void {
    if (this.destroyed) {
      return;
    }

    startClientTimer(clientObj, {
      context: this.context,
      scheduleReconnect: (target) => {
        this.handleReconnect(target);
      },
    });
  }
}

/**

/**
 * 解析队列任务附带数据
 * @param extra slQueue 回传的 extraData
 * @returns 合法则返回数据，否则返回 null

  if (!extra || typeof extra !== "object") {
    return null;

  const data = extra as Partial<ITaskExtraData>;
  if (typeof data.generation !== "number" || !data.data) {
    return null;

  return data as ITaskExtraData;

    slave: clientObj.slave,

/*
 * TCP 连接与安全关闭
 */

import type { IClientRuntime, IModbusRuntimeContext } from "../client/types";
import type { IConnectionDeps } from "./types";

import { EnumClientState, MODBUS_CLOSE_TIMING } from "../constants";
import {
  getClientLabel,
  modbusLog,
  modbusLogDebug,
  modbusLogError,
} from "../logger";

/**
 * 连接 TCP，成功后启动定时采集，失败则进入重连调度
 * @param clientObj 客户端运行时
 * @param deps 连接依赖
 */
export function connectClient(
  clientObj: IClientRuntime,
  deps: IConnectionDeps
): void {
  const label = getClientLabel(clientObj);
  const { context, scheduleReconnect, startTimer } = deps;
  const { options } = context;

  if (
    clientObj.status === EnumClientState.CONNECTING ||
    clientObj.status === EnumClientState.CONNECTED
  ) {
    modbusLogDebug(`${label} 已在连接中或已连接，跳过本次 connect`);
    return;
  }

  clientObj.status = EnumClientState.CONNECTING;

  const { client, id, slave } = clientObj;

  modbusLog(`${label} 正在连接...`, {
    serverIp: options.serverIp,
    serverPort: options.serverPort,
    timeout: options.timeout,
  });

  client.setTimeout(options.timeout);

  client
    .connectTCP(options.serverIp, { port: options.serverPort })
    .then(() => {
      // 连接期间插件可能已被销毁，此时不再对外派发
      if (!context.isActive()) {
        modbusLog(`${label} 插件已销毁，丢弃本次连接结果`);
        return;
      }

      client.setID(slave);
      clientObj.status = EnumClientState.CONNECTED;

      modbusLog(`${label} 连接成功`);
      options.onConnect?.(id, slave);
      startTimer(clientObj);
    })
    .catch((error: unknown) => {
      clientObj.status = EnumClientState.DISCONNECTED;

      if (!context.isActive()) {
        modbusLog(`${label} 插件已销毁，放弃重连`);
        return;
      }

      modbusLogError(`${label} 连接失败，进入重连调度`, error);
      scheduleReconnect(clientObj);
    });
}

/**
 * 计算关闭兜底解锁超时
 *
 * 必须早于下一次重连发起，否则重连时 isClosing 仍为 true，本轮关闭会被静默跳过。
 *
 * @param context 运行时上下文
 * @returns 兜底超时 ms
 */
function resolveCloseFallbackTimeout(context: IModbusRuntimeContext): number {
  const { FALLBACK_TIMEOUT, UNLOCK_DELAY } = MODBUS_CLOSE_TIMING;

  return Math.min(
    FALLBACK_TIMEOUT,
    Math.max(UNLOCK_DELAY, context.options.reconnectDelay / 2)
  );
}

/**
 * 安全关闭 TCP 连接
 *
 * close 回调不保证一定触发（socket 从未成功打开时就不会回调），因此额外挂一个兜底定时器，
 * 超时后强制解除 isClosing，避免该标记被永久锁死导致后续关闭全部静默跳过。
 *
 * @param clientObj 客户端运行时
 * @param context 运行时上下文
 */
export function safeCloseClient(
  clientObj: IClientRuntime,
  context: IModbusRuntimeContext
): void {
  const label = getClientLabel(clientObj);

  if (clientObj.isClosing) {
    modbusLogDebug(`${label} 正在关闭中，跳过重复关闭`);
    return;
  }

  clientObj.isClosing = true;
  modbusLogDebug(`${label} 正在安全关闭...`);

  const unlock = (): void => {
    if (clientObj.closeTimer) {
      clearTimeout(clientObj.closeTimer);
      clientObj.closeTimer = undefined;
    }
    clientObj.isClosing = false;
  };

  clientObj.closeTimer = setTimeout(() => {
    clientObj.closeTimer = undefined;
    clientObj.isClosing = false;
    // modbusLogDebug(`${label} close 回调未触发，已按兜底超时解除关闭标记`);
  }, resolveCloseFallbackTimeout(context));

  try {
    clientObj.client.close(() => {
      modbusLog(`${label} TCP 连接已关闭`);
      context.options.onDisconnect?.(clientObj.id, clientObj.slave);

      // 稍作延迟再解锁，避免关闭尚未完全落地就发起新连接
      setTimeout(() => {
        unlock();
      }, MODBUS_CLOSE_TIMING.UNLOCK_DELAY);
    });
  } catch (err) {
    unlock();
    modbusLogError(`${label} safeClose 异常`, err);
  }
}

/*
 * 定时调度与队列任务入队
 *
 * 定时器只负责入队，实际读取由 slQueue 串行消费，保证同一连接上一问一答。
 */

import { safeSerializeError } from "cy-serialize-error";

import type { IReadConfig } from "../../types";
import type { IClientRuntime, ITaskExtraData } from "../client/types";
import type { ISchedulerDeps } from "./types";

import { isTaskExpired, parseTaskExtra } from "../client";
import { EnumClientState } from "../constants";
import { isConnectionError } from "../errors";
import {
  formatModbusOptionType,
  getClientLabel,
  modbusLog,
  modbusLogDebug,
  modbusLogError,
  modbusLogWarn,
} from "../logger";
import { resolveModbusReadMethod } from "../methods";

/**
 * 清理定时器并清空待执行队列
 * @param clientObj 客户端运行时
 */
export function clearClientTimer(clientObj: IClientRuntime): void {
  if (clientObj.timer) {
    clearInterval(clientObj.timer);
    clientObj.timer = undefined;
  }
  clientObj.queue.clear();
}

/**
 * 判断队列是否已积压
 *
 * 一轮任务数 = readConfigs.length，待执行数达到 `一轮 × maxPendingRounds` 时视为消费跟不上生产。
 *
 * @param clientObj 客户端运行时
 * @param deps 调度依赖
 * @returns 已积压返回 true
 */
function isQueueBacklogged(
  clientObj: IClientRuntime,
  deps: ISchedulerDeps
): boolean {
  const roundSize = clientObj.readConfigs.length;
  if (roundSize === 0) {
    return false;
  }

  const limit = Math.max(
    roundSize,
    roundSize * deps.context.options.maxPendingRounds
  );

  return clientObj.queue.length >= limit;
}

/**
 * 将当前 readConfigs 全部入队执行一次
 * @param clientObj 客户端运行时
 * @param deps 调度依赖
 */
export function appendTasksOnce(
  clientObj: IClientRuntime,
  deps: ISchedulerDeps
): void {
  const { context, scheduleReconnect } = deps;
  const label = getClientLabel(clientObj);

  clientObj.readConfigs.forEach((data: IReadConfig) => {
    const extraData: ITaskExtraData = {
      id: clientObj.id,
      slave: clientObj.slave,
      generation: clientObj.generation,
      data,
    };

    clientObj.queue.push({
      taskParams: data,
      extraData,
      task: async ({ type, start, range }: IReadConfig) => {
        const method = resolveModbusReadMethod(clientObj.client, type);
        return await method(start, range);
      },
      onSuccess: (result, extra) => {
        const taskExtra = parseTaskExtra(extra);
        if (!taskExtra || !isTaskEffective(clientObj, taskExtra, deps)) {
          return;
        }

        clientObj.onTaskSuccess?.(result, {
          start: taskExtra.data.start,
          range: taskExtra.data.range,
        });
        context.options.onQueueSuccess?.(
          taskExtra.id,
          taskExtra.slave,
          result,
          taskExtra.data
        );
      },
      onError: (err, extra) => {
        const taskExtra = parseTaskExtra(extra);
        if (!taskExtra || !isTaskEffective(clientObj, taskExtra, deps)) {
          return;
        }

        const serializedError = safeSerializeError(err);
        modbusLogError(`${label} 读取寄存器失败`, {
          params: {
            ...taskExtra.data,
            type: formatModbusOptionType(taskExtra.data.type),
          },
          error: serializedError,
        });

        clientObj.onTaskError?.(err, {
          start: taskExtra.data.start,
          range: taskExtra.data.range,
        });
        context.options.onQueueError?.(
          taskExtra.id,
          taskExtra.slave,
          err,
          taskExtra.data
        );

        // 运行中掉线 → 自动重连；非连接类错误（如非法功能码）不触发
        if (!isConnectionError(err)) {
          return;
        }

        if (clientObj.status !== EnumClientState.CONNECTED) {
          modbusLog(`${label} 已在连接流程中，忽略重复重连`);
          return;
        }

        modbusLog(`${label} 运行中掉线，准备重连...`);
        scheduleReconnect(clientObj);
      },
    });
  });
}

/**
 * 启动定时采集
 *
 * - `interval > 0`：连接成功后立即读取一次，之后按间隔循环
 * - `interval <= 0` 或未配置：只读取一次
 *
 * @param clientObj 客户端运行时
 * @param deps 调度依赖
 */
export function startClientTimer(
  clientObj: IClientRuntime,
  deps: ISchedulerDeps
): void {
  clearClientTimer(clientObj);

  const label = getClientLabel(clientObj);
  const { interval } = deps.context.options;

  appendTasksOnce(clientObj, deps);

  if (!Number.isFinite(interval) || interval <= 0) {
    modbusLog(`${label} interval 为 0，本次连接只读取一次`);
    return;
  }

  clientObj.timer = setInterval(() => {
    if (!deps.context.isActive()) {
      return;
    }

    if (isQueueBacklogged(clientObj, deps)) {
      modbusLogWarn(`${label} 队列积压，跳过本轮采集`, {
        pendingTasks: clientObj.queue.length,
        roundSize: clientObj.readConfigs.length,
      });
      return;
    }

    appendTasksOnce(clientObj, deps);
  }, interval);
}

/**
 * 判断任务回调是否仍然有效（插件存活且未跨越重连）
 * @param clientObj 客户端运行时
 * @param extra 任务附带数据
 * @param deps 调度依赖
 * @returns 有效返回 true
 */
function isTaskEffective(
  clientObj: IClientRuntime,
  extra: ITaskExtraData,
  deps: ISchedulerDeps
): boolean {
  if (!deps.context.isActive()) {
    return false;
  }

  if (isTaskExpired(clientObj, extra)) {
    modbusLogDebug(`${getClientLabel(clientObj)} 丢弃陈旧任务结果`, {
      taskGeneration: extra.generation,
      currentGeneration: clientObj.generation,
    });
    return false;
  }

  return true;
}

/*
 * 断线重连调度
 */

import type { IClientRuntime } from "../client/types";
import type { IReconnectDeps } from "./types";

import { bumpGeneration } from "../client";
import { safeCloseClient } from "../connection";
import { EnumClientState } from "../constants";
import { getClientLabel, modbusLog } from "../logger";
import { clearClientTimer } from "../scheduler";

/**
 * 清理重连定时器
 * @param clientObj 客户端运行时
 */
export function clearReconnectTimer(clientObj: IClientRuntime): void {
  if (clientObj.reconnectTimer) {
    clearTimeout(clientObj.reconnectTimer);
    clientObj.reconnectTimer = undefined;
  }
}

/**
 * 断开当前连接并延迟重连
 * @param clientObj 客户端运行时
 * @param deps 重连依赖
 */
export function scheduleReconnect(
  clientObj: IClientRuntime,
  deps: IReconnectDeps
): void {
  const label = getClientLabel(clientObj);
  const { context, connect } = deps;

  if (
    clientObj.status === EnumClientState.CONNECTING ||
    clientObj.status === EnumClientState.RECONNECTING
  ) {
    modbusLog(`${label} 当前正在连接/重连中，跳过本次重连调度`);
    return;
  }

  clientObj.status = EnumClientState.RECONNECTING;

  // 代次递增：此前入队任务的回调将被判定为陈旧结果丢弃
  bumpGeneration(clientObj);
  clearClientTimer(clientObj);
  clearReconnectTimer(clientObj);
  safeCloseClient(clientObj, context);

  const delay = context.options.reconnectDelay;
  modbusLog(`${label} ${delay}ms 后开始重连`);

  clientObj.reconnectTimer = setTimeout(() => {
    clientObj.reconnectTimer = undefined;

    if (!context.isActive()) {
      modbusLog(`${label} 插件已销毁，取消重连`);
      return;
    }

    if (clientObj.status !== EnumClientState.RECONNECTING) {
      modbusLog(`${label} 状态已变更为 ${clientObj.status}，停止重连`);
      return;
    }

    clientObj.status = EnumClientState.DISCONNECTED;
    connect(clientObj);
  }, delay);
}

/*
 * Modbus 功能码 → 读方法映射
 */

import type ModbusRTU from "modbus-serial";

import type { TModbusReadMethod } from "./types";

import { EnumModbusOptionType } from "../constants";

/**
 * 按功能码获取绑定后的读方法
 * @param client ModbusRTU 实例
 * @param type 操作类型
 * @returns 绑定 this 后的读方法
 * @throws 传入不支持的功能码时抛出
 */
export function resolveModbusReadMethod(
  client: ModbusRTU,
  type: EnumModbusOptionType
): TModbusReadMethod {
  switch (type) {
    case EnumModbusOptionType.READ_COILS:
      return client.readCoils.bind(client);
    case EnumModbusOptionType.READ_DISCRETE_INPUTS:
      return client.readDiscreteInputs.bind(client);
    case EnumModbusOptionType.READ_HOLDING_REGISTERS:
      return client.readHoldingRegisters.bind(client);
    case EnumModbusOptionType.READ_INPUT_REGISTERS:
      return client.readInputRegisters.bind(client);
    default:
      throw new Error(`Unsupported modbus operation type: ${type}`);
  }
}

/*
 * 连接类错误判断
 */

import { CONNECTION_ERROR_KEYWORDS } from "../constants";

/**
 * 判断是否为连接 / 超时类错误（决定是否触发自动重连）
 * @param err 错误对象
 * @returns 是连接类错误返回 true
 */
export function isConnectionError(err: unknown): boolean {
  if (!err || typeof err !== "object") {
    return false;
  }

  const error = err as { name?: unknown; message?: unknown; errno?: unknown };
  const fields = [error.name, error.message, error.errno].map((item) => {
    return String(item ?? "").toLowerCase();
  });

  return CONNECTION_ERROR_KEYWORDS.some((keyword) => {
    return fields.some((field) => {
      return field.includes(keyword);
    });
  });
}

 * 统一走 logger.modbus，命令行输出由 ENV_LOGGER_ENABLE_MODBUS_STDOUT 控制，

import type { IClientLabelSource } from "./types";

import { EnumModbusOptionType } from "../constants";

/**
 * 生成客户端日志标识
 * @param client 含 id / slave 的对象
 * @returns 形如 `设备ID=1_slave=1 :` 的前缀
 */
export function getClientLabel(client: IClientLabelSource): string {
  return `设备ID=${client.id}_slave=${client.slave} :`;
}

/**
 * 把功能码转成便于阅读的枚举名
 *
 * 依赖数字枚举的反向映射，`3` 还原为 `READ_HOLDING_REGISTERS`。
 *
 * @param type 操作类型（功能码）
 * @returns 枚举名；不在枚举内时回退为 `UNKNOWN(码值)`
 */
export function formatModbusOptionType(type: EnumModbusOptionType): string {
  const typeName: string | undefined = EnumModbusOptionType[type];

  return typeName ?? `UNKNOWN(${type})`;
}

/**
 * 信息日志
 * @param title 日志标题
 * @param detail 日志详情（可选）
 */
export function modbusLog(title: string, detail?: unknown): void {
  if (detail === undefined) {
    logger.modbus.info(title);
    return;
  }
  logger.modbus.info(title, detail);
}

 * 调试日志

export function modbusLogDebug(title: string, detail?: unknown): void {
  if (detail === undefined) {
    logger.modbus.debug(title);

  if (detail === undefined) {
    logger.modbus.warn(title);

  logger.modbus.warn(title, detail);

 * 错误日志

使用说明（摘要） ​

启动采集 ​

ts
import { EnumModbusOptionType, ModbusPlugin } from "./plugins/serialport";

const modbus = new ModbusPlugin({
  clients: [
    {
      id: 1,
      slave: 1,
      readConfigs: [
        {
          type: EnumModbusOptionType.READ_HOLDING_REGISTERS,
          start: 0,
          range: 2,
        },
      ],
      onTaskSuccess: (result, config) => {
        logger.modbus.info("读取成功", { result, config });
      },
      onTaskError: (error, config) => {
        logger.modbus.error("读取失败", { error, config });
      },
    },
  ],
  onConnect: (id, slave) => {
    logger.modbus.info(`设备 ID=${id} (slave=${slave}) 已连接`);
  },
});

modbus.init();

注意事项 ​

  • 不是单例，按需 new；一个实例内的 clients 共用同一组连接参数，靠 slave 区分从站
  • destroy() 之后再 init() 会抛 ModbusPlugin has been destroyed
  • interval 为 0 表示只读一次，不建定时器
  • 非连接类错误（非法地址 / 非法功能码）只回调 error，不触发重连
  • onDisconnect 实际语义是「主动关闭成功」；首次连不上、对端主动断开这两种情况不会触发（详见 desc.md）
  • 慢设备优先调大 ENV_MODBUS_TCP_INTERVAL，MAX_PENDING_ROUNDS 只是防止无限积压的保险
  • 日志统一走 logger.modbus，模块内不要再手动 logger.stdout，否则控制台重复打印
  • 写日志前用 formatModbusOptionType 把功能码转成枚举名，避免日志里出现裸的 type: 3
