/*
 * MQTT Server 门面服务
 * 业务侧唯一入口：组装 Manager / Handler / Publisher，对外暴露稳定 API
 */

import type { IMqttServerPluginConfig } from "../config/types";
import type {
  IMqttServerAuthenticateHandler,
  IMqttServerAuthorizeHandler,
  IMqttServerEventHandler,
  IMqttServerEventName,
  IMqttServerMiddleware,
  IMqttServerPayload,
  IMqttServerPublishOptions,
  IMqttServerService,
  IMqttServerStartOptions,
  IMqttServerTopicHandler,
} from "../types";

import { resolveMqttServerConfig } from "../config";
import { MqttServerEvent, mqttServerMsg } from "../constants";
import {
  MqttServerNotEnabledError,
  MqttServerNotRunningError,
} from "../errors";
import { MqttServerMessageHandler } from "../handler";
import { MqttServerManager } from "../manager";
import { MqttServerPublisherService } from "./publisher";

type IInternalDeliver = (packet: any, cb: () => void) => void;

export class MqttServerService implements IMqttServerService {
  private config: IMqttServerPluginConfig;
  private manager: MqttServerManager;
  private publisher: MqttServerPublisherService;
  private messageHandler: MqttServerMessageHandler;
  private started = false;
  private publishUnbind: (() => void) | null = null;
  private internalDelivers = new Map<string, IInternalDeliver>();

  /**
   * 构造门面：解析配置并创建 Manager / Publisher / Handler
   * @param overrides 可选配置覆盖（一般留给测试）
   */
  constructor(overrides: Partial<IMqttServerPluginConfig> = {}) {
    this.config = resolveMqttServerConfig(overrides);
    this.manager = new MqttServerManager(this.config);
    this.publisher = new MqttServerPublisherService(this.manager);
    this.messageHandler = new MqttServerMessageHandler();
  }

  /**
   * 启动 MQTT Server
   * 绑定消息管道后，委托 Manager 创建 Broker 与 Transport
   * @param options.httpServer 传入 Koa http.Server 以启用 WebSocket
   */
  async start(options: IMqttServerStartOptions = {}) {
    if (this.started && this.manager.isRunning()) {
      logger.mqttServer.debug(mqttServerMsg("已启动"));
      return;
    }

    this.config = resolveMqttServerConfig(this.config);
    this.manager.updateConfig(this.config);

    if (!this.config.enable) {
      logger.mqttServer.warn(
        `ENV_MQTT_SERVER_ENABLE=false，${mqttServerMsg("未启动")}`,
      );
      return;
    }

    this.bindMessagePipeline();
    await this.manager.start(options.httpServer);
    this.started = true;
  }

  /**
   * 停止 MQTT Server：解绑管道、清理内部订阅与 Handler，再关 Manager
   */
  async stop() {
    this.publishUnbind?.();
    this.publishUnbind = null;

    // 清理内部订阅
    const broker = this.manager.getBroker();
    if (broker) {
      for (const [topic, deliver] of this.internalDelivers.entries()) {
        await new Promise<void>((resolve) => {
          broker.unsubscribe(topic, deliver, () => {
            resolve();
          });
        });
      }
    }
    this.internalDelivers.clear();
    this.messageHandler.clear();

    await this.manager.stop();
    this.started = false;
  }

  /** 是否正在运行 */
  isRunning() {
    return this.manager.isRunning();
  }

  /** 当前状态机状态 */
  getState() {
    return this.manager.getState();
  }

  /** 健康检查快照 */
  health() {
    return this.manager.health();
  }

  /**
   * Broker 侧发布消息（推给已订阅客户端）
   * @param topic 主题
   * @param payload 字符串 / Buffer / 对象 / 数组
   * @param options qos / retain / dup
   */
  async publish(
    topic: string,
    payload: IMqttServerPayload,
    options?: IMqttServerPublishOptions,
  ) {
    this.assertEnabled();
    return this.publisher.publish(topic, payload, options);
  }

  /**
   * Broker 内部订阅主题，并注册业务 handler
   * 实际业务处理仍走 publish 事件 → Handler 链
   * @returns 异步取消函数
   */
  async subscribe(topic: string | string[], handler: IMqttServerTopicHandler) {
    this.assertEnabled();
    this.assertRunning();

    const broker = this.manager.getBroker()!;
    const topics = Array.isArray(topic) ? topic : [topic];
    const unbinds: Array<() => void> = [];

    for (const item of topics) {
      const off = this.messageHandler.onTopic(item, handler);

      const deliver: IInternalDeliver = (_packet, cb) => {
        // aedes 内部投递回调，业务逻辑走统一 handler 链（由 publish 事件驱动）
        cb();
      };

      await new Promise<void>((resolve) => {
        broker.subscribe(item, deliver, () => {
          this.internalDelivers.set(item, deliver);
          this.manager.increaseSubscribeCount();
          resolve();
        });
      });

      unbinds.push(off);
      logger.mqttServer.info("Broker 内部订阅", { topic: item });
    }

    return async () => {
      for (const item of topics) {
        const deliver = this.internalDelivers.get(item);
        if (deliver) {
          await new Promise<void>((resolve) => {
            broker.unsubscribe(item, deliver, () => {
              resolve();
            });
          });
          this.internalDelivers.delete(item);
        }
      }
      for (const off of unbinds) {
        off();
      }
    };
  }

  /**
   * 注册主题处理器（支持 + / # 通配）
   * @returns 取消注册函数
   */
  onTopic(topic: string, handler: IMqttServerTopicHandler) {
    return this.messageHandler.onTopic(topic, handler);
  }

  /**
   * 注册消息中间件（在主题路由前执行，Koa 风格）
   */
  use(middleware: IMqttServerMiddleware) {
    this.messageHandler.use(middleware);
  }

  /**
   * 设置自定义连接认证（覆盖默认账号密码逻辑）
   */
  setAuthenticate(handler: IMqttServerAuthenticateHandler) {
    this.manager.auth.setAuthenticate(handler);
  }

  /**
   * 设置自定义 publish/subscribe 授权
   */
  setAuthorize(handler: IMqttServerAuthorizeHandler) {
    this.manager.auth.setAuthorize(handler);
  }

  /** 列出在线客户端 */
  listClients() {
    return this.manager.clients.list();
  }

  /**
   * 按 ID 查询客户端信息
   * @param clientId 客户端 ID
   */
  getClient(clientId: string) {
    return this.manager.clients.get(clientId);
  }

  /**
   * 主动断开指定客户端
   * @param clientId 客户端 ID
   */
  async disconnectClient(clientId: string) {
    this.assertRunning();
    const client = this.manager.clients.getRaw(clientId);
    if (!client) {
      return;
    }
    await new Promise<void>((resolve) => {
      client.close(() => {
        resolve();
      });
    });
  }

  /**
   * 监听服务事件
   * @returns 取消监听函数
   */
  on<E extends IMqttServerEventName>(
    event: E,
    handler: IMqttServerEventHandler<E>,
  ) {
    return this.manager.on(event, handler);
  }

  /**
   * 一次性监听服务事件
   * @returns 取消监听函数
   */
  once<E extends IMqttServerEventName>(
    event: E,
    handler: IMqttServerEventHandler<E>,
  ) {
    return this.manager.once(event, handler);
  }

  /** 移除事件监听 */
  off<E extends IMqttServerEventName>(
    event: E,
    handler: IMqttServerEventHandler<E>,
  ) {
    this.manager.off(event, handler);
  }

  /** 获取底层 aedes（慎用） */
  getBroker() {
    return this.manager.getBroker();
  }

  /** 获取 TCP Server（慎用） */
  getTcpServer() {
    return this.manager.getTcpServer();
  }

  /** 获取 HTTP Server（慎用） */
  getHttpServer() {
    return this.manager.getHttpServer();
  }

  /**
   * 将 Manager 的 publish 事件接到 Handler 消息链
   * 形成：设备消息 → aedes → manager.emit → handler.handle
   */
  private bindMessagePipeline() {
    this.publishUnbind?.();
    this.publishUnbind = this.manager.on(MqttServerEvent.Publish, ({ ctx }) => {
      void this.messageHandler.handle(ctx);
    });
  }

  /** 断言插件已启用 */
  private assertEnabled() {
    if (!this.config.enable) {
      throw new MqttServerNotEnabledError();
    }
  }

  /** 断言服务已在运行 */
  private assertRunning() {
    if (!this.manager.isRunning()) {
      throw new MqttServerNotRunningError();
    }
  }
}

let mqttServerSingleton: MqttServerService | null = null;

  mqttServerSingleton = null;

/*
 * Broker 侧发布服务
 * 将业务 payload 编码后交给 aedes.publish 推送给订阅端
 */

import type { IMqttServerPayload, IMqttServerPublishOptions } from "../types";

import { MQTT_SERVER_DEFAULTS, MqttPacketCommand } from "../constants";
import { encodePayload } from "../utils/payload";
import { MqttServerNotRunningError, MqttServerPublishError } from "../errors";
import { MqttServerManager } from "../manager";

export class MqttServerPublisherService {
  constructor(private manager: MqttServerManager) {}

  /**
   * 向指定主题发布消息
   * @param topic MQTT 主题
   * @param payload 字符串 / Buffer / 对象 / 数组（对象会 JSON 序列化）
   * @param options qos / retain / dup，qos 默认 AtMostOnce
   */
  async publish(
    topic: string,
    payload: IMqttServerPayload,
    options: IMqttServerPublishOptions = {},
  ) {
    if (!topic || !topic.trim()) {
      throw new MqttServerPublishError("topic 不能为空");
    }

    const broker = this.manager.getBroker();
    if (!broker || !this.manager.isRunning()) {
      throw new MqttServerNotRunningError();
    }

    const body = encodePayload(payload);
    const qos = options.qos ?? MQTT_SERVER_DEFAULTS.DEFAULT_QOS;
    const retain = options.retain ?? false;

    await new Promise<void>((resolve, reject) => {
      broker.publish(
        {
          cmd: MqttPacketCommand.Publish,
          topic,
          payload: body,
          qos,
          retain,
          dup: options.dup ?? false,
        },
        (error) => {
          if (error) {
            logger.mqttServer.error("Broker 发布失败", {
              topic,
              qos,
              retain,
              error,
            });
            reject(new MqttServerPublishError(error.message, { topic }));
            return;
          }

          this.manager.increasePublishCount();
          logger.mqttServer.info("Broker 发布成功", {
            topic,
            qos,
            retain,
            size: Buffer.isBuffer(body) ? body.length : Buffer.byteLength(body),
          });
          resolve();
        },
      );
    });
  }
}

import type { IMqttServerPluginConfig } from "../config/types";
import type {
  IMqttServerEventHandler,
  IMqttServerEventMap,
  IMqttServerEventName,
  IMqttServerHealth,
  IMqttServerMessageContext,
  IMqttServerState,
} from "../types";

import { MqttServerAuthService } from "../auth";
import { createAedesBroker } from "../broker";
import {
  AedesBrokerEvent,
  MQTT_SYS_TOPIC_PREFIX,
  MqttServerEvent,
  mqttServerMsg,
  MqttServerQoS,
  MqttServerState,
} from "../constants";
import { startTcpTransport, stopTcpTransport } from "../transport/tcp";
import {
  attachWsTransport,
  type IMqttWsAttachment,
} from "../transport/websocket";
import { MqttServerEventBus } from "../utils/event-bus";
import { decodePayload } from "../utils/payload";
import { ClientRegistry } from "./clients";

export class MqttServerManager {
  private broker: Aedes | null = null;
  private tcpServer: NetServer | null = null;
  private httpServer: HttpServer | null = null;
  private wsAttachment: IMqttWsAttachment | null = null;
  private state: IMqttServerState = MqttServerState.Idle;
  private startedAt: number | null = null;
  private lastError: string | null = null;
  private publishCount = 0;
  private subscribeCount = 0;
  private readonly events = new MqttServerEventBus();
  /** 在线客户端注册表 */
  readonly clients = new ClientRegistry();
  /** 认证 / 授权服务 */
  readonly auth: MqttServerAuthService;

  constructor(private config: IMqttServerPluginConfig) {
    this.auth = new MqttServerAuthService(config);
  }

  /** 获取当前配置 */
  getConfig() {
    return this.config;
  }

  /**
   * 热更新配置（启动前可调用）
   * @param config 新配置
   */
  updateConfig(config: IMqttServerPluginConfig) {
    this.config = config;
    this.auth.updateConfig(config);
  }

  /** 获取 aedes 实例 */
  getBroker() {
    return this.broker;
  }

  /** 获取 TCP Server */
  getTcpServer() {
    return this.tcpServer;
  }

  /** 获取挂载了 WS 的 HTTP Server */
  getHttpServer() {
    return this.httpServer;
  }

  /** 获取运行状态 */
  getState() {
    return this.state;
  }

  /** 是否处于可对外服务的 running 状态 */
  isRunning() {
    return (
      this.state === MqttServerState.Running &&
      !!this.broker &&
      !this.broker.closed
    );
  }

  /**
   * 监听内部事件
   * @returns 取消监听函数
   */
  on<E extends IMqttServerEventName>(
    event: E,
    handler: IMqttServerEventHandler<E>,
  ) {
    return this.events.on(event, handler);
  }

  /**
   * 一次性监听内部事件
   * @returns 取消监听函数
   */
  once<E extends IMqttServerEventName>(
    event: E,
    handler: IMqttServerEventHandler<E>,
  ) {
    return this.events.once(event, handler);
  }

  /** 移除事件监听 */
  off<E extends IMqttServerEventName>(
    event: E,
    handler: IMqttServerEventHandler<E>,
  ) {
    this.events.off(event, handler);
  }

  /**
   * 触发内部事件（Emittery 异步派发，fire-and-forget）
   */
  emit<E extends IMqttServerEventName>(
    event: E,
    ...args: IMqttServerEventMap[E] extends undefined
      ? [] | [undefined]
      : [IMqttServerEventMap[E]]
  ) {
    this.events.emit(event, ...(args as any));
  }

  /**
   * 启动 Broker + Transport
   * 顺序：创建 aedes → 绑事件 → TCP → WebSocket
   * @param httpServer Koa 的 http.Server，用于挂载 MQTT over WS；不传则跳过 WS
   */
  async start(httpServer?: HttpServer) {
    if (!this.config.enable) {
      logger.mqttServer.warn(mqttServerMsg("未启用，跳过启动"));
      return;
    }

    if (this.isRunning()) {
      logger.mqttServer.debug(mqttServerMsg("已在运行"));
      return;
    }

    this.setState(MqttServerState.Starting);
    this.events.emit(MqttServerEvent.Starting);
    this.httpServer = httpServer || null;
    this.startedAt = Date.now();

    try {
      this.broker = await createAedesBroker(this.config, this.auth);
      this.bindBrokerEvents(this.broker);

      // 启动 TCP Transport
      if (this.config.tcpEnable) {
        this.tcpServer = await startTcpTransport(this.broker, this.config);
      }

      // 启动 WebSocket Transport
      if (this.config.wsEnable) {
        if (!httpServer) {
          logger.mqttServer.warn(
            mqttServerMsg(
              "已开启 WS，但未传入 httpServer，跳过 WebSocket 挂载",
            ),
          );
        } else {
          this.wsAttachment = attachWsTransport(
            this.broker,
            httpServer,
            this.config,
          );
        }
      }

      this.setState(MqttServerState.Running);
      this.lastError = null;
      this.events.emit(MqttServerEvent.Ready, this.health());
      logger.mqttServer.info(mqttServerMsg("启动完成"), this.health());
    } catch (error) {
      this.lastError = (error as Error).message;
      this.setState(MqttServerState.Error);
      this.events.emit(MqttServerEvent.Error, { error: error as Error });
      throw error;
    }
  }

  /**
   * 停止服务
   * 顺序：关 WS → 关 TCP → close aedes → 清空客户端
   */
  async stop() {
    if (
      this.state === MqttServerState.Idle ||
      this.state === MqttServerState.Stopped
    ) {
      return;
    }

    this.setState(MqttServerState.Stopping);

    this.wsAttachment?.close();
    this.wsAttachment = null;

    await stopTcpTransport(this.tcpServer, this.config.shutdownTimeout);
    this.tcpServer = null;

    if (this.broker) {
      await new Promise<void>((resolve) => {
        const timer = setTimeout(() => {
          logger.mqttServer.warn("Aedes 关闭超时");
          resolve();
        }, this.config.shutdownTimeout);

        this.broker!.close(() => {
          clearTimeout(timer);
          resolve();
        });
      });
      this.broker = null;
    }

    this.clients.clear();
    this.httpServer = null;
    this.setState(MqttServerState.Stopped);
    this.events.emit(MqttServerEvent.Stopped);
    logger.mqttServer.info(mqttServerMsg("已停止"));
  }

  /** 发布计数 +1（门面 publish / 入站消息时调用） */
  increasePublishCount() {
    this.publishCount += 1;
  }

  /** 订阅计数 +1 */
  increaseSubscribeCount() {
    this.subscribeCount += 1;
  }

  /**
   * 汇总运行健康信息（TCP/WS/连接数/计数/uptime）
   */
  health(): IMqttServerHealth {
    return {
      enabled: this.config.enable,
      state: this.state,
      brokerId: this.config.id,
      tcp: {
        enable: this.config.tcpEnable,
        host: this.config.host,
        port: this.config.port,
        listening: !!this.tcpServer?.listening,
      },
      ws: {
        enable: this.config.wsEnable,
        path: this.config.wsPath,
        attached: !!this.wsAttachment,
      },
      connectedClients: this.broker?.connectedClients ?? this.clients.size(),
      publishCount: this.publishCount,
      subscribeCount: this.subscribeCount,
      lastError: this.lastError,
      uptimeMs: this.startedAt ? Date.now() - this.startedAt : 0,
    };
  }

  /**
   * 切换状态机并广播 stateChange
   * @param next 目标状态
   */
  private setState(next: IMqttServerState) {
    if (this.state === next) {
      return;
    }
    const prev = this.state;
    this.state = next;
    logger.mqttServer.debug(mqttServerMsg("状态变更"), {
      from: prev,
      to: next,
    });
    this.events.emit(MqttServerEvent.StateChange, { next, prev });
  }

  /**
   * 绑定 aedes 原生事件：连接/断开/发布/订阅/错误
   * publish 事件会解码 payload 并转成 IMqttServerMessageContext 再向外 emit
   */
  private bindBrokerEvents(broker: Aedes) {
    broker.on(AedesBrokerEvent.Client, (client: Client) => {
      this.clients.add(client);
      logger.mqttServer.info("客户端连接", {
        clientId: client.id,
        connectedClients: broker.connectedClients,
      });
      this.events.emit(MqttServerEvent.Client, client);
    });

    broker.on(AedesBrokerEvent.ClientReady, (client: Client) => {
      logger.mqttServer.debug("客户端就绪", { clientId: client.id });
      this.events.emit(MqttServerEvent.ClientReady, client);
    });

    broker.on(AedesBrokerEvent.ClientDisconnect, (client: Client) => {
      this.clients.remove(client.id);
      logger.mqttServer.info("客户端断开", {
        clientId: client.id,
        connectedClients: broker.connectedClients,
      });
      this.events.emit(MqttServerEvent.ClientDisconnect, client);
    });

    broker.on(AedesBrokerEvent.Publish, (packet, client) => {
      if (packet.topic.startsWith(MQTT_SYS_TOPIC_PREFIX)) {
        return;
      }

      this.publishCount += 1;
      const decoded = decodePayload(packet.payload as Buffer);
      const ctx: IMqttServerMessageContext = {
        topic: packet.topic,
        payload: packet.payload as Buffer,
        raw: decoded.raw,
        data: decoded.data,
        qos: packet.qos as MqttServerQoS,
        retain: packet.retain,
        dup: packet.dup,
        clientId: client?.id || null,
        receivedAt: Date.now(),
        state: {},
      };

      logger.mqttServer.debug("Broker 收到发布", {
        // topic: ctx.topic,
        // clientId: ctx.clientId,
        // qos: ctx.qos
        ...ctx,
      });

      this.events.emit(MqttServerEvent.Publish, {
        ctx,
        client: client ?? null,
      });
    });

    broker.on(AedesBrokerEvent.Subscribe, (subscriptions, client) => {
      this.subscribeCount += 1;
      logger.mqttServer.info("客户端订阅", {
        clientId: client.id,
        topics: subscriptions.map((item) => {
          return item.topic;
        }),
      });
      this.events.emit(MqttServerEvent.Subscribe, { subscriptions, client });
    });

    broker.on(AedesBrokerEvent.Unsubscribe, (unsubscriptions, client) => {
      logger.mqttServer.info("客户端取消订阅", {
        clientId: client.id,
        topics: unsubscriptions,
      });
      this.events.emit(MqttServerEvent.Unsubscribe, {
        topics: unsubscriptions,
        client,
      });
    });

    broker.on(AedesBrokerEvent.ClientError, (client, error) => {
      this.lastError = error.message;
      logger.mqttServer.error("客户端错误", { clientId: client?.id, error });
      this.events.emit(MqttServerEvent.Error, { error, client });
    });

    broker.on(AedesBrokerEvent.ConnectionError, (client, error) => {
      this.lastError = error.message;
      logger.mqttServer.error("连接错误", { clientId: client?.id, error });
      this.events.emit(MqttServerEvent.Error, { error, client });
    });
  }
}

/*
 * 在线客户端注册表
 * 维护当前连上 Broker 的客户端快照，供查询/踢人使用
 */

import type { Client } from "aedes";
import type { Socket } from "node:net";

import type { IMqttServerClientInfo } from "../types";

export class ClientRegistry {
  private clients = new Map<
    string,
    { client: Client; info: IMqttServerClientInfo }
  >();

  /**
   * 登记客户端上线
   * @param client aedes Client
   * @param username 可选用户名
   */
  add(client: Client, username?: string) {
    const remoteAddress = this.resolveRemoteAddress(client);
    const info: IMqttServerClientInfo = {
      id: client.id,
      connected: true,
      connectedAt: Date.now(),
      username,
      remoteAddress,
    };
    this.clients.set(client.id, { client, info });
  }

  /**
   * 移除已断开的客户端
   * @param clientId 客户端 ID
   */
  remove(clientId: string) {
    this.clients.delete(clientId);
  }

  /**
   * 获取客户端对外信息（不含底层句柄）
   * @param clientId 客户端 ID
   */
  get(clientId: string): IMqttServerClientInfo | null {
    return this.clients.get(clientId)?.info || null;
  }

  /**
   * 获取 aedes 原始 Client（用于主动 disconnect）
   * @param clientId 客户端 ID
   */
  getRaw(clientId: string): Client | null {
    return this.clients.get(clientId)?.client || null;
  }

  /**
   * 列出全部在线客户端信息（拷贝返回）
   */
  list(): IMqttServerClientInfo[] {
    const result: IMqttServerClientInfo[] = [];
    for (const item of this.clients.values()) {
      result.push({
        id: item.info.id,
        connected: item.info.connected,
        connectedAt: item.info.connectedAt,
        username: item.info.username,
        remoteAddress: item.info.remoteAddress,
      });
    }
    return result;
  }

  /** 当前在线数量 */
  size() {
    return this.clients.size;
  }

  /** 清空注册表（服务停止时调用） */
  clear() {
    this.clients.clear();
  }

  /**
   * 解析客户端远端 IP
   * @param client aedes Client
   */
  private resolveRemoteAddress(client: Client): string | undefined {
    const conn = client.conn as Socket;
    return conn?.remoteAddress;
  }
}

/*
 * Aedes Broker 工厂
 *
 * 官方用法：
 *   import { Aedes } from 'aedes'
 *   const aedes = await Aedes.createBroker()
 *
 * 本工程 tsconfig 输出为 CommonJS，而 aedes@1 为纯 ESM，
 * 静态 import 会被编译成 require() 并触发 ERR_REQUIRE_ESM。
 * 因此用动态 import 加载同名导出，API 与文档一致：Aedes.createBroker()
 */

import type { Aedes, AedesOptions, AuthenticateError, Client } from "aedes";

import type { MqttServerAuthService } from "../auth";
import type { IMqttServerPluginConfig } from "../config/types";

import {
  MqttServerAuthorizeAction,
  MqttServerAuthReturnCode,
  mqttServerMsg,
} from "../constants";

/**
 * 创建并配置 Aedes Broker
 * - 动态加载 aedes 命名导出
 * - 绑定 authenticate / authorizePublish / authorizeSubscribe 到 AuthService
 * @param config 插件配置
 * @param auth 认证授权服务
 * @returns aedes 实例
 */
export async function createAedesBroker(
  config: IMqttServerPluginConfig,
  auth: MqttServerAuthService,
): Promise<Aedes> {
  const { Aedes } = await import("aedes");

  const options: AedesOptions = {
    id: config.id,
    concurrency: config.concurrency,
    connectTimeout: config.connectTimeout,
    heartbeatInterval: config.heartbeatInterval,
  };

  const broker = await Aedes.createBroker(options);

  // 连接认证：客户端 CONNECT 时触发
  broker.authenticate = (client, username, password, done) => {
    void (async () => {
      try {
        const result = await auth.authenticate({
          clientId: client.id,
          username: username || undefined,
          password: password ? password.toString("utf8") : undefined,
        });

        if (!result.ok) {
          const error = new Error(
            result.reason || "认证失败",
          ) as AuthenticateError;
          error.returnCode =
            MqttServerAuthReturnCode.BadUsernameOrPassword as unknown as AuthenticateError["returnCode"];
          done(error, false);
          return;
        }

        done(null, true);
      } catch (error) {
        logger.mqttServer.error(mqttServerMsg("authenticate 异常"), error);
        const err = error as AuthenticateError;
        err.returnCode =
          MqttServerAuthReturnCode.ServerUnavailable as unknown as AuthenticateError["returnCode"];
        done(err, false);
      }
    })();
  };

  // 发布授权：客户端 PUBLISH 时触发；client 为 null 表示服务端自身发布
  broker.authorizePublish = (client, packet, done) => {
    void (async () => {
      try {
        if (!client) {
          // 服务端自身发布
          done(null);
          return;
        }

        const ok = await auth.authorize({
          clientId: client.id,
          topic: packet.topic,
          action: MqttServerAuthorizeAction.Publish,
        });

        if (!ok) {
          logger.mqttServer.warn(mqttServerMsg("拒绝发布"), {
            clientId: client.id,
            topic: packet.topic,
          });
          done(new Error("无权发布该主题"));
          return;
        }

        done(null);
      } catch (error) {
        done(error as Error);
      }
    })();
  };

  // 订阅授权：客户端 SUBSCRIBE 时触发
  broker.authorizeSubscribe = (client, subscription, done) => {
    void (async () => {
      try {
        const ok = await auth.authorize({
          clientId: client.id,
          topic: subscription.topic,
          action: MqttServerAuthorizeAction.Subscribe,
        });

        if (!ok) {
          logger.mqttServer.warn(mqttServerMsg("拒绝订阅"), {
            clientId: client.id,
            topic: subscription.topic,
          });
          done(new Error("无权订阅该主题"), null);
          return;
        }

        done(null, subscription);
      } catch (error) {
        done(error as Error, null);
      }
    })();
  };

  logger.mqttServer.info("Aedes Broker 已创建", {
    id: config.id,
    concurrency: config.concurrency,
  });

  return broker;
}

export type { Aedes, Client };

/*
 * TCP Transport
 * 标准 MQTT over TCP：独立端口监听，createServer(aedes.handle)
 */

import type { Aedes } from "aedes";

/**
 * 启动 TCP 传输层
 * @param broker aedes 实例，连接直接交给 broker.handle
 * @param config 使用 host / port
 * @returns 已 listen 的 net.Server
 */
export async function startTcpTransport(
  broker: Aedes,
  config: IMqttServerPluginConfig,
): Promise<NetServer> {
  const server = createServer(broker.handle);

  await new Promise<void>((resolve, reject) => {
    const onError = (error: Error) => {
      server.off(NetServerEvent.Listening, onListening);
      reject(error);
    };

    const onListening = () => {
      server.off(NetServerEvent.Error, onError);
      resolve();
    };

    server.once(NetServerEvent.Error, onError);
    server.once(NetServerEvent.Listening, onListening);
    server.listen(config.port, config.host);
  });

  const connectHost =
    config.host === "0.0.0.0" || config.host === "::"
      ? "localhost"
      : config.host;
  const hostPart = connectHost.includes(":") ? `[${connectHost}]` : connectHost;

  logger.mqttServer.info(mqttMsg("TCP 已监听"), {
    host: config.host,
    port: config.port,
    address: `mqtt://${hostPart}:${config.port}`,
  });

  return server;
}

/**
 * 关闭 TCP 传输层（带超时保护）
 * @param server TCP Server，可为 null
 * @param timeout 关闭超时 ms
 */
export async function stopTcpTransport(
  server: NetServer | null,
  timeout: number,
): Promise<void> {
  if (!server) {
    return;
  }

  await new Promise<void>((resolve) => {
    const timer = setTimeout(() => {
      logger.mqttServer.warn(mqttMsg("TCP 关闭超时，强制结束"));
      resolve();
    }, timeout);

    server.close(() => {
      clearTimeout(timer);
      resolve();
    });
  });

  logger.mqttServer.info(mqttMsg("TCP 已关闭"));
}

/*
 * MQTT 消息处理链
 * Koa 风格：日志中间件 → 用户 middleware → 主题路由
 */

import type {
  IMqttServerMessageContext,
  IMqttServerMiddleware,
  IMqttServerTopicHandler,
} from "../types";

import { matchTopic } from "../utils/payload";
import { mqttServerMsg } from "../constants";

/**
 * 组合中间件为可执行的 dispatch 函数
 * @param middlewares 中间件数组（按注册顺序执行）
 */
export function composeMqttServerMiddlewares(
  middlewares: IMqttServerMiddleware[],
) {
  return async function dispatch(
    ctx: IMqttServerMessageContext,
  ): Promise<void> {
    let index = -1;

    async function next(i: number): Promise<void> {
      if (i <= index) {
        throw new Error("next() 被重复调用");
      }
      index = i;
      const fn = middlewares[i];
      if (!fn) {
        return;
      }
      await fn(ctx, () => {
        return next(i + 1);
      });
    }

    await next(0);
  };
}

export class MqttServerMessageHandler {
  private middlewares: IMqttServerMiddleware[] = [];
  private topicHandlers = new Map<string, Set<IMqttServerTopicHandler>>();
  private runner: (ctx: IMqttServerMessageContext) => Promise<void>;
  private readonly routeMiddleware: IMqttServerMiddleware;

  /**
   * 初始化默认日志中间件 + 主题路由中间件
   */
  constructor() {
    // 主题路由：按 filter 匹配后依次调用 handler
    this.routeMiddleware = async (ctx, next) => {
      for (const [filter, set] of this.topicHandlers.entries()) {
        if (!matchTopic(filter, ctx.topic)) {
          continue;
        }
        for (const handler of set) {
          await this.invokeHandler(handler, ctx);
        }
      }
      await next();
    };

    // 默认日志：记录处理耗时与异常
    this.middlewares.push(async (ctx, next) => {
      const start = Date.now();
      try {
        await next();
        logger.mqttServer.debug(mqttServerMsg("消息处理完成"), {
          topic: ctx.topic,
          clientId: ctx.clientId,
          costMs: Date.now() - start,
        });
      } catch (error) {
        logger.mqttServer.error(mqttServerMsg("消息处理失败"), {
          topic: ctx.topic,
          clientId: ctx.clientId,
          error,
          costMs: Date.now() - start,
        });
        throw error;
      }
    });

    this.runner = composeMqttServerMiddlewares([]);
    this.rebuild();
  }

  /**
   * 注册用户中间件（插入在主题路由之前）
   */
  use(middleware: IMqttServerMiddleware) {
    this.middlewares.push(middleware);
    this.rebuild();
  }

  /**
   * 注册主题处理器（支持 + / #）
   * @returns 取消注册函数
   */
  onTopic(topic: string, handler: IMqttServerTopicHandler) {
    if (!this.topicHandlers.has(topic)) {
      this.topicHandlers.set(topic, new Set());
    }
    this.topicHandlers.get(topic)!.add(handler);
    return () => {
      this.topicHandlers.get(topic)?.delete(handler);
    };
  }

  /**
   * 执行整条消息处理链
   * @param ctx 入站消息上下文
   */
  async handle(ctx: IMqttServerMessageContext) {
    await this.runner(ctx);
  }

  /** 清空全部主题处理器（停止服务时调用） */
  clear() {
    this.topicHandlers.clear();
  }

  /** 重建 runner：用户中间件 + 路由中间件 */
  private rebuild() {
    this.runner = composeMqttServerMiddlewares([
      ...this.middlewares,
      this.routeMiddleware,
    ]);
  }

  /**
   * 安全调用单个 topic handler（内部吞掉异常并记日志）
   */
  private async invokeHandler(
    handler: IMqttServerTopicHandler,
    ctx: IMqttServerMessageContext,
  ) {
    try {
      await handler(ctx);
    } catch (error) {
      logger.mqttServer.error(mqttServerMsg("Topic Handler 执行失败"), {
        topic: ctx.topic,
        error,
      });
    }
  }
}

import type { IMqttServerPluginConfig } from "../config/types";

export class MqttServerAuthService {
  private customAuthenticate: IMqttServerAuthenticateHandler | null = null;
  private customAuthorize: IMqttServerAuthorizeHandler | null = null;

  constructor(private config: IMqttServerPluginConfig) {}

  /**
   * 同步最新配置（Manager.updateConfig 时调用）
   */
  updateConfig(config: IMqttServerPluginConfig) {
    this.config = config;
  }

  /**
   * 注入自定义连接认证
   * @param handler 返回 { ok, reason? }
   */
  setAuthenticate(handler: IMqttServerAuthenticateHandler) {
    this.customAuthenticate = handler;
  }

  /**
   * 注入自定义主题授权（publish / subscribe）
   * @param handler 返回是否允许
   */
  setAuthorize(handler: IMqttServerAuthorizeHandler) {
    this.customAuthorize = handler;
  }

  /**
   * 执行连接认证
   * 优先级：自定义 handler → authEnable=false 放行 → 比对配置账号密码
   */
  async authenticate(params: {
    clientId: string;
    username?: string;
    password?: string;
  }): Promise<IMqttServerAuthResult> {
    if (this.customAuthenticate) {
      return this.customAuthenticate(params);
    }

    if (!this.config.authEnable) {
      return { ok: true };
    }

    const expectUser = this.config.username || "";
    const expectPass = this.config.password || "";
    const ok =
      params.username === expectUser && (params.password || "") === expectPass;

    if (!ok) {
      logger.mqttServer.warn(mqttServerMsg("认证失败"), {
        clientId: params.clientId,
        username: params.username,
      });
      return { ok: false, reason: "用户名或密码错误" };
    }

    return { ok: true };
  }

  /**
   * 执行发布/订阅授权
   * 有自定义 handler 则用之，否则默认全部放行
   */
  async authorize(params: {
    clientId: string;
    username?: string;
    topic: string;
    action: `${MqttServerAuthorizeAction}`;
  }): Promise<boolean> {
    if (this.customAuthorize) {
      return this.customAuthorize(params);
    }

    // 默认放行；业务可通过 setAuthorize 收紧
    return true;
  }
}

/**
 * 将值转换为布尔值
 * @param value 值
 * @param fallback 默认值
 * @returns 布尔值
 */
function toBool(value: unknown, fallback: boolean): boolean {
  if (value === undefined || value === null || value === "") {
    return fallback;
  }
  if (typeof value === "boolean") {
    return value;
  }
  return String(value).toLowerCase() === "true";
}

/**
 * 将值转换为数字
 * @param value 值
 * @param fallback 默认值
 * @returns 数字
 */
function toNumber(value: unknown, fallback: number): number {
  const n = Number(value);
  return Number.isFinite(n) ? n : fallback;
}

/**

/*
 * MQTT Server 事件总线（基于 emittery）
 * - 对内：Emittery 强类型异步派发
 * - 对外：仍暴露 on/once/off/emit，handler 直接收 data（不暴露 { name, data }）
 */

import Emittery from "emittery";

import type {
  IMqttServerEventHandler,
  IMqttServerEventMap,
  IMqttServerEventName,
} from "../types";

import { MQTT_SERVER_DEFAULTS, mqttServerMsg } from "../constants";

type IEmitteryListener = (event: { data?: unknown }) => void | Promise<void>;

export class MqttServerEventBus {
  private readonly emitter = new Emittery<IMqttServerEventMap>({
    debug: { name: MQTT_SERVER_DEFAULTS.EVENT_BUS_DEBUG_NAME },
  });

  /**
   * 业务 handler → Emittery listener 映射
   * Emittery.off 需要同一引用；我们包装了 listener，故需保留映射
   */
  private readonly listenerMap = new WeakMap<
    IMqttServerEventHandler<any>,
    IEmitteryListener
  >();

  /**
   * 订阅事件
   * @returns 取消订阅函数
   */
  on<E extends IMqttServerEventName>(
    event: E,
    handler: IMqttServerEventHandler<E>,
  ): () => void {
    const listener = this.wrapHandler(handler);
    return this.emitter.on(event, listener as any);
  }

  /**
   * 订阅一次（触发后自动取消）
   * @returns 取消订阅函数
   */
  once<E extends IMqttServerEventName>(
    event: E,
    handler: IMqttServerEventHandler<E>,
  ): () => void {
    const off = this.on(event, ((data: IMqttServerEventMap[E]) => {
      off();
      return handler(data);
    }) as IMqttServerEventHandler<E>);
    return off;
  }

  /**
   * 移除指定事件处理器
   */
  off<E extends IMqttServerEventName>(
    event: E,
    handler: IMqttServerEventHandler<E>,
  ): void {
    const listener = this.listenerMap.get(handler);
    if (!listener) {
      return;
    }
    this.emitter.off(event, listener as any);
  }

  /**
   * 异步派发事件（fire-and-forget，错误记日志）
   * Emittery 要求单一 data；无数据事件可不传第二参
   */
  emit<E extends IMqttServerEventName>(
    event: E,
    ...args: IMqttServerEventMap[E] extends undefined
      ? [] | [undefined]
      : [IMqttServerEventMap[E]]
  ): void {
    const data = args[0] as IMqttServerEventMap[E];
    void this.emitter.emit(event, data as any).catch((error) => {
      logger.mqttServer.error(mqttServerMsg("事件派发失败"), { event, error });
    });
  }

  /**
   * 清空监听
   * @param event 指定事件；不传则清空全部
   */
  removeAll(event?: IMqttServerEventName): void {
    if (event) {
      this.emitter.clearListeners(event);
      return;
    }
    this.emitter.clearListeners();
  }

  /**
   * 将业务 handler 包装为 Emittery listener（解包 { data }）
   */
  private wrapHandler<E extends IMqttServerEventName>(
    handler: IMqttServerEventHandler<E>,
  ): IEmitteryListener {
    const cached = this.listenerMap.get(handler);
    if (cached) {
      return cached;
    }

    const listener: IEmitteryListener = (event) => {
      return handler(event.data as IMqttServerEventMap[E]);
    };
    this.listenerMap.set(handler, listener);
    return listener;
  }
}

/*
 * MQTT Server 载荷编解码
 */

import type { IMqttServerPayload } from "../types";

import { mqttServerMsg } from "../constants";

/**
 * 编码发布载荷
 */
export function encodePayload(payload: IMqttServerPayload): string | Buffer {
  if (Buffer.isBuffer(payload)) {
    return payload;
  }

  if (typeof payload === "string") {
    return payload;
  }

  try {
    return JSON.stringify(payload);
  } catch (error) {
    throw new Error(
      `${mqttServerMsg("payload 序列化失败")}: ${(error as Error).message}`,
    );
  }
}

/**
 * 解码入站载荷
 */
export function decodePayload(payload: Buffer): { raw: string; data: unknown } {
  const raw = payload.toString("utf8");

  if (!raw) {
    return { raw, data: raw };
  }

  const trimmed = raw.trim();
  if (
    (trimmed.startsWith("{") && trimmed.endsWith("}")) ||
    (trimmed.startsWith("[") && trimmed.endsWith("]"))
  ) {
    try {
      return { raw, data: JSON.parse(trimmed) };
    } catch {
      return { raw, data: raw };
    }
  }

  return { raw, data: raw };
}

/**
 * MQTT 主题匹配（支持 + / #）
 */
export function matchTopic(filter: string, topic: string): boolean {
  if (filter === topic) {
    return true;
  }

  const filterParts = filter.split("/");
  const topicParts = topic.split("/");

  for (let i = 0; i < filterParts.length; i++) {
    const current = filterParts[i];

    if (current === "#") {
      return true;
    }

    if (current === "+") {
      if (topicParts[i] === undefined) {
        return false;
      }
      continue;
    }

    if (current !== topicParts[i]) {
      return false;
    }
  }

  return filterParts.length === topicParts.length;
}

注意事项 ​

  • 业务一律用单例 mqttServer，不要再 new MqttServerService（测试可用 resetMqttServerService）
  • WS 必须传 httpServer；只开 WS 却不传会跳过挂载
  • ENV_MQTT_SERVER_ENABLE=false 时 start 静默跳过；业务再调 publish 会抛 MqttServerNotEnabledError
  • 自定义认证/授权尽量在 start 之前设置
  • onTopic / on 返回的 off 须在模块卸载时调用
  • $SYS/ 系统主题不进入业务 Handler
  • aedes 为 ESM，本工程用动态 import('aedes')，勿改成静态 require
  • 与 mqtt/client 可同进程共存；注意 TCP 端口与 Client ID 不要冲突
  • 日志 / 错误文案前缀统一走 MQTT_SERVER_DEFAULTS.MODULE_NAME（mqttServerMsg / mqttMsg），改名只改 constants
  • 进程退出用 registerGracefulShutdown，任务里调 stopMqttServerPlugin()；建议先 stopMqttClient 再关 Broker
  • 与业务 WS 共用 HTTP 时注意 upgrade 冲突；同进程客户端优先 TCP
