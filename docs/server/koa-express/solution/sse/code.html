/*

import type { Server as HttpServer, IncomingMessage, ServerResponse } from 'node:http';

import type { SseServerEvent, SseServerState } from './constants';

/** 运行状态 */
export type ISseServerState = `${SseServerState}`;

/** 对外推送的业务载荷 */
export type ISseServerPayload = string | Record<string, unknown> | unknown[];

/** SSE 业务信封（编码为 event + data） */
export interface ISseEnvelope {
  event: string;
  data?: unknown;
  id?: string;
  /** 覆盖默认 retry（ms） */
  retry?: number;
}

/** 连接认证入参 */
export interface ISseAuthenticateParams {
  url: URL;
  headers: IncomingMessage['headers'];
  request: IncomingMessage;
  remoteAddress?: string;
  lastEventId?: string;
}

/** 认证结果 */
export interface ISseAuthResult {
  ok: boolean;
  reason?: string;
  /** 业务侧客户端 ID；不传则自动生成 */
  clientId?: string;
  /** 附加元数据 */
  meta?: Record<string, unknown>;
  /** 连接时自动加入的房间 */
  rooms?: string[];
}

export type ISseAuthenticateHandler = (
  params: ISseAuthenticateParams
) => ISseAuthResult | Promise<ISseAuthResult>;

/** 在线客户端对外信息 */
export interface ISseClientInfo {
  id: string;
  connected: boolean;
  connectedAt: number;
  rooms: string[];
  remoteAddress?: string;
  lastEventId?: string;
  meta: Record<string, unknown>;
}

/** 注册表内部条目 */
export interface ISseClientEntry {
  id: string;
  response: ServerResponse;
  request: IncomingMessage;
  info: ISseClientInfo;
  lastWriteAt: number;
}

/** 连接上下文（中间件 / onConnect） */
export interface ISseConnectionContext {
  clientId: string;
  rooms: string[];
  meta: Record<string, unknown>;
  url: URL;
  headers: IncomingMessage['headers'];
  remoteAddress?: string;
  lastEventId?: string;
  connectedAt: number;
  state: Record<string, unknown>;
}

export type ISseMiddleware = (
  ctx: ISseConnectionContext,
  next: () => Promise<void>
) => Promise<void>;

export type ISseConnectHandler = (ctx: ISseConnectionContext) => void | Promise<void>;

export interface ISseServerHealth {
  enabled: boolean;
  state: ISseServerState;
  path: string;
  attached: boolean;
  connectedClients: number;
  roomCount: number;
  eventCount: number;
  historySize: number;
  lastError: string | null;
  uptimeMs: number;
}

/**
 * Emittery 事件表
 */
export type ISseServerEventMap = {
  [SseServerEvent.Starting]: undefined;
  [SseServerEvent.Ready]: ISseServerHealth;
  [SseServerEvent.Stopped]: undefined;
  [SseServerEvent.StateChange]: { next: ISseServerState; prev: ISseServerState };
  [SseServerEvent.Connection]: ISseClientInfo;
  [SseServerEvent.Disconnect]: ISseClientInfo;
  [SseServerEvent.JoinRoom]: { clientId: string; room: string };
  [SseServerEvent.LeaveRoom]: { clientId: string; room: string };
  [SseServerEvent.Error]: { error: Error; clientId?: string };
};

export type ISseServerEventName = keyof ISseServerEventMap;

export type ISseServerEventHandler<E extends ISseServerEventName = ISseServerEventName> = (
  data: ISseServerEventMap[E]
) => void | Promise<void>;

export interface ISseServerStartOptions {
  /** 复用 Koa HTTP Server */
  httpServer?: HttpServer;
}

export interface ISseSendOptions {
  /** 覆盖 event 名（payload 非信封时） */
  event?: string;
  /** 指定事件 id；不传则自动生成 */
  id?: string;
  /** 是否写入全局历史（默认 true，historySize>0 时生效） */
  history?: boolean;
}

/** 广播过滤 */
export type ISseBroadcastFilter = (client: ISseClientInfo) => boolean;

/** 历史事件条目（供 Last-Event-ID 重放） */
export interface ISseHistoryEntry {
  id: string;
  event: string;
  data: string;
  retry?: number;
  createdAt: number;
  /** 若仅面向房间，重放时按房间过滤；空 = 全局 */
  rooms?: string[];
}

export interface ISseServerService {
  start: (options?: ISseServerStartOptions) => Promise<void>;
  stop: () => Promise<void>;
  isRunning: () => boolean;
  getState: () => ISseServerState;
  health: () => ISseServerHealth;

  send: (
    clientId: string,
    payload: ISseServerPayload | ISseEnvelope,
    options?: ISseSendOptions
  ) => Promise<void>;
  broadcast: (
    payload: ISseServerPayload | ISseEnvelope,
    filter?: ISseBroadcastFilter,
    options?: ISseSendOptions
  ) => Promise<void>;
  toRoom: (
    room: string,
    payload: ISseServerPayload | ISseEnvelope,
    options?: ISseSendOptions
  ) => Promise<void>;

  onConnect: (handler: ISseConnectHandler) => () => void;
  use: (middleware: ISseMiddleware) => void;

  setAuthenticate: (handler: ISseAuthenticateHandler) => void;

  listClients: () => ISseClientInfo[];
  getClient: (clientId: string) => ISseClientInfo | null;
  disconnectClient: (clientId: string, reason?: string) => Promise<void>;

  joinRoom: (clientId: string, room: string) => boolean;
  leaveRoom: (clientId: string, room: string) => boolean;
  getRoomClients: (room: string) => ISseClientInfo[];
  listRooms: () => string[];

  on: <E extends ISseServerEventName>(
    event: E,
    handler: ISseServerEventHandler<E>
  ) => () => void;
  once: <E extends ISseServerEventName>(
    event: E,
    handler: ISseServerEventHandler<E>
  ) => () => void;
  off: <E extends ISseServerEventName>(
    event: E,
    handler: ISseServerEventHandler<E>
  ) => void;

  getHttpServer: () => HttpServer | null;
}

/*
 * SSE Server 插件配置
 * env 有值用值，空值用 SSE_SERVER_DEFAULTS 兜底
 */

import type { ISseServerPluginConfig } from './types';

import { coreConfigEnv } from '../../../core/config/env';
import { SSE_SERVER_DEFAULTS } from '../constants';

/**
 * 数字：env 有值用值，否则用默认
 * @param value coreConfigEnv 数字字段（number | ''）
 * @param fallback 插件默认值
 */
function pickNumber(value: number | '', fallback: number): number {
  return value === '' ? fallback : value;
}

/**
 * 字符串：env 有值用值，否则用默认
 * @param value coreConfigEnv 字符串字段
 * @param fallback 插件默认值
 */
function pickString(value: string, fallback: string): string {
  return value === '' ? fallback : value;
}

/**
 * 解析 SSE Server 插件配置
 * @param overrides 覆盖配置
 * @returns SSE Server 插件配置
 */
export function resolveSseServerConfig(
  overrides: Partial<ISseServerPluginConfig> = {}
): ISseServerPluginConfig {
  const env = coreConfigEnv;
  const defaults = SSE_SERVER_DEFAULTS;

  const base: ISseServerPluginConfig = {
    enable: env.ENV_SSE_ENABLE === true,
    path: pickString(env.ENV_SSE_PATH, defaults.PATH),
    authEnable: env.ENV_SSE_AUTH_ENABLE === true,
    heartbeatInterval: pickNumber(env.ENV_SSE_HEARTBEAT_INTERVAL, defaults.HEARTBEAT_INTERVAL),
    retry: pickNumber(env.ENV_SSE_RETRY, defaults.RETRY),
    maxConnections: pickNumber(env.ENV_SSE_MAX_CONNECTIONS, defaults.MAX_CONNECTIONS),
    historySize: pickNumber(env.ENV_SSE_HISTORY_SIZE, defaults.HISTORY_SIZE),
    shutdownTimeout: pickNumber(env.ENV_SSE_SHUTDOWN_TIMEOUT, defaults.SHUTDOWN_TIMEOUT),
    corsOrigin: pickString(env.ENV_SSE_CORS_ORIGIN, defaults.CORS_ORIGIN)
  };

  return {
    ...base,
    ...overrides
  };
}

export type { ISseServerPluginConfig };

/*
 * SSE Server 门面服务
 * 业务侧唯一入口：组装 Manager / Handler，对外暴露稳定 API
 */

import type { ISseServerPluginConfig } from '../config/types';
import type {
  ISseAuthenticateHandler,
  ISseBroadcastFilter,
  ISseConnectHandler,
  ISseEnvelope,
  ISseMiddleware,
  ISseSendOptions,
  ISseServerEventHandler,
  ISseServerEventName,
  ISseServerPayload,
  ISseServerService,
  ISseServerStartOptions
} from '../types';

import { resolveSseServerConfig } from '../config';
import { sseServerMsg } from '../constants';
import { SseServerNotEnabledError, SseServerNotRunningError } from '../errors';
import { SseServerConnectionHandler } from '../handler';
import { SseServerManager } from '../manager';

export class SseService implements ISseServerService {
  private config: ISseServerPluginConfig;
  private manager: SseServerManager;
  private connectionHandler: SseServerConnectionHandler;
  private started = false;

  /**
   * 构造门面：解析配置并创建 Manager / Handler
   * @param overrides 可选配置覆盖（一般留给测试）
   */
  constructor(overrides: Partial<ISseServerPluginConfig> = {}) {
    this.config = resolveSseServerConfig(overrides);
    this.manager = new SseServerManager(this.config);
    this.connectionHandler = new SseServerConnectionHandler();
  }

  /**
   * 启动 SSE Server
   * @param options.httpServer 传入 Koa http.Server 以挂载
   */
  async start(options: ISseServerStartOptions = {}) {
    if (this.started && this.manager.isRunning()) {
      logger.sseServer.debug(sseServerMsg('已启动'));
      return;
    }

    this.config = resolveSseServerConfig(this.config);
    this.manager.updateConfig(this.config);

    if (!this.config.enable) {
      logger.sseServer.warn(`ENV_SSE_ENABLE=false，${sseServerMsg('未启动')}`);
      return;
    }

    this.manager.setConnectionPipeline((ctx) => {
      return this.connectionHandler.handle(ctx);
    });
    await this.manager.start(options.httpServer);
    this.started = true;
  }

  /**
   * 停止 SSE Server
   */
  async stop() {
    this.manager.setConnectionPipeline(null);
    this.connectionHandler.clear();
    await this.manager.stop();
    this.started = false;
  }

  isRunning() {
    return this.manager.isRunning();
  }

  getState() {
    return this.manager.getState();
  }

  health() {
    return this.manager.health();
  }

  /**
   * 向指定客户端推送
   * @param clientId 客户端 ID
   * @param payload 载荷 / 信封
   * @param options 发送选项
   */
  async send(
    clientId: string,
    payload: ISseServerPayload | ISseEnvelope,
    options?: ISseSendOptions
  ) {
    this.assertEnabled();
    this.assertRunning();
    return this.manager.send(clientId, payload, options);
  }

  /**
   * 广播
   * @param payload 载荷 / 信封
   * @param filter 可选过滤函数
   * @param options 发送选项
   */
  async broadcast(
    payload: ISseServerPayload | ISseEnvelope,
    filter?: ISseBroadcastFilter,
    options?: ISseSendOptions
  ) {
    this.assertEnabled();
    this.assertRunning();
    return this.manager.broadcast(payload, filter, options);
  }

  /**
   * 向房间推送
   * @param room 房间名
   * @param payload 载荷 / 信封
   * @param options 发送选项
   */
  async toRoom(
    room: string,
    payload: ISseServerPayload | ISseEnvelope,
    options?: ISseSendOptions
  ) {
    this.assertEnabled();
    this.assertRunning();
    return this.manager.toRoom(room, payload, options);
  }

  /**
   * 注册连接成功处理器
   * @returns 取消注册函数
   */
  onConnect(handler: ISseConnectHandler) {
    return this.connectionHandler.onConnect(handler);
  }

  /**
   * 注册连接中间件
   */
  use(middleware: ISseMiddleware) {
    this.connectionHandler.use(middleware);
  }

  /**
   * 设置自定义连接认证（覆盖默认策略）
   */
  setAuthenticate(handler: ISseAuthenticateHandler) {
    this.manager.auth.setAuthenticate(handler);
  }

  listClients() {
    return this.manager.clients.list();
  }

  /**
   * 按 ID 查询客户端信息
   * @param clientId 客户端 ID
   */
  getClient(clientId: string) {
    return this.manager.clients.get(clientId);
  }

  /**
   * 主动断开指定客户端
   * @param clientId 客户端 ID
   * @param reason 原因
   */
  async disconnectClient(clientId: string, reason?: string) {
    this.assertRunning();
    return this.manager.disconnectClient(clientId, reason);
  }

  /**
   * 加入房间
   * @param clientId 客户端 ID
   * @param room 房间名
   */
  joinRoom(clientId: string, room: string) {

  /**

   * @param room 房间名
   */

  }

  /**
   * 获取房间内客户端
   * @param room 房间名
   */
  getRoomClients(room: string) {
    return this.manager.clients.getRoomClients(room);
  }

  /** 列出全部房间 */
  listRooms() {
    return this.manager.clients.listRooms();
  }

  on<E extends ISseServerEventName>(event: E, handler: ISseServerEventHandler<E>) {
    return this.manager.on(event, handler);
  }

  once<E extends ISseServerEventName>(event: E, handler: ISseServerEventHandler<E>) {
    return this.manager.once(event, handler);
  }

  off<E extends ISseServerEventName>(event: E, handler: ISseServerEventHandler<E>) {

  getHttpServer() {

      throw new SseServerNotRunningError();
    }
  }

/*
 * SSE 连接与传输管理
 * 负责：挂载 HTTP、认证、心跳、帧编码、房间、历史重放、状态与 health
 */

import type { Server as HttpServer, IncomingMessage, ServerResponse } from 'node:http';

import { randomUUID } from 'node:crypto';

import type { ISseServerPluginConfig } from '../config/types';
import type {
  ISseBroadcastFilter,
  ISseConnectionContext,
  ISseEnvelope,
  ISseSendOptions,
  ISseServerEventHandler,
  ISseServerEventMap,
  ISseServerEventName,
  ISseServerHealth,
  ISseServerPayload,
  ISseServerState
} from '../types';

import { SseServerAuthService } from '../auth';
import {
  SSE_SERVER_DEFAULTS,
  SseBuiltinEvent,
  SseHttpStatus,
  sseMsg,
  SseServerEvent,
  sseServerMsg,
  SseServerState
} from '../constants';
import { SseServerClientNotFoundError, SseServerSendError } from '../errors';
import { attachSseRequestHandler, type ISseAttachResult, matchSsePath } from '../server';
import { SseServerEventBus } from '../utils/event-bus';
import {
  encodeSseFrame,
  encodeSseHeartbeatComment,
  parseRoomsFromUrl,
  SseHistoryBuffer
} from '../utils/payload';
import { ClientRegistry } from './clients';

export class SseServerManager {
  private httpServer: HttpServer | null = null;
  private attach: ISseAttachResult | null = null;
  private state: ISseServerState = SseServerState.Idle;
  private startedAt: number | null = null;
  private lastError: string | null = null;
  private eventCount = 0;
  private heartbeatTimer: ReturnType<typeof setInterval> | null = null;
  private readonly events = new SseServerEventBus();
  private history: SseHistoryBuffer;
  /** 连接成功后交给 facade 的回调 */
  private connectionPipeline: ((ctx: ISseConnectionContext) => Promise<void>) | null = null;
  /** 在线客户端与房间 */
  readonly clients = new ClientRegistry();
  /** 连接认证 */
  readonly auth: SseServerAuthService;

  constructor(private config: ISseServerPluginConfig) {
    this.auth = new SseServerAuthService(config);
    this.history = new SseHistoryBuffer(config.historySize);
  }

  getConfig() {
    return this.config;
  }

  /**
   * 热更新配置（启动前可调用）
   * @param config 新配置
   */
  updateConfig(config: ISseServerPluginConfig) {
    this.config = config;
    this.auth.updateConfig(config);
    this.history.setMaxSize(config.historySize);
  }

  /**
   * 设置连接处理管道（由 Facade 注入）
   */
  setConnectionPipeline(pipeline: ((ctx: ISseConnectionContext) => Promise<void>) | null) {
    this.connectionPipeline = pipeline;
  }

  getHttpServer() {
    return this.httpServer;
  }

  getState() {
    return this.state;
  }

  isRunning() {
    return this.state === SseServerState.Running && !!this.attach;
  }

  on<E extends ISseServerEventName>(event: E, handler: ISseServerEventHandler<E>) {
    return this.events.on(event, handler);
  }

  once<E extends ISseServerEventName>(event: E, handler: ISseServerEventHandler<E>) {
    return this.events.once(event, handler);
  }

  off<E extends ISseServerEventName>(event: E, handler: ISseServerEventHandler<E>) {
    this.events.off(event, handler);
  }

  emit<E extends ISseServerEventName>(
    event: E,
    ...args: ISseServerEventMap[E] extends undefined ? [] | [undefined] : [ISseServerEventMap[E]]
  ) {
    this.events.emit(event, ...(args as any));
  }

  /**
   * 启动 SSE Server
   * @param httpServer Koa 的 http.Server；不传则无法挂载
   */
  async start(httpServer?: HttpServer) {
    if (!this.config.enable) {
      logger.sseServer.warn(sseServerMsg('未启用，跳过启动'));
      return;
    }

    if (this.isRunning()) {
      logger.sseServer.debug(sseServerMsg('已在运行'));
      return;
    }

    if (!httpServer) {
      throw new Error(sseServerMsg('需要传入 httpServer 才能挂载'));
    }

    this.setState(SseServerState.Starting);
    this.events.emit(SseServerEvent.Starting);
    this.httpServer = httpServer;
    this.startedAt = Date.now();

    try {
      this.attach = attachSseRequestHandler(httpServer, this.config, (req, res) => {
        return this.handleHttpRequest(req, res);
      });
      this.startHeartbeat();

      this.setState(SseServerState.Running);
      this.lastError = null;
      this.events.emit(SseServerEvent.Ready, this.health());
      logger.sseServer.info(sseServerMsg('启动完成'), this.health());
    } catch (error) {
      this.lastError = (error as Error).message;
      this.setState(SseServerState.Error);
      this.events.emit(SseServerEvent.Error, { error: error as Error });
      throw error;
    }
  }

  /**
   * 停止服务：关连接 → 停心跳 → detach → 清空注册表 / 历史
   */
  async stop() {
    if (this.state === SseServerState.Idle || this.state === SseServerState.Stopped) {
      return;
    }

    this.setState(SseServerState.Stopping);
    this.stopHeartbeat();

    const closeTasks: Array<Promise<void>> = [];
    this.clients.forEach((entry) => {
      closeTasks.push(this.closeResponse(entry.response, entry.id, 'server shutting down'));
    });

    await Promise.race([
      Promise.allSettled(closeTasks),
      new Promise<void>((resolve) => {
        setTimeout(resolve, this.config.shutdownTimeout);
      })
    ]);

    this.attach?.detach();
    this.attach = null;
    this.clients.clear();
    this.history.clear();
    this.httpServer = null;
    this.setState(SseServerState.Stopped);
    this.events.emit(SseServerEvent.Stopped);
    logger.sseServer.info(sseServerMsg('已停止'));
  }

  /**
   * 向指定客户端发送消息
   */
  async send(
    clientId: string,
    payload: ISseServerPayload | ISseEnvelope,
    options: ISseSendOptions = {}
  ) {
    const entry = this.clients.getEntry(clientId);
    if (!entry) {
      throw new SseServerClientNotFoundError(clientId);
    }
    await this.writeToClient(entry.id, entry.response, payload, options);
  }

  /**
   * 广播消息
   */
  async broadcast(
    payload: ISseServerPayload | ISseEnvelope,
    filter?: ISseBroadcastFilter,
    options: ISseSendOptions = {}
  ) {
    const frame = encodeSseFrame(payload, {
      event: options.event,
      id: options.id,
      retry: this.config.retry
    });

    this.recordHistory(frame, undefined, options.history !== false);
    this.eventCount += 1;

    const tasks: Array<Promise<void>> = [];
    this.clients.forEach((entry) => {
      if (filter && !filter(entry.info)) {
        return;
      }
      tasks.push(this.writeRaw(entry.id, entry.response, frame.raw, frame.id));
    });
    await Promise.allSettled(tasks);
  }

  /**
   * 向房间广播
   */
  async toRoom(
    room: string,
    payload: ISseServerPayload | ISseEnvelope,
    options: ISseSendOptions = {}
  ) {
    if (!room) {
      throw new SseServerSendError('room 不能为空');
    }

    const frame = encodeSseFrame(payload, {
      event: options.event,
      id: options.id,
      retry: this.config.retry
    });

    this.recordHistory(frame, [room], options.history !== false);
    this.eventCount += 1;

    const entries = this.clients.getRoomEntries(room);
    await Promise.allSettled(
      entries.map((entry) => {
        return this.writeRaw(entry.id, entry.response, frame.raw, frame.id);
      })
    );
  }

  /**
   * 加入房间
   */
  joinRoom(clientId: string, room: string): boolean {
    const ok = this.clients.joinRoom(clientId, room);
    if (ok) {
      this.events.emit(SseServerEvent.JoinRoom, { clientId, room });
      logger.sseServer.info('客户端加入房间', { clientId, room });
    }
    return ok;
  }

  /**
   * 离开房间
   */
  leaveRoom(clientId: string, room: string): boolean {
    const ok = this.clients.leaveRoom(clientId, room);
    if (ok) {
      this.events.emit(SseServerEvent.LeaveRoom, { clientId, room });
      logger.sseServer.info('客户端离开房间', { clientId, room });
    }
    return ok;
  }

  /**
   * 主动断开客户端
   */
  async disconnectClient(clientId: string, reason = 'disconnected by server') {
    const entry = this.clients.getEntry(clientId);
    if (!entry) {
      return;
    }
    await this.closeResponse(entry.response, clientId, reason);
  }

  health(): ISseServerHealth {
    return {
      enabled: this.config.enable,
      state: this.state,
      path: this.config.path,
      attached: !!this.attach,
      connectedClients: this.clients.size(),
      roomCount: this.clients.roomCount(),
      eventCount: this.eventCount,
      historySize: this.history.size(),
      lastError: this.lastError,
      uptimeMs: this.startedAt ? Date.now() - this.startedAt : 0
    };
  }

  private setState(next: ISseServerState) {
    if (this.state === next) {
      return;
    }
    const prev = this.state;
    this.state = next;
    logger.sseServer.debug(sseServerMsg('状态变更'), { from: prev, to: next });
    this.events.emit(SseServerEvent.StateChange, { next, prev });
  }

  /**
   * HTTP 请求入口：匹配 path 则建立 SSE，否则放行给 Koa
   */
  private async handleHttpRequest(
    request: IncomingMessage,
    response: ServerResponse
  ): Promise<boolean> {
    if (!matchSsePath(request.url, this.config.path)) {
      return false;
    }

    if (request.method === 'OPTIONS') {
      this.writeCorsHeaders(request, response);
      response.statusCode = SseHttpStatus.Ok;
      response.end();
      return true;
    }

    if (request.method !== 'GET' && request.method !== 'HEAD') {
      this.writeCorsHeaders(request, response);
      response.statusCode = SseHttpStatus.MethodNotAllowed;
      response.setHeader('Allow', 'GET, HEAD, OPTIONS');
      response.end('Method Not Allowed');
      return true;
    }

    if (request.method === 'HEAD') {
      this.writeCorsHeaders(request, response);
      this.writeSseHeaders(response);
      response.statusCode = SseHttpStatus.Ok;
      response.end();
      return true;
    }

    await this.openConnection(request, response);
    return true;
  }

  private async openConnection(request: IncomingMessage, response: ServerResponse) {
    const remoteAddress = request.socket.remoteAddress;
    let clientId = '';

    try {
      if (
        this.config.maxConnections > 0 &&
        this.clients.size() >= this.config.maxConnections
      ) {
        this.writeCorsHeaders(request, response);
        response.statusCode = SseHttpStatus.TooManyRequests;
        response.end('Too Many Connections');
        logger.sseServer.warn(sseMsg('连接数超限'), {
          maxConnections: this.config.maxConnections
        });
        return;
      }

      const url = new URL(request.url || '/', 'http://localhost');
      const lastEventIdHeader = request.headers['last-event-id'];
      const lastEventId =
        (typeof lastEventIdHeader === 'string' && lastEventIdHeader) ||
        url.searchParams.get('lastEventId') ||
        undefined;

      const authResult = await this.auth.authenticate({
        url,
        headers: request.headers,
        request,
        remoteAddress,
        lastEventId
      });

      if (!authResult.ok) {
        logger.sseServer.warn(sseMsg('连接认证失败'), {
          reason: authResult.reason,
          remoteAddress
        });
        this.writeCorsHeaders(request, response);
        response.statusCode = SseHttpStatus.Unauthorized;
        response.end(authResult.reason || 'unauthorized');
        return;
      }

      clientId =
        authResult.clientId ||
        url.searchParams.get('clientId') ||
        `${SSE_SERVER_DEFAULTS.CLIENT_ID_PREFIX}-${randomUUID()}`;

      // 同 ID 重连：踢掉旧连接
      const existing = this.clients.getEntry(clientId);
      if (existing) {
        logger.sseServer.warn('客户端 ID 冲突，断开旧连接', { clientId });
        await this.closeResponse(existing.response, clientId, 'replaced by new connection');
      }

      const queryRooms = parseRoomsFromUrl(url);
      const authRooms = authResult.rooms || [];
      const rooms = [...new Set([...queryRooms, ...authRooms])];

      this.writeCorsHeaders(request, response);
      this.writeSseHeaders(response);
      response.statusCode = SseHttpStatus.Ok;
      // 立即 flush 头，避免中间代理缓冲
      response.write(': ok\n\n');

      const info = this.clients.add(clientId, response, request, {
        remoteAddress,
        meta: authResult.meta,
        lastEventId,
        rooms
      });

      this.bindResponseEvents(request, response, clientId);

      // 下发默认 retry + connected 事件
      const connectedFrame = encodeSseFrame(
        {
          event: SseBuiltinEvent.Connected,
          data: {
            clientId,
            rooms: info.rooms,
            at: Date.now()
          }
        },
        { retry: this.config.retry }
      );
      await this.writeRaw(clientId, response, connectedFrame.raw, connectedFrame.id);

      // Last-Event-ID 重放
      if (lastEventId) {
        const replay = this.history.replayAfter(lastEventId, info.rooms);
        for (const item of replay) {
          const raw = [
            `id: ${item.id}`,
            `event: ${item.event}`,
            ...(typeof item.retry === 'number' ? [`retry: ${item.retry}`] : []),
            ...item.data.split(/\r?\n/).map((line) => {
              return `data: ${line}`;
            }),
            '',
            ''
          ].join('\n');
          await this.writeRaw(clientId, response, raw, item.id);
        }
        if (replay.length > 0) {
          logger.sseServer.info(sseMsg('历史事件已重放'), {
            clientId,
            count: replay.length,
            lastEventId
          });
        }
      }

      this.events.emit(SseServerEvent.Connection, info);
      logger.sseServer.info('客户端已连接', {
        clientId,
        remoteAddress,
        rooms: info.rooms,
        connectedClients: this.clients.size()
      });

      if (this.connectionPipeline) {
        const ctx: ISseConnectionContext = {
          clientId,
          rooms: [...info.rooms],
          meta: { ...info.meta },
          url,
          headers: request.headers,
          remoteAddress,
          lastEventId,
          connectedAt: info.connectedAt,
          state: {}
        };
        await this.connectionPipeline(ctx);
      }
    } catch (error) {
      this.lastError = (error as Error).message;
      logger.sseServer.error(sseMsg('处理连接失败'), error);
      this.events.emit(SseServerEvent.Error, {
        error: error as Error,
        clientId: clientId || undefined
      });
      if (!response.writableEnded) {
        try {
          if (!response.headersSent) {
            this.writeCorsHeaders(request, response);
            response.statusCode = SseHttpStatus.InternalError;
            response.end('internal error');
          } else {
            response.end();
          }
        } catch {
          // ignore
        }
      }
      if (clientId) {
        this.clients.remove(clientId);
      }
    }
  }

  private bindResponseEvents(
    request: IncomingMessage,
    response: ServerResponse,
    clientId: string
  ) {
    let closed = false;
    const onClose = () => {
      if (closed) {
        return;
      }
      // 同 clientId 重连后，旧连接 close 不得误删新条目
      const entry = this.clients.getEntry(clientId);
      if (!entry || entry.response !== response) {
        return;
      }
      closed = true;
      const info = this.clients.remove(clientId);
      logger.sseServer.info('客户端已断开', {
        clientId,
        connectedClients: this.clients.size()
      });
      if (info) {
        this.events.emit(SseServerEvent.Disconnect, info);
      }
    };

    request.on('close', onClose);
    response.on('close', onClose);
    response.on('error', (error) => {
      this.lastError = error.message;
      logger.sseServer.error('客户端 Response 错误', { clientId, error });
      this.events.emit(SseServerEvent.Error, { error, clientId });
    });
  }

  private writeSseHeaders(response: ServerResponse) {
    response.setHeader('Content-Type', 'text/event-stream; charset=utf-8');
    response.setHeader('Cache-Control', 'no-cache, no-transform');
    response.setHeader('Connection', 'keep-alive');
    response.setHeader('X-Accel-Buffering', 'no');
  }

  private writeCorsHeaders(request: IncomingMessage, response: ServerResponse) {
    const originHeader = request.headers.origin;
    const configured = this.config.corsOrigin;

    if (configured === '*') {
      response.setHeader('Access-Control-Allow-Origin', originHeader || '*');
    } else if (configured) {
      response.setHeader('Access-Control-Allow-Origin', configured);
    }

    response.setHeader('Access-Control-Allow-Headers', 'Last-Event-ID, Authorization, Content-Type');
    response.setHeader('Access-Control-Allow-Methods', 'GET, HEAD, OPTIONS');
    if (originHeader && configured !== '*') {
      response.setHeader('Vary', 'Origin');
    }
    // EventSource 带 cookie 时需 credentials；与 * 互斥，仅具体 origin 时开启
    if (configured && configured !== '*') {
      response.setHeader('Access-Control-Allow-Credentials', 'true');
    }
  }

  private async writeToClient(
    clientId: string,
    response: ServerResponse,
    payload: ISseServerPayload | ISseEnvelope,
    options: ISseSendOptions
  ) {
    const frame = encodeSseFrame(payload, {
      event: options.event,
      id: options.id,
      retry: this.config.retry
    });

    this.recordHistory(frame, undefined, options.history !== false);
    this.eventCount += 1;
    await this.writeRaw(clientId, response, frame.raw, frame.id);
  }

  private async writeRaw(
    clientId: string,
    response: ServerResponse,
    raw: string,
    eventId?: string
  ) {
    if (response.writableEnded || response.destroyed) {
      throw new SseServerSendError('客户端连接已关闭', { clientId });
    }

    await new Promise<void>((resolve, reject) => {
      response.write(raw, (error) => {
        if (error) {
          logger.sseServer.error(sseMsg('发送失败'), { clientId, error });
          reject(new SseServerSendError(error.message, { clientId }));
          return;
        }
        this.clients.touchWrite(clientId, eventId);
        resolve();
      });
    });
  }

  private recordHistory(
    frame: { id: string; event: string; data: string; retry?: number },
    rooms: string[] | undefined,
    enabled: boolean
  ) {
    if (!enabled || this.config.historySize <= 0) {
      return;
    }
    this.history.push({
      id: frame.id,
      event: frame.event,
      data: frame.data,
      retry: frame.retry,
      createdAt: Date.now(),
      rooms
    });
  }

  private async closeResponse(
    response: ServerResponse,
    clientId: string,
    reason: string
  ): Promise<void> {
    return new Promise((resolve) => {
      let settled = false;
      const done = () => {
        if (settled) {
          return;
        }
        settled = true;
        // 仅当注册表仍指向本 response 时移除，避免误伤重连后的新连接
        const entry = this.clients.getEntry(clientId);
        if (entry && entry.response === response) {
          const info = this.clients.remove(clientId);
          if (info) {
            this.events.emit(SseServerEvent.Disconnect, info);
          }
        }
        resolve();
      };

      if (response.writableEnded || response.destroyed) {
        done();
        return;
      }

      response.once('close', done);
      try {
        const frame = encodeSseFrame({
          event: SseBuiltinEvent.Error,
          data: { reason }
        });
        response.write(frame.raw, () => {
          response.end();
        });
      } catch {
        try {
          response.end();
        } catch {
          done();
        }
      }

      setTimeout(done, 1_000).unref?.();
    });
  }

  private startHeartbeat() {
    this.stopHeartbeat();
    if (this.config.heartbeatInterval <= 0) {
      return;
    }

    this.heartbeatTimer = setInterval(() => {
      this.clients.forEach((entry) => {
        if (entry.response.writableEnded || entry.response.destroyed) {
          return;
        }
        try {
          const ok = entry.response.write(encodeSseHeartbeatComment());
          if (ok) {
            this.clients.touchWrite(entry.id);
          }
        } catch (error) {
          logger.sseServer.warn('发送心跳失败，断开客户端', {
            clientId: entry.id,
            error
          });
          void this.closeResponse(entry.response, entry.id, 'heartbeat failed');
        }
      });
    }, this.config.heartbeatInterval);

    if (typeof this.heartbeatTimer.unref === 'function') {
      this.heartbeatTimer.unref();
    }
  }

  private stopHeartbeat() {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer);
      this.heartbeatTimer = null;
    }
  }
}

/*
 * 在线客户端与房间注册表
 */

import type { IncomingMessage, ServerResponse } from 'node:http';

import type { ISseClientEntry, ISseClientInfo } from '../types';

export class ClientRegistry {
  private clients = new Map<string, ISseClientEntry>();
  /** room -> clientId set */
  private rooms = new Map<string, Set<string>>();

  /**
   * 登记客户端上线
   * @param id 客户端 ID
   * @param response SSE 响应流
   * @param request 原始请求
   * @param options remoteAddress / meta / lastEventId / rooms
   */
  add(
    id: string,
    response: ServerResponse,
    request: IncomingMessage,
    options: {
      remoteAddress?: string;
      meta?: Record<string, unknown>;
      lastEventId?: string;
      rooms?: string[];
    } = {}
  ): ISseClientInfo {
    const info: ISseClientInfo = {
      id,
      connected: true,
      connectedAt: Date.now(),
      rooms: [],
      remoteAddress: options.remoteAddress,
      lastEventId: options.lastEventId,
      meta: options.meta || {}
    };

    this.clients.set(id, {
      id,
      response,
      request,
      info,
      lastWriteAt: Date.now()
    });

    if (options.rooms) {
      for (const room of options.rooms) {
        this.joinRoom(id, room);
      }
    }

    return { ...info, rooms: [...info.rooms], meta: { ...info.meta } };
  }

  /**
   * 移除已断开的客户端，并清理其房间成员关系
   * @param clientId 客户端 ID
   */
  remove(clientId: string): ISseClientInfo | null {
    const entry = this.clients.get(clientId);
    if (!entry) {
      return null;
    }

    for (const room of [...entry.info.rooms]) {
      this.leaveRoom(clientId, room);
    }

    this.clients.delete(clientId);
    return {
      ...entry.info,
      connected: false,
      rooms: [...entry.info.rooms],
      meta: { ...entry.info.meta }
    };
  }

  /**
   * 获取客户端对外信息
   * @param clientId 客户端 ID
   */
  get(clientId: string): ISseClientInfo | null {
    const entry = this.clients.get(clientId);
    if (!entry) {
      return null;
    }
    return {
      ...entry.info,
      rooms: [...entry.info.rooms],
      meta: { ...entry.info.meta }
    };
  }

  /**
   * 获取注册表原始条目（含 response）
   * @param clientId 客户端 ID
   */
  getEntry(clientId: string): ISseClientEntry | null {
    return this.clients.get(clientId) || null;
  }

  /**
   * 列出全部在线客户端信息
   */
  list(): ISseClientInfo[] {
    const result: ISseClientInfo[] = [];
    for (const entry of this.clients.values()) {
      result.push({
        ...entry.info,
        rooms: [...entry.info.rooms],
        meta: { ...entry.info.meta }
      });
    }
    return result;
  }

  /** 当前在线数量 */
  size() {
    return this.clients.size;
  }

  /**
   * 加入房间
   * @param clientId 客户端 ID
   * @param room 房间名
   */
  joinRoom(clientId: string, room: string): boolean {
    const entry = this.clients.get(clientId);
    if (!entry || !room) {
      return false;
    }

    if (!entry.info.rooms.includes(room)) {
      entry.info.rooms.push(room);
    }

    if (!this.rooms.has(room)) {
      this.rooms.set(room, new Set());
    }
    this.rooms.get(room)!.add(clientId);
    return true;
  }

  /**
   * 离开房间
   * @param clientId 客户端 ID
   * @param room 房间名
   */
  leaveRoom(clientId: string, room: string): boolean {
    const entry = this.clients.get(clientId);
    if (!entry || !room) {
      return false;
    }

    entry.info.rooms = entry.info.rooms.filter((item) => {
      return item !== room;
    });

    const members = this.rooms.get(room);
    if (members) {
      members.delete(clientId);
      if (members.size === 0) {
        this.rooms.delete(room);
      }
    }
    return true;
  }

  /**
   * 获取房间内客户端信息列表
   * @param room 房间名
   */
  getRoomClients(room: string): ISseClientInfo[] {
    const members = this.rooms.get(room);
    if (!members) {
      return [];
    }

    const result: ISseClientInfo[] = [];
    for (const clientId of members) {
      const info = this.get(clientId);
      if (info) {
        result.push(info);
      }
    }
    return result;
  }

  /**
   * 获取房间内客户端条目
   * @param room 房间名
   */
  getRoomEntries(room: string): ISseClientEntry[] {
    const members = this.rooms.get(room);
    if (!members) {
      return [];
    }

    const result: ISseClientEntry[] = [];
    for (const clientId of members) {
      const entry = this.clients.get(clientId);
      if (entry) {
        result.push(entry);
      }
    }
    return result;
  }

  /** 列出全部房间名 */
  listRooms(): string[] {
    return [...this.rooms.keys()];
  }

  /** 房间数量 */
  roomCount() {
    return this.rooms.size;
  }

  /**
   * 更新最近写入时间 / lastEventId
   */
  touchWrite(clientId: string, eventId?: string) {
    const entry = this.clients.get(clientId);
    if (entry) {
      entry.lastWriteAt = Date.now();
      if (eventId) {
        entry.info.lastEventId = eventId;
      }
    }
  }

  /** 遍历全部条目 */
  forEach(fn: (entry: ISseClientEntry) => void) {
    for (const entry of this.clients.values()) {
      fn(entry);
    }
  }

  /** 清空注册表 */
  clear() {
    this.clients.clear();
    this.rooms.clear();
  }
}

/*
 * SSE Server 挂载工厂（挂到 Koa HTTP Server）
 * 在现有 request listener 之前拦截 path，未命中则交给原监听器（Koa）
 */

import type { Server as HttpServer, IncomingMessage, ServerResponse } from 'node:http';

import type { ISseServerPluginConfig } from '../config/types';

import { sseServerMsg } from '../constants';

export type ISseRequestHandler = (
  request: IncomingMessage,
  response: ServerResponse
) => boolean | Promise<boolean>;

export interface ISseAttachResult {
  /** 卸载拦截器，恢复原 request listeners */
  detach: () => void;
}

/**
 * 挂载 SSE 请求拦截器到 http.Server
 * @param httpServer Koa 的 http.Server
 * @param config 插件配置
 * @param handler 返回 true 表示已消费该请求
 */
export function attachSseRequestHandler(
  httpServer: HttpServer,
  config: ISseServerPluginConfig,
  handler: ISseRequestHandler
): ISseAttachResult {
  const previousListeners = httpServer.listeners('request').slice() as Array<
    (req: IncomingMessage, res: ServerResponse) => void
  >;

  httpServer.removeAllListeners('request');

  const dispatcher = (req: IncomingMessage, res: ServerResponse) => {
    void (async () => {
      try {
        const consumed = await handler(req, res);
        if (consumed) {
          return;
        }
      } catch (error) {
        logger.sseServer.error(sseServerMsg('请求拦截异常'), error);
        if (!res.headersSent) {
          res.statusCode = 500;
          res.end('Internal Server Error');
        }
        return;
      }

      for (const listener of previousListeners) {
        listener.call(httpServer, req, res);
      }
    })();
  };

  httpServer.on('request', dispatcher);

  const addr = httpServer.address();
  let address = config.path;
  if (addr && typeof addr !== 'string') {
    const connectHost =
      addr.address === '0.0.0.0' || addr.address === '::' ? 'localhost' : addr.address;
    const hostPart = connectHost.includes(':') ? `[${connectHost}]` : connectHost;
    address = `http://${hostPart}:${addr.port}${config.path}`;
  }

  logger.sseServer.info(sseServerMsg('已挂载'), {
    path: config.path,
    address,
    retry: config.retry,
    maxConnections: config.maxConnections
  });

  return {
    detach: () => {
      httpServer.removeListener('request', dispatcher);
      // 仅当当前没有 request listener 时恢复，避免覆盖后续挂载
      if (httpServer.listeners('request').length === 0) {
        for (const listener of previousListeners) {
          httpServer.on('request', listener);
        }
      }
    }
  };
}

/**
 * 判断请求 path 是否匹配 SSE 挂载路径（忽略 query）
 */
export function matchSsePath(requestUrl: string | undefined, mountPath: string): boolean {
  if (!requestUrl) {
    return false;
  }
  try {
    const url = new URL(requestUrl, 'http://localhost');
    const normalizedMount = mountPath.endsWith('/') && mountPath !== '/'
      ? mountPath.slice(0, -1)
      : mountPath;
    const pathname =
      url.pathname.endsWith('/') && url.pathname !== '/'
        ? url.pathname.slice(0, -1)
        : url.pathname;
    return pathname === normalizedMount;
  } catch {
    return false;
  }
}

/*
 * SSE 连接处理链
 * Koa 风格：日志中间件 → 用户 middleware → onConnect 路由
 */

import type { ISseConnectHandler, ISseConnectionContext, ISseMiddleware } from '../types';

import { sseMsg } from '../constants';

/**
 * 组合中间件为可执行的 dispatch 函数
 * @param middlewares 中间件数组（按注册顺序执行）
 */
export function composeSseMiddlewares(middlewares: ISseMiddleware[]) {
  return async function dispatch(ctx: ISseConnectionContext): Promise<void> {
    let index = -1;

    async function next(i: number): Promise<void> {
      if (i <= index) {
        throw new Error('next() 被重复调用');
      }
      index = i;
      const fn = middlewares[i];
      if (!fn) {
        return;
      }
      await fn(ctx, () => {
        return next(i + 1);
      });
    }

    await next(0);
  };
}

export class SseServerConnectionHandler {
  private middlewares: ISseMiddleware[] = [];
  private connectHandlers = new Set<ISseConnectHandler>();
  private runner: (ctx: ISseConnectionContext) => Promise<void>;
  private readonly routeMiddleware: ISseMiddleware;

  constructor() {
    this.routeMiddleware = async (ctx, next) => {
      for (const handler of this.connectHandlers) {
        await this.invokeHandler(handler, ctx);
      }
      await next();
    };

    this.middlewares.push(async (ctx, next) => {
      const start = Date.now();
      try {
        await next();
        logger.sseServer.debug(sseMsg('连接处理完成'), {
          clientId: ctx.clientId,
          costMs: Date.now() - start
        });
      } catch (error) {
        logger.sseServer.error(sseMsg('连接处理失败'), {
          clientId: ctx.clientId,
          error,
          costMs: Date.now() - start
        });
        throw error;
      }
    });

    this.runner = composeSseMiddlewares([]);
    this.rebuild();
  }

  /**
   * 注册用户中间件（插入在 onConnect 路由之前）
   */
  use(middleware: ISseMiddleware) {
    this.middlewares.push(middleware);
    this.rebuild();
  }

  /**
   * 注册连接成功处理器
   * @returns 取消注册函数
   */
  onConnect(handler: ISseConnectHandler) {
    this.connectHandlers.add(handler);
    return () => {
      this.connectHandlers.delete(handler);
    };
  }

  /**
   * 执行整条连接处理链
   * @param ctx 连接上下文
   */
  async handle(ctx: ISseConnectionContext) {
    await this.runner(ctx);
  }

  /** 清空全部 onConnect 处理器 */
  clear() {
    this.connectHandlers.clear();
  }

  private rebuild() {
    this.runner = composeSseMiddlewares([...this.middlewares, this.routeMiddleware]);
  }

  private async invokeHandler(handler: ISseConnectHandler, ctx: ISseConnectionContext) {
    try {
      await handler(ctx);
    } catch (error) {
      logger.sseServer.error(sseMsg('Connect Handler 执行失败'), {
        clientId: ctx.clientId,
        error
      });
    }
  }
}

/*
 * SSE Server 帧编解码 / 历史缓冲
 */

import { randomUUID } from 'node:crypto';

import type { ISseEnvelope, ISseHistoryEntry, ISseServerPayload } from '../types';

import { sseMsg } from '../constants';

export interface ISseEncodedFrame {
  id: string;
  event: string;
  data: string;
  retry?: number;
  /** 完整 SSE 文本帧（含末尾空行） */
  raw: string;
}

/**
 * 将 data 序列化为 SSE data 行内容（多行会拆成多条 data:）
 * @param value 原始值
 */
export function serializeSseData(value: unknown): string {
  if (typeof value === 'string') {
    return value;
  }
  try {
    return JSON.stringify(value);
  } catch (error) {
    throw new Error(`${sseMsg('payload 序列化失败')}: ${(error as Error).message}`);
  }
}

/**
 * 编码单条 SSE 帧
 * @param payload 载荷 / 信封
 * @param options event / id / retry
 */
export function encodeSseFrame(
  payload: ISseServerPayload | ISseEnvelope,
  options: { event?: string; id?: string; retry?: number } = {}
): ISseEncodedFrame {
  let event = options.event || 'message';
  let dataValue: unknown = payload;
  let id = options.id;
  let retry = options.retry;

  if (
    payload &&
    typeof payload === 'object' &&
    !Array.isArray(payload) &&
    typeof (payload as ISseEnvelope).event === 'string'
  ) {
    const envelope = payload as ISseEnvelope;
    event = envelope.event;
    dataValue = envelope.data;
    if (envelope.id) {
      id = envelope.id;
    }
    if (typeof envelope.retry === 'number') {
      retry = envelope.retry;
    }
  }

  const finalId = id || randomUUID();
  const data = serializeSseData(dataValue === undefined ? null : dataValue);
  const lines: string[] = [];

  lines.push(`id: ${finalId}`);
  lines.push(`event: ${event}`);
  if (typeof retry === 'number' && Number.isFinite(retry) && retry >= 0) {
    lines.push(`retry: ${Math.floor(retry)}`);
  }

  // SSE 规范：多行 data 用多条 data: 行
  const dataLines = data.split(/\r?\n/);
  for (const line of dataLines) {
    lines.push(`data: ${line}`);
  }

  lines.push('');
  lines.push('');

  return {
    id: finalId,
    event,
    data,
    retry,
    raw: lines.join('\n')
  };
}

/**
 * 有界历史缓冲（FIFO），供 Last-Event-ID 重放
 */
export class SseHistoryBuffer {
  private items: ISseHistoryEntry[] = [];
  private readonly index = new Map<string, number>();

  constructor(private maxSize: number) {}

  /**
   * 更新容量；缩容时丢弃最旧
   */
  setMaxSize(maxSize: number) {
    this.maxSize = maxSize;
    this.trim();
  }

  size() {
    return this.items.length;
  }

  /**
   * 追加历史
   */
  push(entry: ISseHistoryEntry) {
    if (this.maxSize <= 0) {
      return;
    }
    this.items.push(entry);
    this.index.set(entry.id, this.items.length - 1);
    this.trim();
  }

  /**
   * 取出 lastEventId 之后的事件（不含 lastEventId 自身）
   * @param lastEventId 客户端 Last-Event-ID
   * @param rooms 当前连接所属房间（用于房间级历史过滤）
   */
  replayAfter(lastEventId: string | undefined, rooms: string[]): ISseHistoryEntry[] {
    if (!lastEventId || this.items.length === 0) {
      return [];
    }

    const start = this.index.get(lastEventId);
    if (start === undefined) {
      // id 已滚出缓冲：不重放，避免重复推送整库
      return [];
    }

    const roomSet = new Set(rooms);
    const result: ISseHistoryEntry[] = [];

    for (let i = start + 1; i < this.items.length; i += 1) {
      const item = this.items[i];
      if (!item) {
        continue;
      }
      if (!item.rooms || item.rooms.length === 0) {
        result.push(item);
        continue;
      }
      const hit = item.rooms.some((room) => {
        return roomSet.has(room);
      });
      if (hit) {
        result.push(item);
      }
    }

    return result;
  }

  clear() {
    this.items = [];
    this.index.clear();
  }

  private trim() {
    if (this.maxSize <= 0) {
      this.clear();
      return;
    }
    while (this.items.length > this.maxSize) {
      const removed = this.items.shift();
      if (removed) {
        this.index.delete(removed.id);
      }
    }
    // 重建 index（shift 后下标变化）
    this.index.clear();
    for (let i = 0; i < this.items.length; i += 1) {
      const item = this.items[i];
      if (item) {
        this.index.set(item.id, i);
      }
    }
  }
}

/**
 * 从 query 解析房间列表：`rooms=a,b` 或多次 `room=a&room=b`
 */
export function parseRoomsFromUrl(url: URL): string[] {
  const rooms: string[] = [];
  const csv = url.searchParams.get('rooms');
  if (csv) {
    for (const part of csv.split(',')) {
      const room = part.trim();
      if (room) {
        rooms.push(room);
      }
    }
  }
  for (const room of url.searchParams.getAll('room')) {
    const trimmed = room.trim();
    if (trimmed) {
      rooms.push(trimmed);
    }
  }
  return [...new Set(rooms)];
}

/*
 * SSE Server 事件总线（基于 emittery）
 */

import Emittery from 'emittery';

import type {
  ISseServerEventHandler,
  ISseServerEventMap,
  ISseServerEventName
} from '../types';

import { SSE_SERVER_DEFAULTS, sseServerMsg } from '../constants';

type IEmitteryListener = (event: { data?: unknown }) => void | Promise<void>;

export class SseServerEventBus {
  private readonly emitter = new Emittery<ISseServerEventMap>({
    debug: { name: SSE_SERVER_DEFAULTS.EVENT_BUS_DEBUG_NAME }
  });

  private readonly listenerMap = new WeakMap<
    ISseServerEventHandler<any>,
    IEmitteryListener
  >();

  /**
   * 订阅事件
   * @returns 取消订阅函数
   */
  on<E extends ISseServerEventName>(
    event: E,
    handler: ISseServerEventHandler<E>
  ): () => void {
    const listener = this.wrapHandler(handler);
    return this.emitter.on(event, listener as any);
  }

  /**
   * 订阅一次
   * @returns 取消订阅函数
   */
  once<E extends ISseServerEventName>(
    event: E,
    handler: ISseServerEventHandler<E>
  ): () => void {
    const off = this.on(event, ((data: ISseServerEventMap[E]) => {
      off();
      return handler(data);
    }) as ISseServerEventHandler<E>);
    return off;
  }

  /**
   * 移除事件处理器
   */
  off<E extends ISseServerEventName>(
    event: E,
    handler: ISseServerEventHandler<E>
  ): void {
    const listener = this.listenerMap.get(handler);
    if (!listener) {
      return;
    }
    this.emitter.off(event, listener as any);
  }

  /**
   * 异步派发事件
   */
  emit<E extends ISseServerEventName>(
    event: E,
    ...args: ISseServerEventMap[E] extends undefined
      ? [] | [undefined]
      : [ISseServerEventMap[E]]
  ): void {
    const data = args[0] as ISseServerEventMap[E];
    void this.emitter.emit(event, data as any).catch((error) => {
      logger.sseServer.error(sseServerMsg('事件派发失败'), { event, error });
    });
  }

  /**
   * 清空监听
   * @param event 指定事件；不传则清空全部
   */
  removeAll(event?: ISseServerEventName): void {
    if (event) {
      this.emitter.clearListeners(event);
      return;
    }
    this.emitter.clearListeners();
  }

  private wrapHandler<E extends ISseServerEventName>(
    handler: ISseServerEventHandler<E>
  ): IEmitteryListener {
    const cached = this.listenerMap.get(handler);
    if (cached) {
      return cached;
    }

    const listener: IEmitteryListener = (event) => {
      return handler(event.data as ISseServerEventMap[E]);
    };
    this.listenerMap.set(handler, listener);
    return listener;
  }
}

import { sseServer } from './plugins/sse';

await sseServer.send(userId, { event: 'notice', data: { tip: '已支付' } });
await sseServer.broadcast({ event: 'notice', data: { tip: '系统维护' } });

sseServer.joinRoom(userId, `order:${orderId}`);
await sseServer.toRoom(`order:${orderId}`, {
  event: 'order.status',
  data: { status: 'paid' }
});

注意事项 ​

  • 业务一律用单例 sseServer，不要再 new SseService（测试可用 resetSseService）
  • 必须传 httpServer，且要在 Koa callback 挂到 server 之后启动（attach 会接管 request 监听器）
  • ENV_SSE_ENABLE=false 时 start 静默跳过
  • authEnable=true 且未 setAuthenticate 会拒绝全部连接
  • onConnect / on 返回的 off 须在模块卸载时调用
  • 同 clientId 重连会踢掉旧连接
  • SSE 单向，客户端无法上行；需要双向用 plugins/websocket
  • Last-Event-ID 已滚出历史缓冲则不重放，业务要能容忍空洞
  • 浏览器对同域 SSE 连接数有限制（HTTP/1.1 下约 6 条）
  • Nginx 反代需 proxy_buffering off（代码已下发 X-Accel-Buffering: no）
  • 日志 / 错误文案前缀统一走 SSE_SERVER_DEFAULTS.MODULE_NAME（sseServerMsg / sseMsg），改名只改 constants
  • 进程退出用 registerGracefulShutdown，任务里调 stopSsePlugin()
