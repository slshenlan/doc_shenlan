/*
 * WebSocket Server 门面服务
 * 业务侧唯一入口：组装 Manager / Handler，对外暴露稳定 API
 */

import type { IWsServerPluginConfig } from "../config/types";
import type {
  IWsAuthenticateHandler,
  IWsBroadcastFilter,
  IWsEnvelope,
  IWsEventHandler,
  IWsMiddleware,
  IWsSendOptions,
  IWsServerEventHandler,
  IWsServerEventName,
  IWsServerPayload,
  IWsServerService,
  IWsServerStartOptions,
} from "../types";

import { resolveWsServerConfig } from "../config";
import { WsServerEvent, wsServerMsg } from "../constants";
import { WsServerNotEnabledError, WsServerNotRunningError } from "../errors";
import { WsServerMessageHandler } from "../handler";
import { WsServerManager } from "../manager";

export class WebSocketService implements IWsServerService {
  private config: IWsServerPluginConfig;
  private manager: WsServerManager;
  private messageHandler: WsServerMessageHandler;
  private started = false;
  private messageUnbind: (() => void) | null = null;

  /**
   * 构造门面：解析配置并创建 Manager / Handler
   * @param overrides 可选配置覆盖（一般留给测试）
   */
  constructor(overrides: Partial<IWsServerPluginConfig> = {}) {
    this.config = resolveWsServerConfig(overrides);
    this.manager = new WsServerManager(this.config);
    this.messageHandler = new WsServerMessageHandler();
  }

  /**
   * 启动 WebSocket Server
   * @param options.httpServer 传入 Koa http.Server 以挂载
   */
  async start(options: IWsServerStartOptions = {}) {
    if (this.started && this.manager.isRunning()) {
      logger.wsServer.debug(wsServerMsg("已启动"));
      return;
    }

    this.config = resolveWsServerConfig(this.config);
    this.manager.updateConfig(this.config);

    if (!this.config.enable) {
      logger.wsServer.warn(`ENV_WS_ENABLE=false，${wsServerMsg("未启动")}`);
      return;
    }

    this.bindMessagePipeline();
    await this.manager.start(options.httpServer);
    this.started = true;
  }

  /**
   * 停止 WebSocket Server
   */
  async stop() {
    this.messageUnbind?.();
    this.messageUnbind = null;
    this.messageHandler.clear();
    await this.manager.stop();
    this.started = false;
  }

  isRunning() {
    return this.manager.isRunning();
  }

  getState() {
    return this.manager.getState();
  }

  health() {
    return this.manager.health();
  }

  /**
   * 向指定客户端发送消息
   * @param clientId 客户端 ID
   * @param payload 载荷 / 信封
   * @param options 发送选项
   */
  async send(
    clientId: string,
    payload: IWsServerPayload | IWsEnvelope,
    options?: IWsSendOptions,
  ) {
    this.assertEnabled();
    this.assertRunning();
    return this.manager.send(clientId, payload, options);
  }

  /**
   * 广播消息
   * @param payload 载荷 / 信封
   * @param filter 可选过滤函数
   * @param options 发送选项
   */
  async broadcast(
    payload: IWsServerPayload | IWsEnvelope,
    filter?: IWsBroadcastFilter,
    options?: IWsSendOptions,
  ) {
    this.assertEnabled();
    this.assertRunning();
    return this.manager.broadcast(payload, filter, options);
  }

  /**
   * 向房间广播
   * @param room 房间名
   * @param payload 载荷 / 信封
   * @param options 发送选项
   */
  async toRoom(
    room: string,
    payload: IWsServerPayload | IWsEnvelope,
    options?: IWsSendOptions,
  ) {
    this.assertEnabled();
    this.assertRunning();
    return this.manager.toRoom(room, payload, options);
  }

  /**
   * 注册 event 处理器
   * @returns 取消注册函数
   */
  onEvent(event: string, handler: IWsEventHandler) {
    return this.messageHandler.onEvent(event, handler);
  }

  /**
   * 注册消息中间件
   */
  use(middleware: IWsMiddleware) {
    this.messageHandler.use(middleware);
  }

  /**
   * 设置自定义握手认证（覆盖默认策略）
   */
  setAuthenticate(handler: IWsAuthenticateHandler) {
    this.manager.auth.setAuthenticate(handler);
  }

  listClients() {
    return this.manager.clients.list();
  }

  /**
   * 按 ID 查询客户端信息
   * @param clientId 客户端 ID
   */
  getClient(clientId: string) {
    return this.manager.clients.get(clientId);
  }

  /**
   * 主动断开指定客户端
   * @param clientId 客户端 ID
   * @param code 关闭码
   * @param reason 原因
   */
  async disconnectClient(clientId: string, code?: number, reason?: string) {
    this.assertRunning();
    return this.manager.disconnectClient(clientId, code, reason);
  }

  /**
   * 加入房间
   * @param clientId 客户端 ID
   * @param room 房间名
   */
  joinRoom(clientId: string, room: string) {
    this.assertRunning();
    return this.manager.joinRoom(clientId, room);
  }

  /**
   * 离开房间
   * @param clientId 客户端 ID
   * @param room 房间名
   */
  leaveRoom(clientId: string, room: string) {
    this.assertRunning();
    return this.manager.leaveRoom(clientId, room);
  }

  /**
   * 获取房间内客户端
   * @param room 房间名
   */
  getRoomClients(room: string) {
    return this.manager.clients.getRoomClients(room);
  }

  /** 列出全部房间 */
  listRooms() {
    return this.manager.clients.listRooms();
  }

  on<E extends IWsServerEventName>(
    event: E,
    handler: IWsServerEventHandler<E>,
  ) {
    return this.manager.on(event, handler);
  }

  once<E extends IWsServerEventName>(
    event: E,
    handler: IWsServerEventHandler<E>,
  ) {
    return this.manager.once(event, handler);
  }

  off<E extends IWsServerEventName>(
    event: E,
    handler: IWsServerEventHandler<E>,
  ) {
    this.manager.off(event, handler);
  }

  getWss() {
    return this.manager.getWss();
  }

  getHttpServer() {
    return this.manager.getHttpServer();
  }

  /**
   * 将 Manager 的 Message 事件接到 Handler 消息链
   */
  private bindMessagePipeline() {
    this.messageUnbind?.();
    this.messageUnbind = this.manager.on(WsServerEvent.Message, ({ ctx }) => {
      void this.messageHandler.handle(ctx);
    });
  }

  private assertEnabled() {
    if (!this.config.enable) {
      throw new WsServerNotEnabledError();
    }
  }

  private assertRunning() {
    if (!this.manager.isRunning()) {
      throw new WsServerNotRunningError();
    }
  }
}

let wsServerSingleton: WebSocketService | null = null;

/**
 * 获取 WebSocket Server 单例
 * @param overrides 仅首次创建时生效
 */
export function getWebSocketService(
  overrides: Partial<IWsServerPluginConfig> = {},
) {
  if (!wsServerSingleton) {
    wsServerSingleton = new WebSocketService(overrides);
  }
  return wsServerSingleton;
}

/**
 * 重置单例（主要用于测试）
 */
export function resetWebSocketService() {
  wsServerSingleton = null;
}

/*
 * WebSocket 连接与传输管理
 * 负责：创建 WSS、握手认证、心跳、消息解码、房间、状态与 health
 */

import type { Server as HttpServer, IncomingMessage } from "node:http";
import type { Duplex } from "node:stream";

import { randomUUID } from "node:crypto";
import { type RawData, WebSocket, type WebSocketServer } from "ws";

import type { IWsServerPluginConfig } from "../config/types";
import type {
  IWsBroadcastFilter,
  IWsEnvelope,
  IWsMessageContext,
  IWsSendOptions,
  IWsServerEventHandler,
  IWsServerEventMap,
  IWsServerEventName,
  IWsServerHealth,
  IWsServerPayload,
  IWsServerState,
} from "../types";

import { WsServerAuthService } from "../auth";
import {
  WS_SERVER_DEFAULTS,
  WsBuiltinEvent,
  WsCloseCode,
  wsMsg,
  WsServerEvent,
  wsServerMsg,
  WsServerState,
} from "../constants";
import { WsServerClientNotFoundError, WsServerSendError } from "../errors";
import { createWebSocketServer } from "../server";
import { WsServerEventBus } from "../utils/event-bus";
import { decodeWsText, encodeWsPayload } from "../utils/payload";
import { ClientRegistry } from "./clients";

export class WsServerManager {
  private wss: WebSocketServer | null = null;
  private httpServer: HttpServer | null = null;
  private state: IWsServerState = WsServerState.Idle;
  private startedAt: number | null = null;
  private lastError: string | null = null;
  private messageCount = 0;
  private heartbeatTimer: ReturnType<typeof setInterval> | null = null;
  private readonly events = new WsServerEventBus();
  /** 在线客户端与房间 */
  readonly clients = new ClientRegistry();
  /** 握手认证 */
  readonly auth: WsServerAuthService;

  constructor(private config: IWsServerPluginConfig) {
    this.auth = new WsServerAuthService(config);
  }

  getConfig() {
    return this.config;
  }

  /**
   * 热更新配置（启动前可调用）
   * @param config 新配置
   */
  updateConfig(config: IWsServerPluginConfig) {
    this.config = config;
    this.auth.updateConfig(config);
  }

  getWss() {
    return this.wss;
  }

  getHttpServer() {
    return this.httpServer;
  }

  getState() {
    return this.state;
  }

  isRunning() {
    return this.state === WsServerState.Running && !!this.wss;
  }

  on<E extends IWsServerEventName>(
    event: E,
    handler: IWsServerEventHandler<E>,
  ) {
    return this.events.on(event, handler);
  }

  once<E extends IWsServerEventName>(
    event: E,
    handler: IWsServerEventHandler<E>,
  ) {
    return this.events.once(event, handler);
  }

  off<E extends IWsServerEventName>(
    event: E,
    handler: IWsServerEventHandler<E>,
  ) {
    this.events.off(event, handler);
  }

  emit<E extends IWsServerEventName>(
    event: E,
    ...args: IWsServerEventMap[E] extends undefined
      ? [] | [undefined]
      : [IWsServerEventMap[E]]
  ) {
    this.events.emit(event, ...(args as any));
  }

  /**
   * 启动 WebSocket Server
   * @param httpServer Koa 的 http.Server；不传则无法挂载
   */
  async start(httpServer?: HttpServer) {
    if (!this.config.enable) {
      logger.wsServer.warn(wsServerMsg("未启用，跳过启动"));
      return;
    }

    if (this.isRunning()) {
      logger.wsServer.debug(wsServerMsg("已在运行"));
      return;
    }

    if (!httpServer) {
      throw new Error(wsServerMsg("需要传入 httpServer 才能挂载"));
    }

    this.setState(WsServerState.Starting);
    this.events.emit(WsServerEvent.Starting);
    this.httpServer = httpServer;
    this.startedAt = Date.now();

    try {
      this.wss = createWebSocketServer(httpServer, this.config);
      this.bindServerEvents(this.wss);
      this.startHeartbeat();

      this.setState(WsServerState.Running);
      this.lastError = null;
      this.events.emit(WsServerEvent.Ready, this.health());
      logger.wsServer.info(wsServerMsg("启动完成"), this.health());
    } catch (error) {
      this.lastError = (error as Error).message;
      this.setState(WsServerState.Error);
      this.events.emit(WsServerEvent.Error, { error: error as Error });
      throw error;
    }
  }

  /**
   * 停止服务：关连接 → 停心跳 → close WSS → 清空注册表
   */
  async stop() {
    if (
      this.state === WsServerState.Idle ||
      this.state === WsServerState.Stopped
    ) {
      return;
    }

    this.setState(WsServerState.Stopping);
    this.stopHeartbeat();

    this.clients.forEach((entry) => {
      try {
        if (entry.socket.readyState === WebSocket.OPEN) {
          entry.socket.close(WsCloseCode.GoingAway, "server shutting down");
        }
      } catch (error) {
        logger.wsServer.warn("关闭客户端连接失败", {
          clientId: entry.id,
          error,
        });
      }
    });

    if (this.wss) {
      await new Promise<void>((resolve) => {
        const timer = setTimeout(() => {
          logger.wsServer.warn(wsServerMsg("关闭超时"));
          resolve();
        }, this.config.shutdownTimeout);

        this.wss!.close(() => {
          clearTimeout(timer);
          resolve();
        });
      });
      this.wss = null;
    }

    this.clients.clear();
    this.httpServer = null;
    this.setState(WsServerState.Stopped);
    this.events.emit(WsServerEvent.Stopped);
    logger.wsServer.info(wsServerMsg("已停止"));
  }

  /**
   * 向指定客户端发送消息
   * @param clientId 客户端 ID
   * @param payload 载荷
   * @param options 发送选项
   */
  async send(
    clientId: string,
    payload: IWsServerPayload | IWsEnvelope,
    options: IWsSendOptions = {},
  ) {
    const entry = this.clients.getEntry(clientId);
    if (!entry) {
      throw new WsServerClientNotFoundError(clientId);
    }
    await this.sendToSocket(entry.socket, payload, options, clientId);
  }

  /**
   * 广播消息
   * @param payload 载荷
   * @param filter 可选过滤
   * @param options 发送选项
   */
  async broadcast(
    payload: IWsServerPayload | IWsEnvelope,
    filter?: IWsBroadcastFilter,
    options: IWsSendOptions = {},
  ) {
    const tasks: Array<Promise<void>> = [];
    this.clients.forEach((entry) => {
      if (filter && !filter(entry.info)) {
        return;
      }
      tasks.push(this.sendToSocket(entry.socket, payload, options, entry.id));
    });
    await Promise.allSettled(tasks);
  }

  /**
   * 向房间广播
   * @param room 房间名
   * @param payload 载荷
   * @param options 发送选项
   */
  async toRoom(
    room: string,
    payload: IWsServerPayload | IWsEnvelope,
    options: IWsSendOptions = {},
  ) {
    if (!room) {
      throw new WsServerSendError("room 不能为空");
    }
    const entries = this.clients.getRoomEntries(room);
    await Promise.allSettled(
      entries.map((entry) => {
        return this.sendToSocket(entry.socket, payload, options, entry.id);
      }),
    );
  }

  /**
   * 加入房间
   * @param clientId 客户端 ID
   * @param room 房间名
   */
  joinRoom(clientId: string, room: string): boolean {
    const ok = this.clients.joinRoom(clientId, room);
    if (ok) {
      this.events.emit(WsServerEvent.JoinRoom, { clientId, room });
      logger.wsServer.info("客户端加入房间", { clientId, room });
    }
    return ok;
  }

  /**
   * 离开房间
   * @param clientId 客户端 ID
   * @param room 房间名
   */
  leaveRoom(clientId: string, room: string): boolean {
    const ok = this.clients.leaveRoom(clientId, room);
    if (ok) {
      this.events.emit(WsServerEvent.LeaveRoom, { clientId, room });
      logger.wsServer.info("客户端离开房间", { clientId, room });
    }
    return ok;
  }

  /**
   * 主动断开客户端
   * @param clientId 客户端 ID
   * @param code 关闭码
   * @param reason 原因
   */
  async disconnectClient(
    clientId: string,
    code: number = WsCloseCode.Normal,
    reason = "disconnected by server",
  ) {
    const entry = this.clients.getEntry(clientId);
    if (!entry) {
      return;
    }

    await new Promise<void>((resolve) => {
      entry.socket.once("close", () => {
        resolve();
      });
      try {
        entry.socket.close(code, reason);
      } catch {
        resolve();
      }
    });
  }

  health(): IWsServerHealth {
    return {
      enabled: this.config.enable,
      state: this.state,
      path: this.config.path,
      attached: !!this.wss,
      connectedClients: this.clients.size(),
      roomCount: this.clients.roomCount(),
      messageCount: this.messageCount,
      lastError: this.lastError,
      uptimeMs: this.startedAt ? Date.now() - this.startedAt : 0,
    };
  }

  private setState(next: IWsServerState) {
    if (this.state === next) {
      return;
    }
    const prev = this.state;
    this.state = next;
    logger.wsServer.debug(wsServerMsg("状态变更"), { from: prev, to: next });
    this.events.emit(WsServerEvent.StateChange, { next, prev });
  }

  private bindServerEvents(wss: WebSocketServer) {
    wss.on("connection", (socket, request) => {
      void this.handleConnection(socket, request);
    });

    wss.on("error", (error) => {
      this.lastError = error.message;
      logger.wsServer.error(wsServerMsg("错误"), error);
      this.events.emit(WsServerEvent.Error, { error });
    });

    wss.on("headers", (_headers, request) => {
      logger.wsServer.debug(wsMsg("升级握手"), {
        url: request.url,
        remoteAddress: request.socket.remoteAddress,
      });
    });
  }

  private async handleConnection(socket: WebSocket, request: IncomingMessage) {
    const remoteAddress = request.socket.remoteAddress;
    let clientId = "";

    try {
      const url = new URL(request.url || "/", "http://localhost");
      const protocolsHeader = request.headers["sec-websocket-protocol"];
      const protocols = protocolsHeader
        ? String(protocolsHeader)
            .split(",")
            .map((item) => {
              return item.trim();
            })
            .filter(Boolean)
        : [];

      const authResult = await this.auth.authenticate({
        url,
        headers: request.headers,
        protocols,
        request,
        remoteAddress,
      });

      if (!authResult.ok) {
        logger.wsServer.warn(wsMsg("握手认证失败"), {
          reason: authResult.reason,
          remoteAddress,
        });
        socket.close(
          WsCloseCode.PolicyViolation,
          authResult.reason || "unauthorized",
        );
        return;
      }

      clientId =
        authResult.clientId ||
        url.searchParams.get("clientId") ||
        `${WS_SERVER_DEFAULTS.CLIENT_ID_PREFIX}-${randomUUID()}`;

      // 同 ID 重连：踢掉旧连接
      const existing = this.clients.getEntry(clientId);
      if (existing) {
        logger.wsServer.warn("客户端 ID 冲突，断开旧连接", { clientId });
        try {
          existing.socket.close(
            WsCloseCode.PolicyViolation,
            "replaced by new connection",
          );
        } catch {
          // ignore
        }
        this.clients.remove(clientId);
      }

      const info = this.clients.add(clientId, socket, {
        remoteAddress,
        meta: authResult.meta,
      });

      this.bindSocketEvents(socket, clientId);
      this.events.emit(WsServerEvent.Connection, info);
      logger.wsServer.info("客户端已连接", {
        clientId,
        remoteAddress,
        connectedClients: this.clients.size(),
      });
    } catch (error) {
      this.lastError = (error as Error).message;
      logger.wsServer.error(wsMsg("处理连接失败"), error);
      this.events.emit(WsServerEvent.Error, {
        error: error as Error,
        clientId: clientId || undefined,
      });
      try {
        socket.close(WsCloseCode.InternalError, "internal error");
      } catch {
        // ignore
      }
    }
  }

  private bindSocketEvents(socket: WebSocket, clientId: string) {
    socket.on("message", (data, isBinary) => {
      void this.handleMessage(clientId, data, isBinary);
    });

    socket.on("pong", () => {
      this.clients.touchPong(clientId);
    });

    socket.on("close", (code, reasonBuf) => {
      const info = this.clients.remove(clientId);
      const reason = reasonBuf?.toString("utf8") || "";
      logger.wsServer.info("客户端已断开", {
        clientId,
        code,
        reason,
        connectedClients: this.clients.size(),
      });
      if (info) {
        this.events.emit(WsServerEvent.Disconnect, info);
      }
    });

    socket.on("error", (error) => {
      this.lastError = error.message;
      logger.wsServer.error("客户端 Socket 错误", { clientId, error });
      this.events.emit(WsServerEvent.Error, { error, clientId });
    });
  }

  private async handleMessage(
    clientId: string,
    data: RawData,
    isBinary: boolean,
  ) {
    const entry = this.clients.getEntry(clientId);
    if (!entry) {
      return;
    }

    this.clients.touchPong(clientId);
    this.messageCount += 1;

    let event: string;
    let messageData: unknown;
    let envelope: IWsEnvelope | null = null;
    let raw: string | Buffer;

    if (isBinary || Buffer.isBuffer(data)) {
      const buf = Buffer.isBuffer(data)
        ? data
        : Buffer.from(data as ArrayBuffer);
      raw = buf;
      event = WsBuiltinEvent.Binary;
      messageData = buf;
    } else {
      raw = data.toString();
      const decoded = decodeWsText(raw);
      event = decoded.event;
      messageData = decoded.data;
      envelope = decoded.envelope;
    }

    // 内置 ping/pong
    if (event === WsBuiltinEvent.Ping) {
      await this.sendToSocket(
        entry.socket,
        {
          event: WsBuiltinEvent.Pong,
          data: messageData,
          id: envelope?.id,
        },
        {},
        clientId,
      );
      return;
    }

    // 客户端通过协议加入/离开房间
    if (event === WsBuiltinEvent.JoinRoom) {
      const room = this.extractRoom(messageData);
      if (room) {
        this.joinRoom(clientId, room);
      }
      return;
    }
    if (event === WsBuiltinEvent.LeaveRoom) {
      const room = this.extractRoom(messageData);
      if (room) {
        this.leaveRoom(clientId, room);
      }
      return;
    }

    const ctx: IWsMessageContext = {
      event,
      data: messageData,
      raw,
      envelope,
      clientId,
      rooms: [...entry.info.rooms],
      meta: { ...entry.info.meta },
      receivedAt: Date.now(),
      state: {},
    };

    logger.wsServer.debug(wsMsg("收到消息"), {
      event: ctx.event,
      clientId: ctx.clientId,
    });

    this.events.emit(WsServerEvent.Message, { ctx });
  }

  private extractRoom(data: unknown): string | null {
    if (typeof data === "string" && data.trim()) {
      return data.trim();
    }
    if (data && typeof data === "object" && !Array.isArray(data)) {
      const room = (data as { room?: unknown }).room;
      if (typeof room === "string" && room.trim()) {
        return room.trim();
      }
    }
    return null;
  }

  private async sendToSocket(
    socket: WebSocket,
    payload: IWsServerPayload | IWsEnvelope,
    options: IWsSendOptions,
    clientId: string,
  ) {
    if (socket.readyState !== WebSocket.OPEN) {
      throw new WsServerSendError("客户端未处于 OPEN 状态", { clientId });
    }

    const body = encodeWsPayload(payload);

    await new Promise<void>((resolve, reject) => {
      socket.send(
        body,
        { binary: options.binary ?? Buffer.isBuffer(body) },
        (error) => {
          if (error) {
            logger.wsServer.error(wsMsg("发送失败"), { clientId, error });
            reject(new WsServerSendError(error.message, { clientId }));
            return;
          }
          resolve();
        },
      );
    });
  }

  private startHeartbeat() {
    this.stopHeartbeat();
    if (this.config.heartbeatInterval <= 0) {
      return;
    }

    this.heartbeatTimer = setInterval(() => {
      const now = Date.now();
      this.clients.forEach((entry) => {
        if (now - entry.lastPongAt > this.config.heartbeatTimeout) {
          logger.wsServer.warn("心跳超时，断开客户端", {
            clientId: entry.id,
            lastPongAt: entry.lastPongAt,
          });
          try {
            entry.socket.terminate();
          } catch (error) {
            logger.wsServer.error("terminate 客户端失败", {
              clientId: entry.id,
              error,
            });
          }
          return;
        }

        if (entry.socket.readyState === WebSocket.OPEN) {
          try {
            entry.socket.ping();
          } catch (error) {
            logger.wsServer.warn("发送 ping 失败", {
              clientId: entry.id,
              error,
            });
          }
        }
      });
    }, this.config.heartbeatInterval);

    // 不阻止进程退出
    if (typeof this.heartbeatTimer.unref === "function") {
      this.heartbeatTimer.unref();
    }
  }

  private stopHeartbeat() {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer);
      this.heartbeatTimer = null;
    }
  }
}

/*
 * 在线客户端与房间注册表
 */

import type { WebSocket } from "ws";

import type { IWsClientEntry, IWsClientInfo } from "../types";

export class ClientRegistry {
  private clients = new Map<string, IWsClientEntry>();
  /** room -> clientId set */
  private rooms = new Map<string, Set<string>>();

  /**
   * 登记客户端上线
   * @param id 客户端 ID
   * @param socket WebSocket 句柄
   * @param options remoteAddress / meta
   */
  add(
    id: string,
    socket: WebSocket,
    options: { remoteAddress?: string; meta?: Record<string, unknown> } = {},
  ): IWsClientInfo {
    const info: IWsClientInfo = {
      id,
      connected: true,
      connectedAt: Date.now(),
      rooms: [],
      remoteAddress: options.remoteAddress,
      meta: options.meta || {},
    };

    this.clients.set(id, {
      id,
      socket,
      info,
      lastPongAt: Date.now(),
    });

    return { ...info, rooms: [...info.rooms], meta: { ...info.meta } };
  }

  /**
   * 移除已断开的客户端，并清理其房间成员关系
   * @param clientId 客户端 ID
   */
  remove(clientId: string): IWsClientInfo | null {
    const entry = this.clients.get(clientId);
    if (!entry) {
      return null;
    }

    for (const room of [...entry.info.rooms]) {
      this.leaveRoom(clientId, room);
    }

    this.clients.delete(clientId);
    return {
      ...entry.info,
      connected: false,
      rooms: [...entry.info.rooms],
      meta: { ...entry.info.meta },
    };
  }

  /**
   * 获取客户端对外信息
   * @param clientId 客户端 ID
   */
  get(clientId: string): IWsClientInfo | null {
    const entry = this.clients.get(clientId);
    if (!entry) {
      return null;
    }
    return {
      ...entry.info,
      rooms: [...entry.info.rooms],
      meta: { ...entry.info.meta },
    };
  }

  /**
   * 获取注册表原始条目（含 socket）
   * @param clientId 客户端 ID
   */
  getEntry(clientId: string): IWsClientEntry | null {
    return this.clients.get(clientId) || null;
  }

  /**
   * 列出全部在线客户端信息
   */
  list(): IWsClientInfo[] {
    const result: IWsClientInfo[] = [];
    for (const entry of this.clients.values()) {
      result.push({
        ...entry.info,
        rooms: [...entry.info.rooms],
        meta: { ...entry.info.meta },
      });
    }
    return result;
  }

  /** 当前在线数量 */
  size() {
    return this.clients.size;
  }

  /**
   * 加入房间
   * @param clientId 客户端 ID
   * @param room 房间名
   * @returns 是否成功加入（已在房间内也返回 true）
   */
  joinRoom(clientId: string, room: string): boolean {
    const entry = this.clients.get(clientId);
    if (!entry || !room) {
      return false;
    }

    if (!entry.info.rooms.includes(room)) {
      entry.info.rooms.push(room);
    }

    if (!this.rooms.has(room)) {
      this.rooms.set(room, new Set());
    }
    this.rooms.get(room)!.add(clientId);
    return true;
  }

  /**
   * 离开房间
   * @param clientId 客户端 ID
   * @param room 房间名
   */
  leaveRoom(clientId: string, room: string): boolean {
    const entry = this.clients.get(clientId);
    if (!entry || !room) {
      return false;
    }

    entry.info.rooms = entry.info.rooms.filter((item) => {
      return item !== room;
    });

    const members = this.rooms.get(room);
    if (members) {
      members.delete(clientId);
      if (members.size === 0) {
        this.rooms.delete(room);
      }
    }
    return true;
  }

  /**
   * 获取房间内客户端信息列表
   * @param room 房间名
   */
  getRoomClients(room: string): IWsClientInfo[] {
    const members = this.rooms.get(room);
    if (!members) {
      return [];
    }

    const result: IWsClientInfo[] = [];
    for (const clientId of members) {
      const info = this.get(clientId);
      if (info) {
        result.push(info);
      }
    }
    return result;
  }

  /**
   * 获取房间内客户端条目（含 socket）
   * @param room 房间名
   */
  getRoomEntries(room: string): IWsClientEntry[] {
    const members = this.rooms.get(room);
    if (!members) {
      return [];
    }

    const result: IWsClientEntry[] = [];
    for (const clientId of members) {
      const entry = this.clients.get(clientId);
      if (entry) {
        result.push(entry);
      }
    }
    return result;
  }

  /** 列出全部房间名 */
  listRooms(): string[] {
    return [...this.rooms.keys()];
  }

  /** 房间数量 */
  roomCount() {
    return this.rooms.size;
  }

  /**
   * 更新心跳时间戳
   * @param clientId 客户端 ID
   */
  touchPong(clientId: string) {
    const entry = this.clients.get(clientId);
    if (entry) {
      entry.lastPongAt = Date.now();
    }
  }

  /** 遍历全部条目 */
  forEach(fn: (entry: IWsClientEntry) => void) {
    for (const entry of this.clients.values()) {
      fn(entry);
    }
  }

  /** 清空注册表 */
  clear() {
    this.clients.clear();
    this.rooms.clear();
  }
}

/**
 * 创建并挂载 WebSocketServer
 * @param httpServer Koa 的 http.Server
 * @param config 插件配置（path / maxPayload）
 * @returns WebSocketServer 实例
 */
export function createWebSocketServer(
  httpServer: HttpServer,
  config: IWsServerPluginConfig,
): WebSocketServer {
  const wss = new WebSocketServer({
    server: httpServer,
    path: config.path,
    maxPayload: config.maxPayload,
    clientTracking: false,
  });

  const addr = httpServer.address();
  let address = config.path;
  if (addr && typeof addr !== "string") {
    const connectHost =
      addr.address === "0.0.0.0" || addr.address === "::"
        ? "localhost"
        : addr.address;
    const hostPart = connectHost.includes(":")
      ? `[${connectHost}]`
      : connectHost;
    address = `ws://${hostPart}:${addr.port}${config.path}`;
  }

  logger.wsServer.info(wsServerMsg("已挂载"), {
    path: config.path,
    address,
    maxPayload: config.maxPayload,
  });

  return wss;
}

/*
 * WebSocket 消息处理链
 * Koa 风格：日志中间件 → 用户 middleware → event 路由
 */

import type {
  IWsEventHandler,
  IWsMessageContext,
  IWsMiddleware,
} from "../types";

import { wsMsg } from "../constants";

/**
 * 组合中间件为可执行的 dispatch 函数
 * @param middlewares 中间件数组（按注册顺序执行）
 */
export function composeWsMiddlewares(middlewares: IWsMiddleware[]) {
  return async function dispatch(ctx: IWsMessageContext): Promise<void> {
    let index = -1;

    async function next(i: number): Promise<void> {
      if (i <= index) {
        throw new Error("next() 被重复调用");
      }
      index = i;
      const fn = middlewares[i];
      if (!fn) {
        return;
      }
      await fn(ctx, () => {
        return next(i + 1);
      });
    }

    await next(0);
  };
}

export class WsServerMessageHandler {
  private middlewares: IWsMiddleware[] = [];
  private eventHandlers = new Map<string, Set<IWsEventHandler>>();
  private runner: (ctx: IWsMessageContext) => Promise<void>;
  private readonly routeMiddleware: IWsMiddleware;

  constructor() {
    this.routeMiddleware = async (ctx, next) => {
      const set = this.eventHandlers.get(ctx.event);
      if (set) {
        for (const handler of set) {
          await this.invokeHandler(handler, ctx);
        }
      }
      await next();
    };

    this.middlewares.push(async (ctx, next) => {
      const start = Date.now();
      try {
        await next();
        logger.wsServer.debug(wsMsg("消息处理完成"), {
          event: ctx.event,
          clientId: ctx.clientId,
          costMs: Date.now() - start,
        });
      } catch (error) {
        logger.wsServer.error(wsMsg("消息处理失败"), {
          event: ctx.event,
          clientId: ctx.clientId,
          error,
          costMs: Date.now() - start,
        });
        throw error;
      }
    });

    this.runner = composeWsMiddlewares([]);
    this.rebuild();
  }

  /**
   * 注册用户中间件（插入在 event 路由之前）
   */
  use(middleware: IWsMiddleware) {
    this.middlewares.push(middleware);
    this.rebuild();
  }

  /**
   * 注册 event 处理器
   * @returns 取消注册函数
   */
  onEvent(event: string, handler: IWsEventHandler) {
    if (!this.eventHandlers.has(event)) {
      this.eventHandlers.set(event, new Set());
    }
    this.eventHandlers.get(event)!.add(handler);
    return () => {
      this.eventHandlers.get(event)?.delete(handler);
    };
  }

  /**
   * 执行整条消息处理链
   * @param ctx 入站消息上下文
   */
  async handle(ctx: IWsMessageContext) {
    await this.runner(ctx);
  }

  /** 清空全部 event 处理器 */
  clear() {
    this.eventHandlers.clear();
  }

  private rebuild() {
    this.runner = composeWsMiddlewares([
      ...this.middlewares,
      this.routeMiddleware,
    ]);
  }

  private async invokeHandler(
    handler: IWsEventHandler,
    ctx: IWsMessageContext,
  ) {
    try {
      await handler(ctx);
    } catch (error) {
      logger.wsServer.error(wsMsg("Event Handler 执行失败"), {
        event: ctx.event,
        clientId: ctx.clientId,
        error,
      });
    }
  }
}

      {
        url: params.url.toString(),

    );
    return { ok: false, reason: "未配置认证处理器" };
  }

/**
 * 将值转换为布尔值
 * @param value 值
 * @param fallback 默认值
 * @returns 布尔值
 */
function toBool(value: unknown, fallback: boolean): boolean {
  if (value === undefined || value === null || value === "") {
    return fallback;
  }
  if (typeof value === "boolean") {
    return value;
  }
  return String(value).toLowerCase() === "true";
}

/**
 * 将值转换为数字
 * @param value 值
 * @param fallback 默认值
 * @returns 数字
 */
function toNumber(value: unknown, fallback: number): number {
  const n = Number(value);
  return Number.isFinite(n) ? n : fallback;
}

/**

/*
 * WebSocket Server 事件总线（基于 emittery）
 */

import Emittery from "emittery";

import type {
  IWsServerEventHandler,
  IWsServerEventMap,
  IWsServerEventName,
} from "../types";

import { WS_SERVER_DEFAULTS, wsServerMsg } from "../constants";

type IEmitteryListener = (event: { data?: unknown }) => void | Promise<void>;

export class WsServerEventBus {
  private readonly emitter = new Emittery<IWsServerEventMap>({
    debug: { name: WS_SERVER_DEFAULTS.EVENT_BUS_DEBUG_NAME },
  });

  private readonly listenerMap = new WeakMap<
    IWsServerEventHandler<any>,
    IEmitteryListener
  >();

  /**
   * 订阅事件
   * @returns 取消订阅函数
   */
  on<E extends IWsServerEventName>(
    event: E,
    handler: IWsServerEventHandler<E>,
  ): () => void {
    const listener = this.wrapHandler(handler);
    return this.emitter.on(event, listener as any);
  }

  /**
   * 订阅一次
   * @returns 取消订阅函数
   */
  once<E extends IWsServerEventName>(
    event: E,
    handler: IWsServerEventHandler<E>,
  ): () => void {
    const off = this.on(event, ((data: IWsServerEventMap[E]) => {
      off();
      return handler(data);
    }) as IWsServerEventHandler<E>);
    return off;
  }

  /**
   * 移除事件处理器
   */
  off<E extends IWsServerEventName>(
    event: E,
    handler: IWsServerEventHandler<E>,
  ): void {
    const listener = this.listenerMap.get(handler);
    if (!listener) {
      return;
    }
    this.emitter.off(event, listener as any);
  }

  /**
   * 异步派发事件
   */
  emit<E extends IWsServerEventName>(
    event: E,
    ...args: IWsServerEventMap[E] extends undefined
      ? [] | [undefined]
      : [IWsServerEventMap[E]]
  ): void {
    const data = args[0] as IWsServerEventMap[E];
    void this.emitter.emit(event, data as any).catch((error) => {
      logger.wsServer.error(wsServerMsg("事件派发失败"), { event, error });
    });
  }

  /**
   * 清空监听
   * @param event 指定事件；不传则清空全部
   */
  removeAll(event?: IWsServerEventName): void {
    if (event) {
      this.emitter.clearListeners(event);
      return;
    }
    this.emitter.clearListeners();
  }

  private wrapHandler<E extends IWsServerEventName>(
    handler: IWsServerEventHandler<E>,
  ): IEmitteryListener {
    const cached = this.listenerMap.get(handler);
    if (cached) {
      return cached;
    }

    const listener: IEmitteryListener = (event) => {
      return handler(event.data as IWsServerEventMap[E]);
    };
    this.listenerMap.set(handler, listener);
    return listener;
  }
}

/*
 * WebSocket Server 载荷编解码
 */

import type { IWsEnvelope, IWsServerPayload } from "../types";

import { WsBuiltinEvent, wsMsg } from "../constants";

/**
 * 编码出站载荷
 * @param payload 字符串 / Buffer / 对象 / 信封
 * @returns 可直接 socket.send 的数据
 */
export function encodeWsPayload(
  payload: IWsServerPayload | IWsEnvelope,
): string | Buffer {
  if (Buffer.isBuffer(payload)) {
    return payload;
  }

  if (typeof payload === "string") {
    return payload;
  }

  try {
    return JSON.stringify(payload);
  } catch (error) {
    throw new Error(
      `${wsMsg("payload 序列化失败")}: ${(error as Error).message}`,
    );
  }
}

/**
 * 解码入站文本载荷为信封
 * @param raw 原始文本
 */
export function decodeWsText(raw: string): {
  event: string;
  data: unknown;
  envelope: IWsEnvelope | null;
} {
  const trimmed = raw.trim();
  if (!trimmed) {
    return { event: WsBuiltinEvent.Raw, data: raw, envelope: null };
  }

  if (
    (trimmed.startsWith("{") && trimmed.endsWith("}")) ||
    (trimmed.startsWith("[") && trimmed.endsWith("]"))
  ) {
    try {
      const parsed = JSON.parse(trimmed) as unknown;
      if (
        parsed &&
        typeof parsed === "object" &&
        !Array.isArray(parsed) &&
        typeof (parsed as IWsEnvelope).event === "string"
      ) {
        const envelope = parsed as IWsEnvelope;
        return {
          event: envelope.event,
          data: envelope.data,
          envelope,
        };
      }
      return { event: WsBuiltinEvent.Raw, data: parsed, envelope: null };
    } catch {
      return { event: WsBuiltinEvent.Raw, data: raw, envelope: null };
    }
  }

  return { event: WsBuiltinEvent.Raw, data: raw, envelope: null };
}

  }

注意事项 ​

  • 业务一律用单例 wsServer，不要再 new WebSocketService（测试可用 resetWebSocketService）
  • 必须传 httpServer；path 勿与 MQTT /mqtt 冲突
  • ENV_WS_ENABLE=false 时 start 静默跳过
  • authEnable=true 且未 setAuthenticate 会拒绝全部连接
  • onEvent / on 返回的 off 须在模块卸载时调用
  • 同 clientId 重连会踢掉旧连接
  • 消息统一用 { event, data, id? } 信封
  • 日志 / 错误文案前缀统一走 WS_SERVER_DEFAULTS.MODULE_NAME（wsServerMsg / wsMsg），改名只改 constants
  • 进程退出用 registerGracefulShutdown，任务里调 stopWebSocketPlugin()
  • 与 MQTT over WS 共用同一 HTTP Server 时注意 upgrade 冲突（详见 desc.md）
