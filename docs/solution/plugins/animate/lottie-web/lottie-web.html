插件安装 ​

 "lottie-web": "^5.12.2",

示例 ​

vue
<template>
  <div :id="animationDomId" class="lottie-dom">
    <p>{{ title }}</p>
  </div>
</template>
<script setup>
import { nextTick, ref } from 'vue';
import lottie from 'lottie-web';
import LoadingJson from './lottie/data.json';

const { proxy } = getCurrentInstance();
const { $pub } = proxy;
const animationDomId = $pub.slGenUid();

const jsonAnimation = ref('');
const title = ref('');

onMounted(() => {
  nextTick(() => {});
});

// 导出可以被父组件直接调用的函数
defineExpose({
  stop() {
    jsonAnimation.value.destroy();
  },
  play(item) {
    // TODO: 确认字段
    const { name } = item;
    title.value = name;
    jsonAnimation.value = lottie.loadAnimation({
      container: document.querySelector(`#${animationDomId}`), //进行播放的元素
      renderer: 'svg',
      loop: true, //循环播放
      autoplay: true, //自动播放
      animationData: LoadingJson //要播放的文件
    });
    // jsonAnimation.value.play();
  }
});
</script>
