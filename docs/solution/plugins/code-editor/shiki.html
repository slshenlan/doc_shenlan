使用 ​

vue
<template>
  <div class="sl-shiki" v-html="previewCode"></div>
</template>
<script setup>
import { ref } from 'vue';
import beautify from 'js-beautify';
import { codeToHtml } from 'shiki';

const beautifyCss = beautify.css;
const beautifyHtml = beautify.html;
const beautifyJs = beautify.js;
const { proxy } = getCurrentInstance();

// 定义传递属性
const props = defineProps({
  lang: {
    type: String,
    default: 'javascript'
  },
  code: {
    type: String,
    default: ''
  }
});

const previewCode = ref('');

// 监听其他属性、路由或者状态管理的都这样
watch(
  () => props.code,
  async (newCode) => {
    let code = newCode;

    switch (props.lang) {
      case 'css':
        code = beautifyCss(code, {});
        break;
      case 'javascript':
        code = beautifyJs(code, {});
        break;
    }

    previewCode.value = await codeToHtml(code, {
      lang: props.lang,
      theme: 'github-dark'
    });
  }
);
</script>
