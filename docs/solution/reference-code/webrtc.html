// 1v1

import { SignalClient } from "./signal-client";
import type { ISignalMessage } from "./types";

export function useWebRTC() {
  const localVideo = ref<HTMLVideoElement | null>(null);
  const remoteVideo = ref<HTMLVideoElement | null>(null);

  let pc: RTCPeerConnection;
  let localStream: MediaStream;

  const userId = Math.random().toString(36).slice(2);
  const roomId = "room1";

  let currentTargetId = "";

  const signal = new SignalClient("ws://192.168.1.49:8881"); // 改成你的IP

  // 初始化媒体
  async function initMedia() {
    localStream = await navigator.mediaDevices.getUserMedia({
      video: true,
      audio: true,
    });

    if (localVideo.value) {
      localVideo.value.srcObject = localStream;
    }
  }

  // 创建连接
  function createPeerConnection() {
    pc = new RTCPeerConnection({
      iceServers: [],
    });

    // 添加流
    localStream.getTracks().forEach((track) => {
      pc.addTrack(track, localStream);
    });

    // 远程流
    pc.ontrack = (event) => {
      if (remoteVideo.value) {
        remoteVideo.value.srcObject = event.streams[0];
      }
    };

    // ICE
    pc.onicecandidate = (event) => {
      if (event.candidate) {
        signal.send({
          type: "candidate",
          roomId,
          userId,
          targetId: currentTargetId,
          data: event.candidate,
        });
      }
    };
  }

  // 发起呼叫
  async function call(targetId: string) {
    currentTargetId = targetId;

    createPeerConnection();

    const offer = await pc.createOffer();
    await pc.setLocalDescription(offer);

    signal.send({
      type: "offer",
      roomId,
      userId,
      targetId,
      data: offer,
    });
  }

  // 处理信令
  signal.onMessage(async (msg: ISignalMessage) => {
    const { type, userId: fromId, data } = msg;

    switch (type) {
      case "join":
        // 有人加入，可自动发起连接
        break;

      case "offer":
        currentTargetId = fromId;

        createPeerConnection();

        await initMedia();

        await pc.setRemoteDescription(data);

        const answer = await pc.createAnswer();
        await pc.setLocalDescription(answer);

        signal.send({
          type: "answer",
          roomId,
          userId,
          targetId: fromId,
          data: answer,
        });
        break;

      case "answer":
        await pc.setRemoteDescription(data);
        break;

      case "candidate":
        await pc.addIceCandidate(data);
        break;
    }
  });

  // 加入房间
  function joinRoom() {
    signal.send({
      type: "join",
      roomId,
      userId,
    });
  }

  return {
    localVideo,
    remoteVideo,
    initMedia,
    joinRoom,
    call,
    userId,
  };
}

// 多人
import { SignalClient } from "./signal-client";
import type { ISignalMessage } from "./types";

interface IPeerContext {
  pc: RTCPeerConnection;
  targetId: string;
}

export function useWebRTC() {
  // 本机
  const userId = Math.random().toString(36).slice(2);
  // 房间
  const roomId = "room1";
  // 信令交换(WebSocket)
  const signal = new SignalClient("ws://192.168.1.49:8881");

  // 本地视频播放器
  const localVideo = ref<HTMLVideoElement | null>(null);

  // 远端视频流
  const remoteStreams = ref<Map<string, MediaStream>>(new Map());

  // PeerConnection
  const peers = new Map<string, IPeerContext>();

  // 本地流
  let localStream: MediaStream;

  /**
   * 信令处理
   */
  signal.onMessage(async (msg: ISignalMessage) => {
    const { type, userId: fromId, data } = msg;

    switch (type) {
      case "join":
        // 自动连接新加入的人
        if (fromId !== userId) {
          await call(fromId);
        }
        break;

      case "offer": {
        const { pc } = getOrCreatePeer(fromId);

        await pc.setRemoteDescription(data);

        const answer = await pc.createAnswer();
        await pc.setLocalDescription(answer);

        signal.send({
          type: "answer",
          roomId,
          userId,
          targetId: fromId,
          data: answer,
        });
        break;
      }

      case "answer": {
        const ctx = peers.get(fromId);
        if (!ctx) {
          return;
        }

        await ctx.pc.setRemoteDescription(data);
        break;
      }

      case "candidate": {
        const ctx = peers.get(fromId);
        if (!ctx) {
          return;
        }

        await ctx.pc.addIceCandidate(data);
        break;
      }

      case "leave":
        removePeer(fromId);
        break;
    }
  });

  /**
   * 初始化本地媒体
   */
  async function initMedia() {
    localStream = await navigator.mediaDevices.getUserMedia({
      video: true,
      audio: true,
    });

    if (localVideo.value) {
      localVideo.value.srcObject = localStream;
    }
  }

  /**
   * 获取或创建 PeerConnection
   */
  function getOrCreatePeer(targetId: string): IPeerContext {
    if (peers.has(targetId)) {
      return peers.get(targetId)!;
    }

    const pc = new RTCPeerConnection({
      iceServers: [],
    });

    // 本地流
    localStream.getTracks().forEach((track) => {
      pc.addTrack(track, localStream);
    });

    // 远端流
    pc.ontrack = (event) => {
      const stream = event.streams[0];
      remoteStreams.value.set(targetId, stream);
    };

    // ICE
    pc.onicecandidate = (event) => {
      if (event.candidate) {
        signal.send({
          type: "candidate",
          roomId,
          userId,
          targetId,
          data: event.candidate,
        });
      }
    };

    const ctx = { pc, targetId };
    peers.set(targetId, ctx);

    return ctx;
  }

  /**
   * 删除 peer
   */
  function removePeer(targetId: string) {
    const ctx = peers.get(targetId);
    if (!ctx) {
      return;
    }

    ctx.pc.close();
    peers.delete(targetId);
    remoteStreams.value.delete(targetId);
  }

  /**
   * 发起呼叫
   */
  async function call(targetId: string) {
    const { pc } = getOrCreatePeer(targetId);

    const offer = await pc.createOffer();
    await pc.setLocalDescription(offer);

    signal.send({
      type: "offer",
      roomId,
      userId,
      targetId,
      data: offer,
    });
  }

  /**
   * 加入房间
   */
  function joinRoom() {
    signal.send({
      type: "join",
      roomId,
      userId,
    });
  }

  return {
    userId,
    localVideo,
    remoteStreams,
    initMedia,
    joinRoom,
    call,
  };
}

import type { ISignalMessage } from "./types";

type MessageHandler = (msg: ISignalMessage) => void;

export class SignalClient {
  private ws: WebSocket;
  private handler?: MessageHandler;

  constructor(url: string) {
    this.ws = new WebSocket(url);

    this.ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      console.log(data);

      this.handler?.(data);
    };
  }

  onMessage(fn: MessageHandler) {
    this.handler = fn;
  }

  send(msg: ISignalMessage) {
    this.ws.send(JSON.stringify(msg));
  }
}

import WebSocket from "ws";

interface IClient {
  userId: string;
  ws: WebSocket;
}

export class RoomManager {
  private rooms = new Map<string, Map<string, IClient>>();

  // 加入房间
  join(roomId: string, client: IClient) {
    logger.stdout("加入房间", {
      roomId,
      client: client.userId,
    });
    if (!this.rooms.has(roomId)) {
      this.rooms.set(roomId, new Map());
    }
    this.rooms.get(roomId)!.set(client.userId, client);
  }

  // 离开房间
  leave(roomId: string, userId: string) {
    const room = this.rooms.get(roomId);
    if (!room) return;

    room.delete(userId);

    if (room.size === 0) {
      this.rooms.delete(roomId);
    }
  }

  // 获取房间用户
  getRoomUsers(roomId: string) {
    return this.rooms.get(roomId);
  }

  // 转发消息
  sendTo(roomId: string, targetId: string, message: any) {
    const room = this.rooms.get(roomId);
    const target = room?.get(targetId);

    if (target && target.ws.readyState === WebSocket.OPEN) {
      target.ws.send(JSON.stringify(message));
    }
  }

  // 广播（除自己）
  broadcast(roomId: string, userId: string, message: any) {
    const room = this.rooms.get(roomId);
    if (!room) return;

    room.forEach((client, id) => {
      if (id !== userId && client.ws.readyState === WebSocket.OPEN) {
        client.ws.send(JSON.stringify(message));
      }
    });
  }
}

import { WebSocketServer } from "ws";
import { RoomManager } from "./room";
import { ISignalMessage } from "./types";

const roomManager = new RoomManager();

export function initWebSocket(server: any) {
  const wss = new WebSocketServer({ server });

  wss.on("connection", (ws) => {
    let currentUserId = "";
    let currentRoomId = "";

    ws.on("message", (msg) => {
      const message: ISignalMessage = JSON.parse(msg.toString());

      const { type, roomId, userId, targetId, data } = message;

      switch (type) {
        case "join":
          currentUserId = userId;
          currentRoomId = roomId;

          roomManager.join(roomId, { userId, ws });

          // 通知房间其他人
          roomManager.broadcast(roomId, userId, {
            type: "join",
            userId,
          });
          break;

        case "offer":
        case "answer":
        case "candidate":
          if (!targetId) return;

          roomManager.sendTo(roomId, targetId, {
            type,
            userId,
            data,
          });
          break;

        case "leave":
          roomManager.leave(roomId, userId);

          roomManager.broadcast(roomId, userId, {
            type: "leave",
            userId,
          });
          break;
      }
    });

    ws.on("close", () => {
      if (currentRoomId && currentUserId) {
        roomManager.leave(currentRoomId, currentUserId);

        roomManager.broadcast(currentRoomId, currentUserId, {
          type: "leave",
          userId: currentUserId,
        });
      }
    });
  });
}

import type { DeviceState } from "./types";

export interface IDevice {
  id: string;
  state: DeviceState;
}

export class DeviceManager {
  private devices = new Map<string, IDevice>();

  update(id: string, state: DeviceState) {
    const dev = this.devices.get(id) || { id, state };
    dev.state = state;
    this.devices.set(id, dev);
  }

  remove(id: string) {
    this.devices.delete(id);
  }

  list() {
    return Array.from(this.devices.values());
  }
}

import { type IRTCParams } from "./types";

export class WebRTCPlayer {
  private peer: RTCPeerConnection;
  public stream: MediaStream;

  constructor() {
    this.peer = new RTCPeerConnection();
    this.stream = new MediaStream();

    this.peer.ontrack = (event) => {
      this.stream.addTrack(event.track);
    };
  }

  async play(params: IRTCParams) {
    this.peer.addTransceiver("audio", { direction: "recvonly" });
    this.peer.addTransceiver("video", { direction: "recvonly" });

    const offer = await this.peer.createOffer();
    await this.peer.setLocalDescription(offer);

    const res = await fetch(params.url, {
      method: "POST",
      body: JSON.stringify({
        sdp: offer.sdp,
        streamurl: params.streamurl,
      }),
    });

    const data = await res.json();

    await this.peer.setRemoteDescription({
      type: "answer",
      sdp: data.sdp,
    });
  }

  close() {
    this.stream.getTracks().forEach((t) => t.stop());
    this.peer.close();
  }
}

import { type IRTCParams } from "./types";

export class WebRTCPublisher {
  private peer: RTCPeerConnection;
  private stream: MediaStream | null = null;

  constructor() {
    this.peer = new RTCPeerConnection();
  }

  async publish(params: IRTCParams) {
    this.stream = await navigator.mediaDevices.getUserMedia({
      video: true,
      audio: true,
    });

    this.stream.getTracks().forEach((track) => {
      this.peer.addTrack(track);
    });

    const offer = await this.peer.createOffer();
    await this.peer.setLocalDescription(offer);

    const res = await fetch(params.url, {
      method: "POST",
      body: JSON.stringify({
        sdp: offer.sdp,
        streamurl: params.streamurl,
      }),
    });

    const data = await res.json();

    await this.peer.setRemoteDescription({
      type: "answer",
      sdp: data.sdp,
    });
  }

  getStream() {
    return this.stream;
  }

  close() {
    this.stream?.getTracks().forEach((t) => t.stop());
    this.peer.close();
  }
}

export interface ISignalMessage {
  event?: "join" | "leave" | "publish";
  display: string;
  peer?: {
    display: string;
    publishing: boolean;
  };
  participants?: any[];
}

export class SignalClient {
  private ws: WebSocket | null = null;
  private url: string;
  private onMessage: (msg: ISignalMessage) => void;

  constructor(url: string, onMessage: (msg: ISignalMessage) => void) {
    this.url = url;
    this.onMessage = onMessage;
  }

  connect() {
    return new Promise<void>((resolve) => {
      this.ws = new WebSocket(this.url);

      this.ws.onopen = () => {
        console.log("WebSocket 已连接");
        resolve();
      };

      this.ws.onmessage = (e) => {
        try {
          const data = JSON.parse(e.data);
          this.onMessage(data);
        } catch (err) {
          console.error("WS 解析失败", err);
        }
      };

      this.ws.onclose = () => {
        console.warn("WebSocket 断开，准备重连...");
        setTimeout(() => this.connect(), 2000);
      };
    });
  }

  send(data: any) {
    this.ws?.send(JSON.stringify(data));
  }

  close() {
    this.ws?.close();
  }
}

import { WebRTCPlayer } from "./player";

export class StreamManager {
  private players = new Map<string, WebRTCPlayer>();

  async play(
    id: string,
    video: HTMLVideoElement,
    url: string,
    streamurl: string
  ) {
    this.stop(id);

    const player = new WebRTCPlayer();
    await player.play({ url, streamurl });

    video.srcObject = player.stream;
    this.players.set(id, player);
  }

  stop(id: string) {
    const p = this.players.get(id);
    if (p) {
      p.close();
      this.players.delete(id);
    }
  }

  stopAll() {
    this.players.forEach((p) => p.close());
    this.players.clear();
  }
}

import { DeviceManager } from "./device-manager";
import { SignalClient } from "./signal-client";
import { StreamManager } from "./stream-manager";

export function useRTCVideo(host: string, room: string, display: string) {
  const deviceManager = new DeviceManager();
  const streamManager = new StreamManager();

  const devices = ref<any[]>([]);
  const videoRefs = new Map<string, HTMLVideoElement>();

  let signal: SignalClient;

  function setVideoRef(id: string, el: HTMLVideoElement | null) {
    if (el) {
      videoRefs.set(id, el);
    }
  }

  function handleSignal(msg: any) {
    console.log("收到信令:", msg);

    // 1️⃣ publish 事件
    if (msg.event === "publish") {
      const peer = msg.peer;

      if (peer?.publishing && peer.display !== display) {
        deviceManager.update(peer.display, "publishing");

        const video = videoRefs.get(peer.display);
        if (video) {
          streamManager.play(
            peer.display,
            video,
            `http://${host}:1985/rtc/v1/play/`,
            `webrtc://${host}/${room}/${peer.display}`
          );
        }
      }
    }

    // 2️⃣ join / leave
    if (msg.event === "join" || msg.event === "leave") {
      msg.participants?.forEach((p: any) => {
        if (p.display === display) {
          return;
        }

        deviceManager.update(p.display, p.publishing ? "publishing" : "online");
      });
    }

    // 3️⃣ 初始 participants
    if (msg.participants) {
      msg.participants.forEach((p: any) => {
        if (p.display === display) {
          return;
        }

        if (p.publishing) {
          deviceManager.update(p.display, "publishing");

          const video = videoRefs.get(p.display);
          if (video) {
            streamManager.play(
              p.display,
              video,
              `http://${host}:1985/rtc/v1/play/`,
              `webrtc://${host}/${room}/${p.display}`
            );
          }
        }
      });
    }

    devices.value = deviceManager.list();
  }

  // 初始化
  async function init() {
    const wsUrl = `ws://${host}/sig/v1/rtc?room=${room}&display=${display}`;

    signal = new SignalClient(wsUrl, handleSignal);

    await signal.connect();

    // join 房间
    signal.send({
      action: "join",
      room,
      display,
    });

    // 告诉别人我在推流（可选）
    signal.send({
      action: "publish",
      room,
      display,
    });
  }

  function destroy() {
    signal?.close();
    streamManager.stopAll();
  }

  onBeforeUnmount(destroy);

  return {
    devices,
    setVideoRef,
    init,
  };
}

// 视频流推送
function webRTCPublish() {
  this.peer = new RTCPeerConnection(null);
  this.stream = new MediaStream();

  this.publish = async (params) => {
    const { url, streamurl } = params;

    this.peer.addTransceiver("audio", { direction: "sendonly" });
    this.peer.addTransceiver("video", { direction: "sendonly" });

    const stream = await navigator.mediaDevices.getUserMedia({
      audio: true,
      video: {
        width: { ideal: 320, max: 576 },
      },
    });

    stream.getTracks().forEach((track) => {
      this.peer.addTrack(track);
      this.stream.addTrack(track);
    });

    const offer = await this.peer.createOffer();
    await this.peer.setLocalDescription(offer);

    const answer = await fetch(url, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        streamurl,
        sdp: offer.sdp,
        api: url,
        tid: slGenUid(8),
        clientip: null,
      }),
    }).then((response) => {
      if (!response.ok) {
        throw new Error(`HTTP error! Status: ${response.status}`);
      }
      return response.json();
    });

    await this.peer.setRemoteDescription(
      new RTCSessionDescription({ type: "answer", sdp: answer.sdp })
    );
    console.log("---推送视频成功---");
  };

  this.close = function () {
    console.log("---关闭webRTCPublish---");

    this.peer && this.peer.close();
    this.peer = null;
  };
}

export default function useWebRTCVideo(pageParams) {
  const { host, room, display } = config.socket.video;

  const handleStartPlay = pageParams?.handleStartPlay
    ? pageParams.handleStartPlay
    : () => {};
  const handleJoinAndLeave = pageParams?.handleJoinAndLeave
    ? pageParams.handleJoinAndLeave
    : () => {};

  const socket = ref({
    ws: null,
    config: {
      host,
      room,
      display,
    },
  });
  async function socketInit() {
    console.log(
      "摄像头websocket地址：",
      `ws://${host}/sig/v1/rtc?room=${room}&display=${display}`
    );

    socket.value.ws = new slPubWebSocket({
      url: `ws://${host}/sig/v1/rtc?room=${room}&display=${display}`,

      onMessage: (message) => {
        const msg = message.msg;

        console.log(msg);

        if (msg) {
          if (Object.hasOwnProperty.call(msg, "event")) {
            const event = msg.event;

            if (event === "publish") {
              if (
                msg.peer &&
                msg.peer.publishing &&
                msg.peer.display !== display
              ) {
                // 开始播放
                handleStartPlay(msg.peer.display);
              }
            }

            if (event === "join" || event === "leave") {
              // 处理 join 和 leave
              handleJoinAndLeave({
                data: msg.participants,
                devceId: display,
              });
            }
          } else {
            if (msg.participants) {
              msg.participants.forEach((ele) => {
                if (ele.display === display || !ele.publishing) return;
                // 开始播放
                handleStartPlay(ele.display);
              });
            }
          }
        }
      },
      // 异常断开重新连接
      onErrorClose: () => {
        console.log("摄像头 socket 断开了");
      },
    });
  }

  // 断开连接
  function webRTCVideoClose() {
    console.log("关闭 socket");
    socket.value.ws && socket.value.ws.close();
  }

  let publisher = null;

  // 开始推送流
  function webRTCStartPublish() {
    if (publisher) {
      publisher.close();
    }
    publisher = new webRTCPublish();
    publisher.publish({
      url: `http://${host}:1985/rtc/v1/publish/`,
      streamurl: `webrtc://${host}/${room}/${display}`,
    });
    console.log("推送请求发出了");
  }

  // webRTC 初始化
  async function webRTCVideoInit(deviceCloud) {
    socketInit();

    const ws = socket.value.ws;
    await ws.connect();

    await ws.send({
      action: "join",
      room: room,
      display: display,
    });

    await webRTCStartPublish();

    await ws.send({
      action: "publish",
      room: room,
      display: display,
    });

    if (deviceCloud) {
      deviceCloud();
    }
  }

  // 播放视频流
  function webRTCVideoPlay(params) {
    // streamId  视频流ID
    // videoId 播放器ID
    const { streamId, videoId } = params;
    const player = new webRTCPlayer();

    player.play({
      url: `http://${host}:1985/rtc/v1/play/`,
      streamurl: `webrtc://${host}/${room}/${streamId}`,
    });

    const video = document.getElementById(videoId);
    video.srcObject = player.stream;

    return player;
  }

  // 播放摄像头
  function webRTCVideoPlayDeviceCloud(params) {
    // streamId  视频流ID
    // videoId 播放器ID
    const { streamId, videoId } = params;
    const player = new webRTCPlayer();

    console.log("videoId:", videoId);

    player.play({
      url: `http://${host}:1985/rtc/v1/play/`,
      streamurl: `webrtc://${host}/${room}/${streamId}`,
      // url: `http://192.168.1.61:1985/rtc/v1/play/`,
      // streamurl: `webrtc://192.168.1.61/live/zr30`
    });

    const video = document.getElementById(videoId);
    video.srcObject = player.stream;

    return player;
  }

  onBeforeUnmount(() => {
    console.log("离开本页面啦-------------------");
    publisher && publisher.close();
    webRTCVideoClose();
  });

  return {
    webRTCVideoInit,
    webRTCVideoPlay,
    webRTCVideoPlayDeviceCloud,
    webRTCVideoClose,
  };
}

import useWebRTCVideo from "./webRTC-video.js";

export default function useVideo() {
  const { proxy } = getCurrentInstance();
  const { $pub, $api } = proxy;

  const { webRTCVideoInit, webRTCVideoPlay, webRTCVideoPlayDeviceCloud } =
    useWebRTCVideo({
      handleStartPlay: handleStartPlay,
      handleJoinAndLeave: handleJoinAndLeave,
    });

  const videoContentId = $pub.slGenUid();
  const videoDivId = $pub.slGenUid();

  // 视频播放对象
  const videoProps = ref({
    player: null,
    // 视频原大小
    style: {
      width: null,
      height: null,
    },
  });

  // 当前是否是全屏
  const isCurrentFullscreen = ref(false);

  // 全屏切换
  function videoFullScreenToggle() {
    isCurrentFullscreen.value = !isCurrentFullscreen.value;
  }

  // 有设备加入 开始播放视频
  function handleStartPlay(id) {
    emits("onJoin", id);
  }

  // 设备离开
  function handleJoinAndLeave(params) {
    console.log("有设备离开了");
    emits("onLeave", params);
  }

  // 播放流视频
  function videoPlay(id) {
    console.log("id:", id);

    const player = webRTCVideoPlay({
      streamId: id,
      videoId,
    });

    emits("onPlay", {
      id,
      player,
    });
  }
  // 关闭视频
  function videoClose() {
    console.log("videoClose");

    if (videoProps.value.player) {
      console.log("关闭当前视频");

      videoProps.value.player.close();
      videoProps.value.player = null;
    }
  }

  // 一体机摄像头初始化
  function cameraVideoDataInit() {
    webRTCVideoInit(() => {
      // 获取摄像头 ID
      videoProps.value.player = webRTCVideoPlayDeviceCloud({
        streamId: res[0].DeviceNo,
        videoId,
      });
    });
  }

  // 计算视频播放窗口大小
  function handleVideoWindow() {
    const videoContent = document.getElementById(videoContentId);
    const videoDiv = document.getElementById(videoDivId);

    const style = videoProps.value.style;
    const videoWidth = style.width;
    const videoHeight = style.height;

    if (videoContent) {
      let videoParentWidth = videoContent.offsetWidth;
      let videoParentHeight = videoContent.offsetHeight;

      if (isCurrentFullscreen.value) {
        // 全屏
        videoParentWidth = window.innerWidth;
        videoParentHeight = window.innerHeight;
      }

      if ((videoWidth * videoParentHeight) / videoParentWidth < videoHeight) {
        // 视频高度超出了
        videoDiv.style.width = `${
          (videoWidth * videoParentHeight) / videoHeight
        }px`;
      }
    }
  }

  onMounted(() => {
    nextTick(() => {
      const video = document.getElementById(videoId);

      // 当视频尺寸信息加载完成后
      video.addEventListener("loadedmetadata", function () {
        videoProps.value.style = {
          width: video.videoWidth,
          height: video.videoHeight,
        };

        handleVideoWindow();
      });
    });
  });

  onBeforeUnmount(() => {
    console.log("关闭所有视频流，关闭视频socket");

    if (videoProps.value.player) {
      videoProps.value.player.close();
    }
  });

  // 监听 全屏状态 动态计算播放器窗口大小
  watch(
    () => isCurrentFullscreen.value,
    () => {
      nextTick(() => {
        handleVideoWindow();
      });
    }
  );

  return {};
}
