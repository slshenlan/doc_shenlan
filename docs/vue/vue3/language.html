js
import { catchLanguage } from '@/lib/js/cache';
import { createI18n } from 'vue-i18n';
import config from './config';

import zh from './lang/zh-CN.json';
import en from './lang/en-US.json';
import ja from './lang/ja-JP.json';

import iviewZh from 'view-ui-plus/dist/locale/zh-CN';
import iviewEn from 'view-ui-plus/dist/locale/en-US';
import iviewJa from 'view-ui-plus/dist/locale/ja-JP';

export default createI18n({
  locale: catchLanguage.get(),
  legacy: false,
  globalInjection: true, // 全局注册 $t
  // allowComposition: true,
  messages: {
    [config.key.zh]: { ...zh, ...iviewZh },
    [config.key.en]: { ...en, ...iviewEn },
    [config.key.ja]: { ...ja, ...iviewJa }
  }
});
